\section{Probability Spaces}

\begin{definition}{Probability Spaces}{}
    A \dfntxt{probability space} is a measure space \(  \left(  \Omega ,\mathcal{F},P \right)   \) with \[
    P: \mathcal{F}\to \left[ 0,1 \right] 
    \] 
\end{definition}

\begin{example}{Discrete probability spaces}{}
    Let \(   \Omega   \) be a countable set .  Let \(  \mathcal{F}  \) be the collection of all subsets of \(   \Omega   \) .  Let \[
    P\left( A \right)= \sum _{ \omega \in A} p\left(  \omega  \right)  ,\quad \text{where } p\left(  \omega  \right)\ge 0 \text{ and } \sum _{ \omega \in  \Omega }p\left(  \omega  \right)= 1    
    \]   
\end{example}


\begin{example}{Measures on the real line.}{}
Measures on $(\mathbf{R}, \mathcal{R})$ are defined by giving probability a \dfntxt{Stieltjes measure function} with the following properties:
\begin{enumerate}
    \item[(i)] $F$ is nondecreasing.
    \item[(ii)] $F$ is right continuous, that is, $\lim_{y \downarrow x} F(y) = F(x)$.
\end{enumerate}
\end{example}

\begin{theorem}{}{}
Associated with each Stieltjes measure function F there is a unique measure $\mu$ on $(\mathbf{R}, \mathcal{R})$ with $\mu(a, b]) = F(b) - F(a)$
\begin{equation}
    \mu((a, b]) = F(b) - F(a) 
\end{equation}
When $F(x) = x$ the resulting measure is called \dfntxt{Lebesgue measure}
\end{theorem}

\begin{definition}{Semialgebra}{}

A collection $\mathcal{S}$ of sets is said to be a \dfntxt{semialgebra} if (i) it is closed under intersection, that is, $S, T \in \mathcal{S}$ implies $S \cap T \in \mathcal{S}$, and (ii) if $S \in \mathcal{S}$ then $S^c$ is a finite disjoint union of sets in $\mathcal{S}$.

\end{definition}

\begin{definition}{Algebra}{}
    A collection $\mathcal{A}$ of subsets of $\Omega$ is called an \dfntxt{algebra} (or \dfntxt{field}) if $A, B \in \mathcal{A}$ implies $A^c$ and $A \cup B$ are in $\mathcal{A}$.
\end{definition}

\begin{remark}
     Since $A \cap B = (A^c \cup B^c)^c$, it follows that $A \cap B \in \mathcal{A}$. Obviously a $\sigma$-algebra is an algebra. 
\end{remark}

\begin{lemma}{}{}
If $S$ is a semialgebra, then $\bar{S} = \{\text{finite disjoint unions of sets in } S\}$ is an algebra, called the \dfntxt{algebra generated by S}.
\end{lemma}
