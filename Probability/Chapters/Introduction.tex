\section{Probability Spaces}

\begin{definition}{Probability Spaces}{}
    A \dfntxt{probability space} is a measure space \(  \left(  \Omega ,\mathcal{F},P \right)   \) with \[
    P: \mathcal{F}\to \left[ 0,1 \right] 
    \] 
\end{definition}

\begin{example}{Discrete probability spaces}{}
    Let \(   \Omega   \) be a countable set .  Let \(  \mathcal{F}  \) be the collection of all subsets of \(   \Omega   \) .  Let \[
    P\left( A \right)= \sum _{ \omega \in A} p\left(  \omega  \right)  ,\quad \text{where } p\left(  \omega  \right)\ge 0 \text{ and } \sum _{ \omega \in  \Omega }p\left(  \omega  \right)= 1    
    \]   
\end{example}


\begin{example}{Measures on the real line.}{}
Measures on $(\mathbf{R}, \mathcal{R})$ are defined by giving probability a \dfntxt{Stieltjes measure function} with the following properties:
\begin{enumerate}
    \item[(i)] $F$ is nondecreasing.
    \item[(ii)] $F$ is right continuous, that is, $\lim_{y \downarrow x} F(y) = F(x)$.
\end{enumerate}
\end{example}

\begin{theorem}{}{}
Associated with each Stieltjes measure function F there is a unique measure $\mu$ on $(\mathbf{R}, \mathcal{R})$ with $\mu(a, b]) = F(b) - F(a)$
\begin{equation}
    \mu((a, b]) = F(b) - F(a) 
\end{equation}
When $F(x) = x$ the resulting measure is called \dfntxt{Lebesgue measure}
\end{theorem}

\begin{remark}
    The proof is long and winding. We will show the main idea following. 
\end{remark}

\begin{definition}{Semialgebra}{}

A collection $\mathcal{S}$ of sets is said to be a \dfntxt{semialgebra} if (i) it is closed under intersection, that is, $S, T \in \mathcal{S}$ implies $S \cap T \in \mathcal{S}$, and (ii) if $S \in \mathcal{S}$ then $S^c$ is a finite disjoint union of sets in $\mathcal{S}$.

\end{definition}
\begin{example}{}{}
$S_d = $ the empty set plus all sets of the form
\[
(a_1, b_1] \times \cdots \times (a_d, b_d] \subset \mathbf{R}^d \quad \text{where } -\infty \le a_i < b_i \le \infty
\]
Then \(  S_{d}  \) is a esmialgebra. 
\end{example}
\begin{remark}
    The above Stieljes measure function gives the values of \(  \mu   \) on the semialgebra.  
\end{remark}
\begin{definition}{Algebra}{}
    A collection $\mathcal{A}$ of subsets of $\Omega$ is called an \dfntxt{algebra} (or \dfntxt{field}) if $A, B \in \mathcal{A}$ implies $A^c$ and $A \cup B$ are in $\mathcal{A}$.
\end{definition}

\begin{remark}
     Since $A \cap B = (A^c \cup B^c)^c$, it follows that $A \cap B \in \mathcal{A}$. Obviously a $\sigma$-algebra is an algebra. 
\end{remark}

\begin{lemma}{}{}
If $\mathcal{S}$ is a semialgebra, then $\bar{\mathcal{S}} = \{\text{finite disjoint unions of sets in } \mathcal{S}\}$ is an algebra, called the \dfntxt{algebra generated by \(  \mathcal{S}  \) }.
\end{lemma}

\begin{remark}
    Given a set function \(  \mu   \) on \(  \mathcal{S}  \) , we can extend it to \(  \bar{\mathcal{S}}  \) by \[
    \mu \left( + _{i= 1}^{n}A_{i} \right)= \sum _{i = 1}^{n}\mu \left( A_{i} \right)  
    \]   
\end{remark}

\begin{proof}
    Suppose \(  A = + _{i} \mathcal{S}_{i}  \) and \(  B= + _{j} T_{j}  \) , where \(  +   \) denots disjoint union and we assume the index sets are finite. Then \(  A\cup B=  + _{i,j} \mathcal{S}_{i}\cap T_{j}  \) , where \(  \mathcal{S}_{j}\cap T_{j} \in \mathcal{S}  \) by definition.      Thus \(  \bar{\mathcal{S}}  \) is closed under intersection.  As for complements, if \(  A= + _{i}\mathcal{S}_{i}  \) , then \(  A^{c}= \cap _{i}\mathcal{S}_{i}^{c}  \), where \(  \mathcal{S}_{i}^{c}  \) is a finite disjoint union of sets in \(  \mathcal{S}  \) , thus is in \(  \bar{\mathcal{S}}  \) .  Since \(  \bar{\mathcal{S}}  \) is closed under intersection, then \(  A^{c} \in \bar{\mathcal{S}}  \) by induction.        
\end{proof}

\begin{definition}{measure on an algebra}{}
    By a \dfntxt{measure on an algebra} $\mathcal{A}$, we mean a set function $\mu$ with
\begin{enumerate}
    \item[(i)] $\mu(A) \geq \mu(\emptyset) = 0$ for all $A \in \mathcal{A}$, and
    \item[(ii)] if $A_i \in \mathcal{A}$ are \dfntxt{disjoint} and \dfntxt{their union is in} $\mathcal{A}$, then
\end{enumerate}
\[
\mu \left(\bigcup_{i=1}^{\infty} A_i\right) = \sum_{i=1}^{\infty} \mu(A_i)
\]

\end{definition}
\begin{definition}{\(   \sigma   \)-finite }{}
   $\mu$ is said to be \dfntxt{σ-finite} if there is a sequence of sets $A_n \in \mathcal{A}$ so that $\mu(A_n) < \infty$ and $\cup_n A_n = \Omega$. 
\end{definition}
\begin{remark}
    Letting $A'_1 = A_1$ and for $n \geq 2$,
\[
A'_n = \bigcup_{m=1}^n A_m \quad \text{or} \quad A'_n = A_n \cap \left(\bigcap_{m=1}^{n-1} A_m^c\right) \in \mathcal{A}
\]
we can without loss of generality assume that $A_n \uparrow \Omega$ or the $A_n$ are disjoint.

\end{remark}

\begin{theorem}{}{}
Let $\mathcal{S}$ be a semialgebra and let $\mu$ defined on $\mathcal{S}$ have $\mu(\emptyset) = 0$. Suppose (i) if $S \in \mathcal{S}$ is a finite disjoint union of sets $S_i \in \mathcal{S}$ then $\mu(S) = \sum_i \mu(S_i)$, and (ii) if $S_i, S \in \mathcal{S}$ with $S = +_{i \geq 1} S_i$ then $\mu(S) \leq \sum_{i \geq 1} \mu(S_i)$. Then $\mu$ has a unique extension $\bar{\mu}$ that is a measure on $\bar{\mathcal{S}}$ the algebra generated by $\mathcal{S}$. If $\bar{\mu}$ is sigma-finite then there is a unique extension $\nu$ that is a measure on $\sigma(\mathcal{S})$.
\end{theorem}

\begin{lemma}{}{}
    Suppose only that (i) holds. 
    \begin{enumerate}
        \item If \(  A,B_{i} \in \bar{\mathcal{S}}  \)  with \(  A= + _{i= 1}^{n}B_{i}  \) , then \(  \bar{\mu}\left( A \right)= \sum _{i} \bar{\mu}\left( B_{i} \right)    \). 
        \item If \(  A,B_{i} \in \bar{\mathcal{S}}  \)    with \(  A\subseteq \bigcup _{i = 1}^{n} B_{i}  \) then \(  \bar{\mu}\left( A \right)\le \sum _{i} \bar{\mu}\left( B_{i} \right)    \)  
    \end{enumerate}
    
\end{lemma}

\subsection{Exersice}

\begin{exercise}{}{}
A set $A \subset \{1, 2, \dots\}$ is said to have \dfntxt{asymptotic density} $\theta$ if
\[
\lim_{n\to\infty} |A \cap \{1, 2, \dots, n\}|/n = \theta
\]
Let $\mathcal{A}$ be the collection of sets for which the asymptotic density exists. Is $\mathcal{A}$ a $\sigma$-algebra? an algebra?
\end{exercise}
\begin{proof}
    Suppose that \(  A\in \mathcal{A}  \) has esymptotic density \(   \theta   \) .  \[
    \left| A\cap \left\{ 1,2,\cdots ,n \right\} \right|/n+ \left| A^{c}\cap \left\{ 1,2,\cdots ,n \right\} \right|/n= 1  
    \]  Take the limit of \(  n\to \infty  \) ,we have \[
    \lim_{n\to \infty} \left| A\cap \left\{ 1,2,\cdots ,n \right\} \right|/n + \lim_{n\to \infty} \left| A^{c}\cap \left\{  1,\cdots,n  \right\} \right|/n = 1  
    \] which implies that \(  A^{c}  \) has the asymptotic density \(  1- \theta   \) .  
    
    Let \(  A = \left\{ 2k \right\} _{k= 1}^{\infty} \) .  Let \(  B   \) contained all the even numbers among \(  \left[ 2^{2m}, 2^{2m+ 1}-1 \right]   \)  , and contained all the odd numbers among \(  \left[ 2^{2m+ 1}, 2^{2m+ 2}-1 \right]   \)   . Since there are as many even numbers as odd numbers among \(  \left[ 2^{k},2^{k}-1 \right]   \) . It is not hard to see \(  B  \) has asymptotic density \(  \frac{1}{2}  \).  Then \[
   \begin{aligned}
    \lim_{m\to \infty} \left| \left( A\cap B \right) \cap \left\{ 1,2,\cdots ,2^{2m+ 1} \right\}  \right|/2^{2m+ 1}&=\lim_{m\to \infty}\sum _{k = 0}^{m} \frac{1 }{2 }\left( 2^{2k+ 1}-2^{2k} \right)/2^{2m+ 1}\\ 
     &= \lim_{m\to \infty} \sum _{k = 0}^{m} 4^{k-m-1}\\ 
      &= \lim_{m\to \infty} 4^{-m-1} \frac{1-4^{m+ 1} }{ 1-4}= \frac{1 }{3 }  
   \end{aligned}
    \]   \[
    \begin{aligned}
    \lim_{m\to \infty} \left| \left( A\cap B \right) \cap \left\{ 1,2,\cdots ,2^{2m+ 2} \right\} \right|/ 2^{2m+ 2} &=  \lim_{m\to \infty}  \sum _{k = 0}^{m} \frac{1}{2}\left( 2^{2m+ 1}-2^{2m} \right)/2^{2m+ 2} = \frac{1}{6}
    \end{aligned}
    \] which implies that  \(  A\cap B  \) is not in \(  \mathcal{A}  \) . But \(  \mathcal{A}  \) is closed under complements ,  \(  \mathcal{A}  \) is not closed under finite union. Thus \(  \mathcal{A}  \) is not an algebra, then is neither a \(   \sigma   \)- algebra.         
\end{proof}

\section{Distributions}

\begin{definition}{}{}
    A real-valued function \(  X  \) defined on \(   \Omega   \) is said to be a \dfntxt{random variable} if for every Borel set \(  B\subseteq \mathbb{R}   \) we have \(  X ^{-1} \left( B \right) \in \mathcal{F}   \). That is  a  real valued measurable  function on \(  \mathcal{F}  \) .  We write \(  X \in \mathcal{F}  \) .  
\end{definition}
\begin{example}{}{}
 \begin{enumerate}
    \item If \(   \Omega    \) is a discrete probability space , then any function \(  X: \Omega \to \mathbb{R}   \) is a random variable.   
    \item For any \(  A\in \mathcal{F}  \),the \dfntxt{indicator  function} of  \(  A  \) is:  \[
    1 _{A}\left(  \omega  \right)= \begin{cases} 1,&  \omega \in A\\ 
     0,& \omega \not\in A \end{cases}  
    \]
 \end{enumerate}
 
\end{example}

\begin{definition}{distribution}{}
    \begin{itemize}
        \item If \(  X  \) is a random variable, then \(  X  \) induces a probability measure on \(  \mathbb{R}   \) called \dfntxt{distribution} by setting \(  \mu \left( A \right)= P\left( X \in A \right)    \) for Borel sets \(  A  \).   
        \item The distribution of a random variable \(  X  \) is usually described by giving its \dfntxt{distribution function}, \[
        F\left( x \right)= P\left( X\le x \right)  
        \] 
    \end{itemize}
       
\end{definition}
\begin{remark}
    The right-hand side can be written as \(  P\left( X ^{-1} \left( A \right)  \right)   \) . In words, we pull \(  A\in \mathbb{R}   \) back to \(  X^{-1} \left( A \right) \in \mathcal{F}   \) and then take \(  P  \) of that set.    
\end{remark}
\begin{proof}
    If the \(  A_{i}  \) are disjoint, \[
    \begin{aligned}
    \mu \left( \bigcup _{i}A_{i} \right)  &=  P\left( X \in A \right) \\ 
     &= P\left( \bigcup _{i}\left\{ X \in A_{i} \right\} \right)\\ 
      &= \sum _{i} P\left( X \in A_{i} \right)= \sum _{i}\mu \left( A_{i} \right)   
    \end{aligned}
    \] 
\end{proof}

\begin{theorem}{}{}
Any distribution function $F$ has the following properties:
\begin{enumerate}
\item[(i)] $F$ is \dfntxt{nondecreasing}.
\item[(ii)] $\lim_{x\to\infty} F(x) = 1$, $\lim_{x\to-\infty} F(x) = 0$.
\item[(iii)] $F$ is \dfntxt{right continuous}, that is, $\lim_{y\downarrow x} F(y) = F(x)$.
\item[(iv)] If $F(x-) = \lim_{y\uparrow x} F(y)$ then $F(x-) = P(X < x)$.
\item[(v)] $P(X = x) = F(x) - F(x-)$.
\end{enumerate}
\end{theorem}

\begin{theorem}{}{}
    If \(  F  \) satisfies (i), (ii), and (iii) in the above theorem, then it is the distribution function of some random variable.  
\end{theorem}

\begin{remark}
    Even thoug \(  F  \) may not be 1-1 and onto, we will call \(  X  \) the inverse of \(  F  \) and denote by \(  F^{-1}   \).     
\end{remark}

\begin{proof}
    Let \(  P  \) be Lebesgue measure,  Note that \[
    F\left( x \right)= P\left(  \omega :  \omega \le F\left( x \right)  \right)   
    \] And if \(  X  \) is a random variable which we need , then \[
    F\left( x \right)= P\left(  \omega :X\left(  \omega  \right)\le x   \right)  
    \] Thus we only need to show that \[
    \left\{  \omega :X\left(  \omega  \right)\le x  \right\}= \left\{  \omega : \omega \le F\left( x \right)  \right\}
    \] holds for some \(  X  \).  If \(  F  \) is a bejiction, we may define \[
    X\left(  \omega  \right)= F^{-1} \left(  \omega  \right)  
    \] It motivates us to define \[
    X\left(  \omega  \right)= \sup \left\{ y: F\left( y \right)<  \omega   \right\} 
    \] in a general case. To check that it is just what we need , we observe that if \(   \omega \le F\left( x \right)   \), and if \(  x\le X\left(  \omega  \right)   \) , then \(  F\left( x \right)<  \omega    \), which is a contradiction.Thus \(  x > X\left(  \omega  \right)   \). On the other hand if \(   \omega > F\left( x \right)   \), then since \(  F  \) is right continuous, there is an \(   \varepsilon > 0  \) so that \(  F\left( x+  \varepsilon  \right)<  \omega    \), and \(  X\left(  \omega  \right)\ge x+  \varepsilon > x   \).     
\end{proof}

\begin{definition}{Equal in Distribution}{}
   \begin{itemize}
    \item  If \(  X  \) and \(  Y  \) induce the same distribution \(  \mu   \) on \(  \left( \mathbb{R} ,\mathcal{R} \right)   \), we say \(  X  \) and \(  Y  \) are \dfntxt{equal in distribution}, written by \[
    X = _{d} Y
    \]    
    \item When the distribution function \(  F\left( x \right)= P\left( X\le x \right)    \)  has the form \[
    F\left( x \right)= \int_{-\infty}^{x}f\left( y \right)\,\mathrm{d} y  
    \]We say that \(  X  \) has \dfntxt{density function f}.  We write \(  f_{X}\left( x \right)   \).  
   \end{itemize}
   
\end{definition}
\begin{remark}
   \begin{itemize}
    \item   \(  X  \) and \(  Y  \) are equal in distribution, iff \(  X  \) and \(  Y  \) have the same distribution function, that is , \(  P\left( X\le x \right)= P\left( Y\le x \right)    \) for all \(  x  \).      
    \item In remembering formulas, it is often useful to think of \(  f\left( x \right)   \) as being \(  P\left( X= x \right)   \) alghough \[
    P\left( X= x \right)= F\left( x \right)-F\left( x- \right)= \lim_{ \varepsilon \to 0}\int_{x- \varepsilon }^{x+  \varepsilon }f\left( y \right)\,\mathrm{d} y= 0   
    \]  
   \end{itemize}
   
      
\end{remark}
\begin{example}{Uniform distribution on (0,1)}{}
$f(x) = 1$ for $x \in (0, 1)$ and 0 otherwise. Distribution function:
\[
F(x) = \begin{cases} 
0 & x \le 0 \\
x & 0 \le x \le 1 \\
1 & x > 1 
\end{cases}
\]
\end{example}
 \begin{example}{Exponential distribution with rate $\lambda$}{}
$f(x) = \lambda e^{-\lambda x}$ for $x \geq 0$ and $0$ otherwise. Distribution function:
\[
F(x) = \begin{cases}
    0 & x \leq 0 \\
    1 - e^{- \lambda x} & x \geq 0
\end{cases}
\]
\end{example}
\begin{example}{Standard normal distribution}{}
\[
f(x) = (2\pi)^{-1/2} \exp(-x^2/2)
\]
\end{example}

\begin{remark}
    In this case, there is no closed-form expression for $F(x)$, but we have the following bounds that are useful for large $x$:
\end{remark}

\begin{theorem}{For $x > 0$}{}
\[
(x^{-1} - x^{-3}) \exp(-x^2/2) \leq \int_{x}^{\infty} \exp(-y^2/2)dy \leq x^{-1} \exp(-x^2/2)
\]
\end{theorem}


\section{Random Variables}

\begin{definition}{}{}
    A function \(  X: \Omega \to S  \) is said to be a \dfntxt{measurable map} from \(  \left(  \Omega ,\mathcal{F} \right)\to \left( S,\mathcal{S} \right)    \) if \[
    X^{-1} \left( B \right)\equiv \left\{  \omega :X\left(  \omega  \right)\in B  \right\}\in \mathcal{F} \quad \text{for all } B\in \mathcal{S} 
    \]  If \(  \left( S,\mathcal{S} \right)= \left( \mathbb{R} ^{d},\mathcal{R}^{d} \right)    \) and \(  d> 1  \), then \(  X  \) is called a \dfntxt{random vector}. If \(  d= 1  \), \(  X  \) is called a \dfntxt{random variable}, ro r.v.for short.      
\end{definition}

\begin{theorem}{}{}
    If \(  \left\{  \omega :X\left(  \omega  \right)\in A  \right\} \in \mathcal{F} \) for all \(  A\in \mathcal{A}  \) and \(  \mathcal{A}  \) \dfntxt{generates} \(  \mathcal{S}  \) , then \(  X  \) is measurable.      
\end{theorem}

\begin{definition}{}{8-1-1}
    Suppose that \(  X  \) is a measurable map. Then we define \[
     \sigma \left( X \right):= \left\{ \left\{ X \in B \right\}:B\in \mathcal{S} \right\} 
    \] as the  \dfntxt{\(   \sigma   \)-field generated by \(  X  \)  }. 
\end{definition}
\begin{remark}
    where we write \( \left\{ X \in B \right\} \) as shorthand for \(  \left\{  \omega :X\left(  \omega  \right)\in B  \right\}  \).   
\end{remark}

\begin{example}{}{}
    If \(  \left( S,\mathcal{S} \right)= \left( \mathbb{R},\mathcal{R} \right)    \), then possible choices of \(  \mathcal{A}  \) in  \ref{dfn:8-1-1} are \(  \left\{ \left( -\infty,x \right] : x\in \mathbb{R} \right\}  \) or \(  \left\{ \left( -\infty,x \right):x \in \mathbb{Q}   \right\}  \).  
\end{example}

\begin{example}{}{}
    If \(  \left( S,\mathcal{S} \right)= \left( \mathbb{R}^{d},\mathcal{R}^{d} \right)    \), a useful choice of \(  \mathcal{A}  \) is \[
    \left\{ \left( a_1,b_1 \right)\times \cdots \times \left( a_{d},b_{d} \right):-\infty< a_{i}< b_{i}< \infty   \right\}
    \]  
\end{example}
\begin{theorem}{}
If $X : (\Omega, \mathcal{F}) \to (S, \mathcal{S})$ and $f : (S, \mathcal{S}) \to (T, \mathcal{T})$ are measurable maps, then $f(X)$ is a measurable map from $(\Omega, \mathcal{F})$ to $(T, \mathcal{T})$
\end{theorem}
\begin{theorem}{}
If $X_1, \dots, X_n$ are random variables and $f : (\mathbf{R}^n, \mathcal{R}^n) \to (\mathbf{R}, \mathcal{R})$ is measurable, then $f(X_1, \dots, X_n)$ is a random variable.
\end{theorem}
\begin{theorem}{}
If $X_1,\dots,X_n$ are random variables then $X_1+\dots+X_n$ is a random variable.
\end{theorem}
\begin{theorem}{}
If $X_1, X_2, \dots$ are random variables then so are
\[ \inf_n X_n \qquad \sup_n X_n \qquad \limsup_n X_n \qquad \liminf_n X_n \]
\end{theorem}

\section{Expected Value}

\begin{definition}{Expected Value}{}
    \begin{itemize}
        \item If \(  X\ge 0  \) is a random variable on \(  \left(  \Omega ,\mathcal{F},P \right)   \), then we define its \dfntxt{expected value}to be \[
        EX= \int X\,\mathrm{d} P
        \] We declare that \(  EX  \) \dfntxt{exists} and set \[
        EX = EX^{+ }-EX^{-}
        \]whenever the subtraction makes sense.    
    \end{itemize}
    
\end{definition}
\begin{remark}
    \(  EX  \) is often called the \dfntxt{mean} of \(  X  \) and denoted by \(  \mu   \) .    
\end{remark}
\begin{theorem}{}{}
Suppose $X, Y \ge 0$ or $E|X|, E|Y| < \infty$.
\begin{enumerate}
    \item[(a)] $E(X + Y) = EX + EY$.
    \item[(b)] $E(aX + b) = aE(X) + b$ for any real numbers $a, b$.
    \item[(c)] If $X \ge Y$ then $EX \ge EY$.
\end{enumerate}
\end{theorem}

\subsection{Inequalities}

\begin{theorem}{Jensen}{}
    Suppose \(   \varphi   \) is \dfntxt{convex}. Then \[
    E\left(  \varphi \left( X \right)  \right)\ge  \varphi \left( EX \right)  
    \] provided both expectations exists, that is , \(  E\left| X \right|    \) and \(  E\left|  \varphi \left( X \right)  \right|< \infty   \)   
\end{theorem}

\begin{theorem}{Holder}{}
    If \(  p,q\in \left[ 1,\infty \right]   \) with \(  \frac{1}{p}+ \frac{1}{q}= 1  \) , then \[
    E\left| XY \right|\le \left\| X \right\|_{p}\left\| Y \right\|_{q} 
    \] Here \(  \left\| X \right\|_{r}= \left( E\left| X \right|^{r}  \right)^{\frac{1}{r}}   \)    for \(  r\in \left[ 1,\infty \right)   \); \(  \left\| X \right\|_{\infty}= \inf \left\{ M:P\left( \left| X \right|> M  \right)  = 0\right\}  \)  
\end{theorem}

\subsection{Integration to the Limit}
\begin{theorem}{Fatou's lemma}{}
If $X_n \ge 0$ then
\[
\liminf_{n\to\infty} E X_n \ge E\left(\liminf_{n\to\infty} X_n\right)
\]
\end{theorem}

\begin{theorem}{Monotone convergence theorem}{}
If $0 \le X_n \uparrow X$ then $EX_n \uparrow EX$.
\end{theorem}

\begin{theorem}{Dominated convergence theorem}{}
If $X_n \to X \text{ a.s.}$, $|X_n| \le Y$ for all $n$, and $EY < \infty$, then $EX_n \to EX$.
\end{theorem}

\subsection{Computing Expected Values}
\begin{theorem}{Change of variables formula}{}
Let $X$ be a random element of $(S, \mathcal{S})$ with distribution $\mu$, that is, $\mu(A) = P(X \in A)$. If $f$ is a measurable function from $(S, \mathcal{S})$ to $(\mathbb{R}, \mathcal{R})$ so that $f \geq 0$ or $E|f(X)| < \infty$, then
\[ Ef(X) = \int_S f \,\mathrm{d} \mu = : \int_{S}f\left( y \right)\mu \left( \,\mathrm{d} y \right)   \]
\end{theorem}



\begin{definition}{}{}
  \begin{itemize}
    \item If \(  k  \) is a positive integer, then \(  EX^{k}  \) is called the \dfntxt{kth moment} of \(  X  \). The first moment \(  EX   \) is usually called the \dfntxt{mean} and denoted by \(  \mu   \). 
    \item If \(  EX^{2}< \infty  \) , then the \dfntxt{variance} of \(  X  \) is defined to be \[
    \begin{aligned}
    \operatorname{var}\left( X \right)&=  E\left( X-\mu  \right)^{2}\\ 
     &=    EX^{2}-\mu ^{2}
    \end{aligned}
    \]
  \end{itemize}
  
\end{definition}

\begin{proposition}{}{}
    \begin{itemize}
        \item   \(
    \mathrm{var}\left( X \right)\le EX^{2} 
    \)
    \item \(
    \operatorname{var}\left( aX+ b \right)= a^{2}\mathrm{var}\left( X \right)  
    \)
    \end{itemize}
    
\end{proposition}

\begin{example}{}{}
    If \(  X  \) has the \dfntxt{exponential distribution} with rate 1, then \[
    EX^{k}= \int_{0}^{\infty}x^{k}e^{-x}\,\mathrm{d} x= k!
    \] 
\end{example}
\begin{proof}
    \[
    EX^{k}= \int_{S}y^{k}\,\mathrm{d} \mu =  \int_{S}y^{k}\,\mathrm{d} F= \int_{\mathbb{R} }x^{k}f\left( x \right)\,\mathrm{d} x=  \int_{0}^{\infty}x^{k} e^{-x}\,\mathrm{d} x
    \] \[
    \begin{aligned}
    \int_{0}^{\infty}x^{k}e^{-x}\,\mathrm{d} x=  -\int_{0}^{\infty}x^{k}\,\mathrm{d} e^{-x}&= -\left[ e^{-x}x^{k} \right] _{0}^{\infty}+  k \int_{0}^{\infty}x^{k-1}e^{-x}\,\mathrm{d} x \\ 
     &=  k \int_{0}^{\infty}x^{k-1}e^{-x}\,\mathrm{d} x
    \end{aligned}
    \] Then it follows from induction.
\end{proof}
