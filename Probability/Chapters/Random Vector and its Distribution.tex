\section{The additivity of some distributions}

\begin{example}{The additivity of Possion Distribution}{}
    Suppose that \(  X \sim P\left(  \lambda _1  \right)   \), \(  Y\sim P\left(  \lambda _2  \right)   \), and \(  X,Y  \) are independent. Then \[
    Z= X+ Y\sim P\left(  \lambda _1 +  \lambda _2  \right) 
    \]   
\end{example}

\begin{example}{The additivity of Binomal Distribution}{}
    Suppose that \(  X\sim b\left( n,p \right)   \), \(  Y\sim b\left( m,p \right)   \) and \(  X,Y  \) are independent. Then \[
    Z= X+ Y\sim b\left( n+ m,p \right) 
    \]   
\end{example}

\begin{example}{the additivity of Nomal Distribution}{}
    Suppose that \(  X\sim N\left( \mu _1 , \sigma _1 ^{2} \right)   \) , \(  Y\sim N\left( \mu _2 , \sigma _2 ^{2} \right)   \) and \(  X,Y  \) are independent . Then \[
    Z= X+ Y\sim N\left( \mu _1 + \mu _2 , \sigma _1 ^{2}+  \sigma _2 ^{2} \right) 
    \]  
\end{example}

\begin{example}{the additivity of Gamma Distribution}{}
    Suppose that \(  X\sim Ga\left( \alpha _1 , \lambda  \right)   \), \(  Y_2\sim Ga\left( \alpha _2 , \lambda \right)   \), and \(  X,Y  \) are independent. Then \[
    Z= X+ Y\sim Ga\left( \alpha _1 + \alpha _2 , \lambda  \right) 
    \]   
\end{example}
