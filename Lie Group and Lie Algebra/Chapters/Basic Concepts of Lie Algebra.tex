\chapter{ Basic Concepts of Lie Algebra }

\section{Definitions and first examples}

\begin{definition}{}{}
A vector space $L$ over a field $\mathbb{F}$, with an operation $L \times L \to L$, denoted $(x, y) \mapsto [xy]$ and called the \dfntxt{bracket} or \dfntxt{commutator} of $x$ and $y$, is called a \dfntxt{Lie algebra} over $\mathbb{F}$ if the following axioms are satisfied:
\begin{enumerate}
\item[(L1)] The bracket operation is bilinear.
\item[(L2)] $[xx] = 0$ for all $x \in L$.
\item[(L3)] $[x[yz]]+[y[zx]]+[z[xy]] = 0 \quad (x, y, z \in L)$.
\end{enumerate}
\end{definition}

\begin{remark}
   \begin{itemize}
    \item  (L1) and (L2), applied to \(  \left[ x+ y, x+ y \right]   \) imply anticommutativity : (L2\(  ^{\prime}   \)) \(  \left[ xy \right]= \left[ -yx \right]    \). 
    \item Conversely, if \(  \operatorname{char}\mathbb{F}\neq 2  \), it is clear that (L2\(  ^{\prime}   \) ) will imply (L2). 
   \end{itemize}
     
\end{remark}

\begin{definition}{}{}
   We  define a \dfntxt{Lie subalgebra} of $L$ to be a vector subspace $K \subseteq L$ such that
$$[x, y] \in K \quad \text{for all } x, y \in K.$$
\end{definition}

\begin{example}{}{}
   Suppose that $V$ is a finite-dimensional vector space over $F$. Write $\mathfrak{gl}(V)$ for the set of all linear maps from $V$ to $V$. This is again a vector space over $F$, and it becomes a Lie algebra, known as the \dfntxt{general linear algebra}, if we define the Lie bracket $[-,-]$ by

\[
[x,y] := x \circ y - y \circ x \quad \text{for } x, y \in \mathfrak{gl}(V),
\]

where $\circ$ denotes the composition of maps.
\end{example}

\section{Lie algebras of derivations}

\begin{definition}{}{}
   \begin{itemize}
    \item   By an \dfntxt{F-algebra} (not necessarily associative) we simply mean a vector space $\mathfrak{A}$ over F endowed with a bilinear operation $\mathfrak{A} \times \mathfrak{A} \to \mathfrak{A}$, usually denoted by juxtaposition (unless $\mathfrak{A}$ is a Lie algebra, in which case we always use the bracket).
    \item  By a \dfntxt{derivation} of $\mathfrak{A}$ we mean a linear map $\delta: \mathfrak{A} \to \mathfrak{A}$ satisfying the familiar product rule $\delta(ab) = a\delta(b) + \delta(a)b$ for each \(  n\in \mathbb{Z}   \) . 
   \end{itemize}
   
\end{definition}

\section{Ideals and homomorphism}

\subsection{Ideals}
\begin{definition}{}{}
   A subspace $I$ of a Lie algebra $L$ is called an \dfntxt{ideal} of $L$ if $x \in L, y \in I$ together imply $[xy] \in I$.
\end{definition}

\subsection{Homomorphisms and representations}
\begin{definition}{}{}
 \begin{itemize}
   \item   A linear transformation $\phi: L \to L'$ ($L, L'$ Lie algebras over F) is called a \dfntxt{homomorphism} if $\phi([xy]) = [\phi(x)\phi(y)]$, for all $x, y \in L$.
   \item $\phi$ is called a \dfntxt{monomorphism} if $\mathrm{Ker} \phi = 0$, an \dfntxt{epimorphism} if $\mathrm{Im} \phi = L'$, an \dfntxt{isomorphism} if it is both mono- and epi-. 
   
 \end{itemize}
 
\end{definition}

\begin{definition}{}{}
   If $L_1$ and $L_2$ are Lie algebras over a field $F$, then we say that a map $\varphi : L_1 \to L_2$ is a \dfntxt{homomorphism} if $\varphi$ is a linear map and

\[
\varphi([x,y]) = [\varphi(x), \varphi(y)] \quad \text{for all } x, y \in L_1.
\]

Notice that in this equation the first Lie bracket is taken in $L_1$ and the second Lie bracket is taken in $L_2$. We say that $\varphi$ is an \dfntxt{isomorphism} if $\varphi$ is also bijective.
\end{definition}

\begin{example}{}{}
     An extremely important homomorphism is the \dfntxt{adjoint homomorphism}. If $L$ is a Lie algebra, we define
\[
\operatorname{ad}: L \to \mathfrak{gl}(L)
\]
by $(\operatorname{ad} x)(y) := [x, y]$ for $x, y \in L$. It follows from the bilinearity of the Lie bracket that the map $\operatorname{ad} x$ is linear for each $x \in L$. For the same reason, the map $x \mapsto \operatorname{ad} x$ is itself linear. So to show that ad is a homomorphism, all we need to check is that
\[
\operatorname{ad}([x, y]) = \operatorname{ad} x \circ \operatorname{ad} y - \operatorname{ad} y \circ \operatorname{ad} x \quad \text{for all } x, y \in L;
\]

this turns out to be equivalent to the Jacobi identity. The kernel of ad is the centre of $L$.

\end{example}
\begin{proof}
    We have from Jacobi identity that \[
    \begin{aligned}
    \mathrm{ad}\left( \left[ x,y \right]  \right)\left( z \right)&=  \left[ \left[ x,y \right],z  \right]\\ 
     &= \left[ x,\left[ y,z \right]  \right]+ \left[ y,\left[ z,x \right]  \right]      \\ 
      &=\left[ x, \operatorname{ad}y\left( z \right)  \right]-\left[ y, \operatorname{ad}x\left( z \right)  \right]\\ 
       &= \operatorname{ad}x\circ \operatorname{ad}y\left( z \right)-\operatorname{ad}y\circ \operatorname{ad}x\left( z \right)      
    \end{aligned}
    \]
\end{proof}
