\chapter{Measure Theory}

\section{Outer Measure}
\begin{definition}{Outer Measure}{}
A function $\varphi$ defined for every subset $A$ of an arbitrary set $X$ is called an \dfntxt{outer measure} on $X$ if the following conditions are satisfied:
\begin{enumerate}
    \item[(i)] $\varphi(\emptyset) = 0$,
    \item[(ii)] $0 \leq \varphi(A) \leq \infty$ whenever $A \subset X$,
    \item[(iii)] $\varphi(A_1) \leq \varphi(A_2)$ whenever $A_1 \subset A_2$,
    \item[(iv)] $\varphi(\bigcup_{i=1}^{\infty} A_i) \leq \sum_{i=1}^{\infty} \varphi(A_i)$ for every countable collection of sets $\{A_i\}$ in $X$.
\end{enumerate}
\end{definition}

\begin{example}{Outer Measure}{}
\begin{enumerate}
\item[(i)] In an arbitrary set $X$, define $\varphi(A) = 1$ if $A$ is nonempty and $\varphi(\emptyset) = 0$.
\item[(ii)] Let $\varphi(A)$ be the number (possibly infinite) of points in $A$.
\item[(iii)] Let $\varphi(A)=\left\{\begin{array}{ll} 0 & \text { if card } A \leq \aleph_{0}, \\ 1 & \text { if card } A > \aleph_{0} . \end{array}\right.$
\item[(iv)] If $X$ is a metric space, fix $\varepsilon > 0$. Let $\varphi(A)$ be the smallest number of balls of radius $\varepsilon$ that cover $A$.
\item[(v)] Select a fixed $x_0$ in an arbitrary set $X$, and let
\[
\varphi(A)=\left\{\begin{array}{ll}
0 & \text { if } x_{0} \notin A, \\
1 & \text { if } x_{0} \in A;
\end{array}\right.
\]
$\varphi$ is called the \dfntxt{Dirac measure} concentrated at $x_0$.
\end{enumerate}
\end{example}

\begin{definition}{}{}
Let $\varphi$ be an outer measure on a set $X$. A set $E \subset X$ is called \dfntxt{$\varphi$-measurable} if
\[
\varphi(A) = \varphi(A \cap E) + \varphi(A - E)
\]
for every set $A \subset X$. In view of property (iv) above, observe that $\varphi$-measurability requires only
\[
\varphi(A) \ge \varphi(A \cap E) + \varphi(A - E).
\]
\end{definition}

\begin{note}
    也就是说, \(  E  \)是可测的, 当且仅当任意集合 \(  A  \)都可以被 \(  E  \)按外测度被良好地拆解为 \(  A\cap E  \)和 \(  A\setminus E  \).      
\end{note}

\begin{lemma}{}{7-23-1}
A set $E \subset X$ is $\varphi$-measurable if and only if
\[
\varphi(P \cup Q) = \varphi(P) + \varphi(Q)
\]
for all sets $P$ and $Q$ such that $P \subset E$ and $Q \subset E^{c}$.
\end{lemma}

\begin{note}
    也就是说 \(  E  \)可测, 当且仅当 落在\(  E  \)的内, 外的集合 在外测度意义下是无关的. 
\end{note}

\begin{proof}
   For Necessity, if \(  E   \) is \(   \varphi   \)-measurable , then for all such \(  P,Q  \),  \[
     \varphi \left( P\cup Q \right)=  \varphi \left( \left( P\cup Q \right)\cap E  \right) +  \varphi \left( \left( P\cup Q \right)-E  \right)=  \varphi \left( P \right)+  \varphi \left( Q \right)     
    \]  

    For sufficiency ,  let \(  P= A\cap E  \) , \(  Q= A\cap E^{c}= A\setminus  E  \) .     Then \[
       \varphi \left( A \right)=  \varphi \left( P\cup Q \right)=   \varphi \left( P \right)+  \varphi \left( Q \right)    =\varphi \left( A\cap E \right)+  \varphi \left( A\setminus E \right)  
    \]
    
    
\end{proof}

\begin{theorem}{}{7-24-1}
Suppose $\varphi$ is an outer measure on an arbitrary set $X$.
Then the following four statements hold:
\begin{enumerate}
\item[(i)] $E$ is $\varphi$-measurable whenever $\varphi(E) = 0$.
\item[(ii)] $\emptyset$ and $X$ are $\varphi$-measurable.
\item[(iii)] $E_1 - E_2$ is $\varphi$-measurable whenever $E_1$ and $E_2$ are $\varphi$-measurable.
\item[(iv)] If $\{E_i\}$ is a \dfntxt{countable collection of disjoint $\varphi$-measurable sets}, then $\bigcup_{i=1}^\infty E_i$ is $\varphi$-measurable and
\[
\varphi\left(\bigcup_{i=1}^\infty E_i\right) = \sum_{i=1}^\infty \varphi(E_i).
\]
\end{enumerate}
More generally, if $A \subset X$ is an arbitrary set, then
\[
\varphi(A) = \sum_{i=1}^\infty \varphi(A \cap E_i) + \varphi(A \cap S^{c}),
\]
where
\[
S = \bigcup_{i=1}^\infty E_i.
\]
\end{theorem}
\begin{proof}
    \begin{enumerate}
        \item Take any \(  A\subseteq X  \) . Since \(  A\cap E\subseteq E  \) , \(   \varphi \left( A\cap E \right)= 0   \) .  Thus \[
         \varphi \left( A\cap E \right)+   \varphi \left( A\cap E^{c} \right)=  \varphi \left( A\cap E^{c} \right)\le  \varphi \left( A \right)    
        \]   
        \item Follows immediately frome lemma \ref{lem:7-23-1}.
        \item Take any \(  P,Q  \) such that \(  P\subseteq \left( E_1\setminus E_2 \right) = E_1\cap  E_2^{c}  \) , \(  Q\subseteq \left( E_1\setminus E_2 \right)^{c}= E_1^{c}\cup E_2   \) .   \[
        \begin{aligned}
         \varphi \left( Q \right)&=  \varphi \left( Q\cap E_2 \right)+  \varphi \left( Q-  E_2 \right)
        \end{aligned}   
        \] Note that \(  Q-E_2 \subseteq E_1^{c}  \) , \(  Q\cap E_2\subseteq E_2  \) . Since \(  P\subseteq E_1  \) , the measurability of \(  E_1  \) implis that  \[
         \varphi \left( P \right)+  \varphi \left( Q-E_2 \right)=  \varphi \left( P\cup \left( Q-E_2 \right)  \right)   
        \]  Note that \(  P\cup \left( Q-E_2 \right)\subseteq E_2^{c}   \)  . Since \(  Q\cap E_2\subseteq E_2  \) , we have \[
         \varphi \left( P\cup \left( Q-E_2 \right) \right)+  \varphi \left( Q\cap E_2 \right)=  \varphi \left( P\cup Q \right)   
        \]  Finally , we have \[
        \begin{aligned}
         \varphi \left( P \right)+  \varphi \left( Q \right)&=   \varphi \left( P \right)+  \varphi \left( Q\cap E_2 \right)+  \varphi \left( Q- E_2 \right)\\ 
          &=        \varphi \left( P\cup \left( Q-E_2 \right)  \right)+  \varphi \left( Q\cap E_2 \right)\\ 
           &=  \varphi \left( P\cup Q \right)    
        \end{aligned}
        \]
        \item  Let \(  S_{k} = \bigcup _{i= 1}^{k}E_{i}  \).  We proceed by finite induction and first note that the result is obviously true for \(  k= 1  \) . For \(  k> 1  \) assume that \(  S_{k}  \) is \(   \varphi   \)-measurable and that \[
         \varphi \left( A \right)\ge  \sum _{i = 1}^{k} \varphi \left( A\cap E_{i} \right)+  \varphi \left( A\cap S_{k}^{c} \right)   ,\quad \forall A\subseteq X
        \]   Then \[
        \begin{aligned}
         \varphi \left( A \right)&=  \varphi \left( A\cap E_{k+ 1} \right)+  \varphi \left( A\cap E_{k+ 1}^{c} \right)\\ 
          &=  \varphi \left( A\cap E_{k+ 1} \right)+  \varphi \left( A\cap E_{k+ 1}^{c}\cap S_{k}^{c} \right)+  \varphi \left( A\cap E_{k+ 1}^{c}\cap S_{k} \right)\\ 
           &=  \varphi \left( A\cap E_{k+ 1} \right)+  \varphi \left( A\cap S_{k+ 1}^{c} \right)+  \varphi \left( A\cap S_{k} \right)        
        \end{aligned}
        \]  Replace \(  A  \) by \(  A\cap S_{k}  \) , we have \[
         \varphi \left( A\cap S_{k} \right)\ge \sum _{i = 1}^{k} \varphi \left( A\cap S_{k}\cap E_{i} \right)+  \varphi \left( A\cap S_{k}\cap S_{k}^{c} \right)=     \sum _{i = 1}^{k} \varphi \left( A\cap E_{i} \right) 
        \]   Thus \[
         \varphi \left( A \right) \ge \sum _{i = 1}^{k+ 1} \varphi \left( A\cap E_{k+ 1} \right)+  \varphi \left( A\cap S_{k+ 1}^{c} \right)  
        \] Then the subadditivity gives that \[
         \varphi \left( A \right)\ge   \varphi \left( A\cap S_{k+ 1} \right)+  \varphi \left( A\cap S_{k+ 1}^{c} \right)   
        \] this , in turn , implies that \(  S_{k+ 1}  \) is \(   \varphi   \)-measurable. Since we now know  for every set \(  A\subseteq X  \) and for all positive integers \(  k  \) \[
         \varphi \left( A \right)\ge \sum _{i = 1}^{k} \varphi \left( A\cap E_{i} \right)+  \varphi \left( A\cap S_{k}^{c} \right)   
        \]     We have \[
         \varphi \left( A \right)\ge \sum _{i = 1}^{\infty} \varphi \left( A\cap E_{i} \right)+  \varphi \left( A\cap S^{c} \right)  \ge  \varphi \left( A\cap S \right)+  \varphi \left( A\cap S^{c} \right)   
        \] where \(  S= \bigcup _{i = 1}^{\infty}E_{k}  \) .  Thus \(  S  \) is measurable.   The subadditivity gives that \[
         \varphi \left( A \right)\le \sum _{i= 1}^{\infty} \varphi \left( A\cap E_{i} \right)+  \varphi \left( A\cap S^{c} \right)   
        \]Finally, we have \[
         \varphi \left( A \right)= \sum _{i= 1}^{\infty} \varphi \left( A\cap E_{i} \right)+  \varphi \left( A\cap S^{c} \right)   
        \] Replace \(  A  \) by \(  S  \) , we can get \[
         \varphi \left( S \right)= \sum _{i= 1}^{\infty} \varphi \left( E_{i} \right)  
        \]  
    \end{enumerate}
    
\end{proof}

\begin{lemma}{}{}
Let $\{E_i\}$ be a sequence of arbitrary sets. Then there exists a sequence of disjoint sets $\{A_i\}$ such that each $A_i \subset E_i$ and
\[
\bigcup_{i=1}^{\infty} E_i = \bigcup_{i=1}^{\infty} A_i.
\]
If each $E_i$ is $\varphi$-measurable, and so is $A_i$.
\end{lemma}
\begin{proof}
    Let \(  S_{k}= \bigcup _{i= 1}^{k}E_{i}  \) .  Let  \(  A_{k}= S_{k}\setminus S_{k-1} ,k \ge 1  \) , where \(  S_0   = \varnothing\) .  Then \[
    \bigcup _{i = 1}^{\infty}E_{i}= \bigcup _{i = 1}^{\infty}\left( S_{k}\setminus S_{k-1} \right)  = \bigcup _{i = 1}^{\infty}A_{i} .
    \]  If each \(  E_{i}  \) is \(   \varphi   \)-measurable.   \(  E_{i+ 1}\cup E_{i}= \left( E_{i+ 1}-E_{i} \right)\cup E_{i}    \) is measurable follows from the (ii), (iii) of the above theorem . Inductively , every \(  S_{k}=  \left( S_{k-1}\setminus E_{k} \right)\cup E_{k}   \)  is measurable.  Then  each \(  A_{k}   \) is measurable .  
\end{proof}

\begin{theorem}{}{7-24-2}
If $\{E_i\}$ is a sequence of \dfntxt{$\varphi$-measurable} sets in X, then $\cup_{i=1}^{\infty} E_i$ and $\cap_{i=1}^{\infty} E_i$ are \dfntxt{$\varphi$-measurable}.
\end{theorem}

\begin{proof}
    By the lemma above, there exists a sequence of disjoint \(   \varphi   \) - measurable sets \(  \left\{ A_{i} \right\}  \) such that each \(  A_{i}\subseteq E_{i}  \) , and \[
    \bigcup _{i = 1}^{\infty}E_{i}= \bigcup _{i = 1}^{\infty}A_{i}
    \]  It follows from Theorem \ref{thm:7-24-1} that \(  \bigcup _{i = 1}^{\infty}A_{i}  \) is measurable , then \(  \bigcup _{i = 1}^{\infty}E_{i}  \) is as well .   Since \(  X   \) is measurable , \(  E_{i}^{c}= X\setminus E_{i}  \) is measurable , then   \(  \bigcup _{i = 1}^{\infty}E_{i}^{c}  \) is measurable .  Thus \[
    \bigcap_{i= 1}^{\infty}E_{i}=  X\setminus \bigcup _{i= 1}^{\infty}E_{i}^{c} 
    \] is measurable . 
\end{proof}

\begin{definition}{}{}
A nonempty collection $\Sigma$ of sets $E$ satisfying the following two conditions is called a \dfntxt{$\sigma$-algebra}:
\begin{enumerate}
    \item[(i)] if $E \in \Sigma$, then $E^{c} \in \Sigma$;
    \item[(ii)] $\cup_{i=1}^{\infty} E_i \in \Sigma$ if each $E_i$ is in $\Sigma$.
\end{enumerate}
\end{definition}

\begin{definition}{}{}
In a topological space, the elements of the smallest $\sigma$-algebra that contains all open sets are called \dfntxt{Borel sets}. The term ``smallest'' is taken in the sense of inclusion.
\end{definition}
\begin{proof}
    We need to show that the ``smallest'' exists. 
\end{proof}

\begin{corollary}{}{}
If $\varphi$ is an \dfntxt{outer measure} on an arbitrary set $X$, then the class of $\varphi$-measurable sets forms a \dfntxt{$\sigma$-algebra}.
\end{corollary}
\begin{proof}
    Follows immediately from Theorem \ref{thm:7-24-1} and Theorem \ref{thm:7-24-2}. 
\end{proof}

\begin{corollary}{}{8-14-1}
Suppose $\varphi$ is an outer measure on $X$ and $\{E_i\}$ a countable collection of $\varphi$-measurable sets.
\begin{enumerate}
    \item[(i)] If $E_1 \subset E_2$ with $\varphi(E_1) < \infty$, then
    \[
        \varphi(E_2 \setminus E_1) = \varphi(E_2) - \varphi(E_1).
    \]
    \item[(ii)] (Countable additivity) If $\{E_i\}$ is a disjoint sequence of sets, then
    \[
        \varphi(\bigcup_{i=1}^{\infty} E_i) = \sum_{i=1}^{\infty} \varphi(E_i).
    \]
    \item[(iii)] (Continuity from the left) If $\{E_i\}$ is an increasing sequence of sets, that is, if $E_i \subset E_{i+1}$ for each $i$, then
    \[
        \varphi(\bigcup_{i=1}^{\infty} E_i) = \varphi(\lim_{i \to \infty} E_i) = \lim_{i \to \infty} \varphi(E_i).
    \]
    \item[(iv)] (Continuity from the right) If $\{E_i\}$ is a decreasing sequence of sets, that is, if $E_i \supset E_{i+1}$ for each $i$, and if $\varphi(E_{i_0}) < \infty$ for some $i_0$, then
    \[
        \varphi(\bigcap_{i=1}^{\infty} E_i) = \varphi(\lim_{i \to \infty} E_i) = \lim_{i \to \infty} \varphi(E_i).
    \]
    \item[(v)] If $\{E_i\}$ is any sequence of $\varphi$-measurable sets, then
    \[
        \varphi(\liminf_{i \to \infty} E_i) \le \liminf_{i \to \infty} \varphi(E_i).
    \]
    \item[(vi)] If
    \[
        \varphi(\bigcup_{i=i_0}^{\infty} E_i) < \infty
    \]
    for some positive integer $i_0$, then
    \[
        \varphi(\limsup_{i \to \infty} E_i) \ge \limsup_{i \to \infty} \varphi(E_i).
    \]
    
\end{enumerate}
\end{corollary}

\begin{proof}
    (v)  Let \(  A_{j}=  \bigcap_{k= j}^{\infty}E_{k}   \) , then \[
    \liminf_{i\to \infty}E_{i}= \lim_{j\to \infty}A_{j}= \bigcup _{j = 1}^{\infty}A_{j}
    \] It follows from (iv) that \[
     \varphi \left( \liminf_{i\to \infty}E_{i} \right)= \lim_{j\to \infty} \varphi \left( A_{j} \right)  
    \] Since \(  A_{j}\subseteq  E_{j}  \), we have \(   \varphi \left( A_{j} \right)\le  \varphi \left( E_{j} \right)    \), then \[
    \lim_{j\to \infty} \varphi \left( A_{j} \right)\le \liminf_{i\to \infty} \varphi \left( E_{j} \right)  
    \]   Thus \[
     \varphi \left( \liminf_{i\to \infty}E_{i} \right)\le \liminf_{i\to \infty} \varphi \left( E_{i} \right)  
    \]
\end{proof}

\begin{definition}{}{}
    \begin{itemize}
        \item An outer measure \(   \varphi   \) on a topological space \(  X  \) is called a \dfntxt{Borel outer measure} if all Borel sets are \(   \varphi   \)-measurable. 
        \item A Borel outer measure is \dfntxt{finite} if \(   \varphi \left( X \right)    \) is finite.
        \item An outer measure \(   \varphi   \) on a set \(  X  \) is called \dfntxt{regular} if for each \(  A\subseteq X  \) there exists a \(   \varphi   \)-measurable set \(  B\supseteq A  \) such that \(   \varphi \left( B \right)=  \varphi \left( A \right)    \). 
        \item A \dfntxt{Borel rugular outer measure} is a Borel outer measure such that for each \(  A\subseteq X  \), there exists a Borel set \(  B  \) such that \(   \varphi \left( B \right)=  \varphi \left( A \right)    \).    
        \item A \dfntxt{Radon outer measure} is a Borel regular outer measure that is finite on compact sets.             
    \end{itemize}
    
\end{definition}
\section*{Exercises for Outer Measure}

\begin{exercise}{}{}
     If \(  X  \) is a metric space, fix \(   \varepsilon > 0  \). Let \(   \varphi \left( A \right)   \) be the smallest number of balls of radius \(   \varepsilon   \) that cover \(  A  \).      

Let \(   \varphi _{ \varepsilon }\left( A \right): =  \varphi \left( A \right)    \) denote the dependence on \(   \varepsilon   \), and define \[
\psi \left( A \right) : = \lim_{ \varepsilon \to 0 }  \varphi _{ \varepsilon }\left( A \right)  
\] What is \(  \psi \left( A \right)   \)  and what are the correspoding \(  \psi   \)-measurable sets. 
\end{exercise}
\begin{proof}
    If there are only finite points in \(  A  \). Let \[
    l : =  \min \left\{ d\left( x,y \right) :x,y \in A ,x \neq y\right\}
    \] Then for any \(  x,y \in A  \), \(  x \neq y  \). If \(  B_1,B_2  \) are ball of radius \( \frac{l }{2 }   \) such that \(  x\in B_1, y \in B_2  \).     Then \(  B_1\cap B_2= \varnothing  \).  But each point in \(  A  \) must covered by one of such ball, which shows that \(  \psi \left( A \right)\ge  \left| A \right|    \).    Since it is obvious that \(  \psi \left( A \right)\le \left| A \right|    \),  then \(  \psi \left( A \right)= \left| A \right|    \). 
    
    Now suppose that \(  A  \) is a infinite set. If \(  \psi \left( A \right)< \infty   \), say \(  \psi \left( A \right)= n   \) . Then for any sufficiently small \(   \varepsilon   \), there are balls \(  B_1,\cdots ,B_{n}  \)  with radius \(   \varepsilon   \) cover \(  A  \). Take any \(  n+ 1  \) different points \(  p_1,\cdots ,p_{n+ 1}  \)  of \(  A  \). Let  \(   \varepsilon = \frac{1}{2} \min \left\{ d\left( p_{i},p_{j} \right): i\neq j  \right\}  \), then \(   B_1,\cdots,B_n   \) cannot cover these \(  n+ 1  \) points, thus cannot cover  \(  A  \), which is a contradiction. Thus we have \(  \psi \left( A \right)= \infty   \) . 

    We find that \(  \psi   \) is just the counting measure on \(  X  \).Take any \(  E\subseteq X  \),and take any \(  A\subseteq X  \), if \(  A  \) is finite, then \(  A\cap E, A- E  \) are as well, and \(  \left| A \right|= \left| A\cap E \right|+ \left| A-E \right|     \)   , that is \(  \psi \left( A \right)= \psi \left( A\cap E \right)+ \psi \left( A-E \right)     \).  If \(  A  \) is infinite, then one of the \(  A\cap E  \)   and \(  A-E  \) is as well. Then \[
    \psi \left( A \right)= \infty= \psi \left( A\cap E \right)+ \psi \left( A-E \right)   
    \]The above shows that \(  E  \) is \(  \psi   \)-measurable. Thus all subset of \(  X  \) are \(  \psi   \)-measurable.        
\end{proof}

\section{Carath\'eodory Outer Measure}

\begin{definition}{Carath\'eodory outer measure}{}
    An outer measure \(   \varphi  \) defined on a metric space \(  \left( X,p \right)   \) is called a \dfntxt{Carath\'eodory outer measure} if \[
     \varphi \left( A\cup B \right)=  \varphi \left( A \right)+  \varphi \left( B \right)   
    \] whenever \(  A,B  \) are arbitrary subsets of \(  X  \) with \(  d\left( A,B \right)> 0   \). Where \[
    d\left( A,B \right): = \inf \left\{ \rho \left( a,b \right): a \in A, b\in B  \right\} 
    \]     
\end{definition}

\begin{theorem}{}{8-10-1}
    If \(   \varphi   \) is a Carath\'eodory outer measure on a metric space \(  X  \), then all closed sets are \(   \varphi   \)-measurable.    
\end{theorem}
\begin{proofsketch}
    用一列集合近似代替\(  C  \)是标准的想法. 证明的核心要素有: 1.\(  C  \)是闭集, 使得按距离向外逼近的行为是良好的. 2. 距离这个属性与Carath\'odory条件是相适应的.   此外, 将命题转化为极限 \[
    \lim_{i \to \infty}  \varphi \left( A\setminus C_{i} \right)=  \varphi \left( A\setminus C \right)  
    \]是标准的流程. 要证明该极限成立, 则需要一定的技巧. 这里将 \(  A\setminus C_{i}  \)和 \(  A\setminus C  \)相差的中间部分进行细分量化, 将原极限转化为 \[
\lim_{i \to \infty}    \sum _{j = i}^{\infty} \varphi \left( T_{j} \right) 
    \]的极限问题, 从而转化为级数的收敛性问题. 将集合隔一个分组, 变成两组两两相隔正距离的集合, 以便套用Carath\'eodory条件.   
\end{proofsketch}
\begin{proof}
    Let \(  C \subseteq X  \) be any closed sets. Take \(  A \subseteq X  \), only need to show that \[
     \varphi \left( A \right)\ge  \varphi \left( A\cap C \right) +  \varphi \left( A\setminus C \right)   
    \]We assume that \(   \varphi \left( A \right)< \infty   \).  Define \[
    C_{i}: = \left\{ x: d\left( x,C \right)\le \frac{1 }{i }   \right\}
    \] Since \(  d\left( A\setminus C_{i}, A\cap C \right)   > 0\),  \[
     \varphi \left( A \right)\ge   \varphi \left( \left( A\setminus C_{i} \right)\cup \left( A\cap C \right)   \right)=   \varphi \left( A\setminus C_{i} \right)+  \varphi \left( A\cap C \right)    
    \] Only need to show that \[
    \lim_{i \to \infty}  \varphi \left( A\setminus C_{i} \right)=  \varphi \left( A\setminus C \right)  
    \] Define \[
    T_{i}:=  \left\{ x \in A: \frac{1 }{i+ 1 }< d\left( x,C \right)\le \frac{1 }{i }    \right\}
    \], then \[
    A\setminus C\subseteq  \left( A\setminus C_{i} \right)\cup \bigcup _{j = i}^{\infty}T_{i} 
    \] Thus \[
     \varphi \left( A\setminus C \right)\le  \varphi \left( A\setminus C_{i} \right)+ \sum _{j = i}^{\infty} \varphi \left( T_{i} \right)   
    \] Once we have \(  \sum _{j= 1}^{\infty} \varphi \left( T_{i} \right)< \infty   \), then \(  \lim_{i\to \infty}\sum _{j = i}^{\infty} \varphi \left( T_{i} \right)= 0   \),  the conclusion follows from it. To show this, note that \(  \,\mathrm{d} \left( T_{i},T_{j} \right)> 0   \) whenever \(  \left| i-j \right|\ge 2   \). Then it by induction follows from Carath\'odory condition that \[
    \sum _{ j= 1}^{\infty} \varphi \left( T_{2i} \right) \le  \varphi \left( A \right) < \infty
    \] \[
    \sum _{j = 1}^{\infty} \varphi \left( T_{2j-1} \right)\le  \varphi \left( A \right) < \infty 
    \]    Thus \(  \sum _{j = 1}^{\infty} \varphi \left( T_{i} \right)< \infty   \), which completes the proof.  
\end{proof}

\begin{theorem}{}{8-10-2}
Suppose $\mathcal{F}$ is a family of subsets of a topological space X that contains all open and all closed subsets of X. Suppose also that $\mathcal{F}$ is closed under countable unions and countable intersections. Then $\mathcal{F}$ contains all Borel sets; that is, $\mathcal{B} \subset \mathcal{F}$.
\end{theorem}
\begin{note}
    可以将包含\(  \mathcal{B}  \)描述成对以下三种集合的封闭性: 1.开集 2. 补集 3.可数并. 

    如果我们事先没有对补集的封闭性, 可以通过加强1.3.的条件,使得当我们强制使得补集封闭后, 得到的新集族依旧满足1.3. 具体地, 我们希望原集族对于开集的补集,也就是闭集也封闭; 对于可数并的补集, 也就是可数交也封闭. 
\end{note}
\begin{proof}
    Let \[
    \mathcal{H}=  \mathcal{F}\cap \left\{ A:A^{c}\in \mathcal{F} \right\}
    \] Obseve that \(  \mathcal{H}  \) contains all closed sets. Moreover, it is easily seen that \(  \mathcal{H}  \) is closed under complementation and countable unions. Thus \(  \mathcal{H}  \) is a \(   \sigma   \)-algebra that contains the open sets and therefore contains all Borel sets.    
\end{proof}

\begin{theorem}{}{}
    If \(   \varphi   \) is a Carath\'eodory outer measure on a metric space \(  X  \), then the Borel sets of \(  X  \) are \(   \varphi   \)-measurable.    
\end{theorem}
\begin{proof}
    Since the collection of \(   \varphi   \)-measurable sets is itself a \(   \sigma   \)-algebra , it is closed under complements and countable union and countable intersection. Theorem \ref{thm:8-10-1} gives that \(   \varphi   \)-measurable sets contains closed sets, thus contains open sets as well.  Then  the conclusion follows from Theorem \ref{thm:8-10-2} .
\end{proof}

\begin{theorem}{}{}
Suppose each edge $I_k = [a_k, b_k]$ of an $n$-dimensional interval $I$ is partitioned into $\alpha_k$ subintervals. The products of these intervals produce a partition of $I$ into $\beta := \alpha_1 \cdot \alpha_2 \cdots \alpha_n$ subintervals $I_i$ and

\]
\end{theorem}


\begin{theorem}{}{}
For each interval $I$ and each $\epsilon > 0$, there exists an interval $J$ whose interior contains $I$ and
\[

\]
\end{theorem}


\begin{definition}{Lebesgue outer measure}{}
    The \dfntxt{Lebesgue outer measure} of an arbitrary set \(  E\subseteq \mathbb{R} ^{n}  \), denoted by \(   \lambda ^{*}\left( E \right)   \), is defined by \[
     \lambda ^{*}\left( E \right): = \inf \left\{ \sum _{k= 1}^{\infty}v\left( I _{k} \right)  \right\}, 
    \] where the infimum is taken over all countable collections of closed intervals \(  I _{k}  \) such that \[
    E\subseteq \bigcup _{k = 1}^{\infty}I _{k}. 
    \]  
\end{definition}

\begin{remark}
   \begin{itemize}
    \item  It may be necessary at times to emphasize the dimension of the Euclidean space in which Lebesgue outer measure if defined. When clarification is needed, we will write \(   \lambda _{n}^{*}\left( E \right)   \) in plce place of \(   \lambda ^{*}\left( E \right)   \).   
    \item We have not prove that \(  \dfntxt{Lebesgue outer measure}  \) is actually a outer measure till now. 
   \end{itemize}
   
\end{remark}


\begin{theorem}{}{}
    For a closed interval \(  I\subseteq \mathbb{R} ^{n}  \), \(   \lambda ^{*}\left( I \right)= v\left( I \right)    \).   
\end{theorem}

\begin{proof}
    The inequality \(   \lambda ^{*}\left( I \right)\le v\left( I \right)    \) holds, since \(  I  \) itself givs a element of \(  \left\{ \sum _{k = 1}^{\infty}v\left( I _{k} \right)  \right\}  \)   in the definition of \(   \lambda ^{*}\left( I \right)   \). 

    To prove the opposite inequality, choose \(   \varepsilon > 0  \), and a sequence of closed intervals \(  I_1,I_2,\cdots   \) such that \[
    I\subseteq \bigcup _{k = 1}^{\infty}I _{k},\quad \sum _{ k= 1}^{\infty}v\left( I _{k} \right) \le  \lambda ^{*}\left( E \right)+  \varepsilon   
    \]  We can choose closed intervals \(  J_{k}  \) such that \[
    v\left( J_{k} \right)\le v\left( I _{k} \right)+ \frac{ \varepsilon  }{2^{k} },\quad  I _{k}\subseteq J_{k}^{\circ}   
    \]Let \(  \mathcal{F}  : =  \left\{ J_{k}^{\circ}:k \in \mathbb{N}  \right\}\).    Then \(  \mathcal{F}  \) is an open cover of \(  I  \), since \(  I  \) is compact, it has a lebesgue number \(  \eta   \).    There is a finitely many partition \(   K_1,\cdots,K_m   \) of \(  I  \) such that each  with diameter less than \(  \eta   \). Then \(  K_{i}  \) is contained in some  element of \(  \mathcal{F}  \), we say \(  J_{k_{i}}  \)    althoung two \(  K_{i}  \) may be contained in a same \(  J_{k_{i}}  \).  Let \(  N_{m}  \) be the smallest number of \(  J_{k_{i}}  \) containing \(  K_{i}  \), then \[
    v\left( I \right)= \sum _{i= 1}^{m}v\left( K_{i} \right)\le \sum _{i = 1}^{N_{m}}v\left( J_{k_{i}} \right)\le \sum _{ k = 1}^{\infty}v\left( J_{k} \right)    \le   \lambda ^{*}\left( E \right)+ 2 \varepsilon  
    \]     Since \(   \varepsilon   \) is arbitrary, \(  v\left( I \right)\le  \lambda ^{*}\left( E \right)    \).   
\end{proof}

\begin{theorem}{}{8-17-1}
    Lebesgue outer measure, \(   \lambda ^{*}  \), defined on \(  \mathbb{R} ^{n}  \) is a Carath\'eorody outer measure.  
\end{theorem}

\begin{proof}
    We first verify that \(   \lambda ^{*}  \) is an outer measure. We only show the subadditivity of \(   \lambda ^{*}  \), the other conditions are obvious. Let \( \left\{ A_{i} \right\}  \) be a countable collection of arbitrary subsets of \(  \mathbb{R} ^{n}  \). Denote \(  A: = \bigcup _{i = 1}^{\infty}A_{i}  \).       There is a countable family of closed intervals \(  \left\{ I _{j}^{\left( i \right) } \right\}_{j \in \mathbb{N} }  \) such that \[
    A_{i}\subseteq \bigcup _{j = 1}^{\infty} I^{\left( i \right) }_{j},\quad   \sum _{j = 1}^{\infty}v\left( I _{j}^{\left( i \right) } \right)<  \lambda ^{*}\left( A_{i} \right)+ \frac{ \varepsilon  }{2^{i} }   
    \] Collection of closed interval \(  \left\{ I _{j}^{\left( i \right) } \right\}_{i,j\in \mathbb{N} }  \)   can be aranged to a sequence of closed invervals, such that \[
    A\subseteq  \bigcup _{i,j\in \mathbb{N} } I^{\left( i \right) }_{j}
    \] Thus \[
     \lambda ^{*}\left( A \right)\le \sum _{i,j}v\left( I _{j}^{\left( i \right) } \right)\le \sum _{i = 1}^{\infty} \lambda ^{*}\left( A_{i} \right)+  \varepsilon    
    \] Since \(   \varepsilon > 0  \) is arbitrary, the countable subadditivity of \(   \lambda ^{*}  \) is established.      

    To show that \(   \lambda ^{*}  \) is Carath\'eorody outer measure. We take any \(   \varepsilon > 0  \) , and take a countable conllection of closed intervals \(  I_1,I_2,\cdots   \)   such that \[
    A\cup B\subseteq \bigcup _{ k= 1}^{\infty}I _{k},\quad  \sum _{k = 1}^{\infty}v\left( I _{k} \right)\subseteq  \lambda ^{*}\left( A\cup B \right).   
    \] We can subdividing each \(  I _{k}  \) into smaller invervals if necessary, we may assume that each interval \(   I _{k}  \) which diameter less that \(  \mathbf{d}\left( A,B \right)   \). Then \(  \left\{  I _{k} \right\}  \) can be seperated into two subfamiles \(  \left\{ I _{k}^{\prime}  \right\}  \) and \(  \left\{ I _{k}^{\prime \prime}  \right\}  \). The members in the former have empty intersection with \(  B  \) , and that  in the later have empty intersection with \(  A  \). \(  \left\{ I _{k} \right\}= \left\{ I _{k}^{\prime}  \right\} \sqcup \left\{ I _{k}^{\prime \prime}  \right\}  \). Thus \[
    A\subseteq \bigcup _{k = 1}^{\infty}\left\{ I _{k}^{\prime}  \right\},\quad B\subseteq \bigcup _{k= 1}^{\infty} \left\{ I _{k}^{\prime \prime}  \right\}
    \]          We have \[
    \begin{aligned}
     \lambda ^{*}\left( A \right)+  \lambda ^{*}\left( B \right)&\le  \sum _{k= 1}^{\infty}v\left( I _{k}^{\prime}  \right)+ \sum _{k = 1}^{\infty}v\left( I _{k}^{\prime \prime}  \right)     \\ 
      & = \sum _{ k= 1}^{\infty}v\left( I _{k} \right)\\ 
       &<   \lambda ^{*}\left( A\cup B \right)+  \varepsilon   
    \end{aligned}
    \] Since \(   \varepsilon > 0  \) is arbitrary, we have \[
     \lambda ^{*}\left( A \right)+  \lambda ^{*}\left( B \right)\le  \lambda ^{*}\left( A\cup B \right)   
    \] 
\end{proof}

\begin{corollary}{}{}

\end{corollary}

    U : =  \bigcup _{k = 1}^{\infty} \left( I _{k}^{\prime}  \right)^{\circ} 
    \] is a open set.Since \(  U,E  \) is \(   \lambda   \)-measurable,   \[
    \begin{aligned}
     \lambda ^{*}\left( U\setminus E \right)=  \lambda \left( U\setminus E \right)  &=  \lambda \left( U \right)- \lambda \left( E \right)  \\ 
      &=  \lambda \left( U \right)- \lambda ^{*}\left( E \right)\\ 
       &<   \sum _{k = 1}^{\infty} v\left(  I _{k}^{\prime}  \right)- \left( \sum _{k = 1}^{\infty}v\left( I _{k} \right)- \frac{ \varepsilon  }{2 }   \right)\\ 
        &\le  \varepsilon      
    \end{aligned}
    \]
    Now, note that \(  \mathbb{R} ^{n}  \) is \(   \sigma   \)-finite, we suppose that \(  A_1,A_2,\cdots   \) are measruable sets such that \[
    \mathbb{R} ^{n}= \bigcup _{ j = 1}^{\infty}A_{j},\quad  \lambda \left( A_{j} \right)< \infty,\forall j\in \mathbb{Z} _{> 0} 
    \]   Then \(  E_{j}: = E\cap A_{j}  \) is a \(   \lambda ^{*}  \)-measurable set with \(   \lambda \left( E_{j} \right)< \infty   \) for each \(  j  \). For each \(   \varepsilon > 0  \), and each \(  j \in \mathbb{Z} _{> 0}  \), there is an open set \(  U_{j}\supseteq E  \) such that \[
     \lambda ^{*}\left( U_{j}\setminus E_{j} \right)<  \frac{ \varepsilon  }{2^{j} } 
    \] Let \(  U: = \bigcup _{j = 1}^{\infty}U_{j}  \)        ,then \[
     \lambda ^{*}\left( U\setminus E \right)\le \sum _{j= 1}^{\infty} \lambda ^{*}\left( U_{j}\setminus E \right)\le \sum _{j = 1}^{\infty}   \lambda ^{*}\left( U_{j}\setminus E_{j} \right)\le  \varepsilon  
    \] which completes the proof. 
    
    2. \(  \implies   \) 3. : For each \(  n \in \mathbb{Z} _{> 0}  \), there is a open set \(  U_{n}  \), such that \[
     \lambda ^{*}\left( U_{n}\setminus E \right)< \frac{1 }{n }  
    \]Let \[
    U:= \bigcap_{n = 1}^{\infty} U_{n} 
    \]   then \(  U  \) is a \(  G_{ \delta }  \) set, such that \[
     \lambda ^{*}\left( U\setminus E \right)\le  \lambda ^{*}\left( U_{n}\setminus E \right)< \frac{1 }{n },\quad \forall n.   
    \]That is \(   \lambda ^{*}\left( U\setminus E \right)= 0   \). 

    3.\(  \implies   \) 1. : Since \(   \lambda ^{*}  \) is a Carathéorody measure, then all Borel sets is measurable. Specially, the \(  G_{ \delta }  \) set \(  U  \)  is measurable. And since \(   \lambda ^{*}  \)-zero set is measurable,  then \[
  E =  U\setminus \left( U\setminus E \right)  
    \]   is measurable.
    
    2. \(  \implies   \) 4. : There is an open set \(  U\supseteq  E^{c}  \)   such that \(   \lambda ^{*}\left( U\setminus E^{c} \right)<  \varepsilon    \) . Let \(  F = U^{c}  \), then \[
    E\setminus F= E\cap F^{c}=  E\cap U=  U\setminus E^{c} 
    \] Thus \[
     \lambda ^{*}\left( E\setminus F \right)=  \lambda ^{*}\left( U\setminus E^{c} \right)<  \varepsilon   
    \]

    4.\(  \implies   \) 5. : For each \(  n \in \mathbb{Z} _{> 0}  \), there is a closed set \(  F_{n}\subseteq E  \) such taht \[
     \lambda ^{*}\left( E\setminus F_{n} \right)< \frac{1 }{n }  
    \] Let \[
    F: = \bigcup _{n = 1}^{\infty}F_{n} , 
    \] then \(  F  \) is a \(  F_{ \sigma }  \) set, such that \[
     \lambda ^{*}\left( E\setminus F \right)\le  \lambda ^{*}\left( E\setminus F_{n} \right)< \frac{1 }{n },\quad \forall n .   
    \] That is \(   \lambda ^{*}\left( E\setminus F \right)= 0   \).
    
    5. \(  \implies   \) 1. : Since \(  E\setminus F  \) and \(  F  \) is measurable, we have \[
    E = \left( E\setminus F \right)\cup F 
    \]    is measurable.
 \end{proof}

\begin{exercise}{}{}
Let $I, I_1, I_2, \dots, I_k$ be intervals in $\mathbb{R}$ such that $I \subset \bigcup_{i=1}^k I_i$. Prove that
\[
v(I) \le \sum_{i=1}^k v(I_i),
\]
where $v(I)$ denotes the length of the interval $I$.
\end{exercise}

\begin{proof}
   We may assume that \(  v\left( I \right)< \infty   \), that is \(  I  \) is bounded.     Since \(  v\left( I \right)= v\left( \overline{I} \right)    \), \(  v\left( I _{i} \right)=  v\left( \overline{I} _{i} \right)    \) , and \(  \bar{I}\subseteq \bigcup _{ i = 1}^{k} \bar{I}_{i}  \), we may assume that \(  I,I_1,\cdots , I _{k}  \) are closed intervals. For any \(   \varepsilon > 0  \), there are closed intervals \(  I^{\prime} _{i}  \) such that \[
     I _{i}\subseteq \left( I^{\prime} _{i} \right)^{\circ},\quad  v\left( I _{i}^{\prime}  \right)\le v\left( I _{i} \right)+ \frac{ \varepsilon  }{2^{i} }   
     \] Then \(  \left\{\left(  I _{i}^{\prime} \right) ^{\circ}    \right\}\) is a open cover of  \(  I  \).        Since \(  I  \) is compact, it has a Lebesgue number \(  \eta   \).   We can subdivide \(  I  \) into subintervals  \(  K_1,\cdots ,K_{m}  \) with diameter  less that \(  \eta   \). Then each \(K_j\) is contained in some \((I_{i_j}')^\circ\).

We can formalize this by defining a map \(f: \{1, \dots, m\} \to \{1, \dots, k\}\) such that for each \(j\), \(K_j \subset (I_{f(j)}')^\circ\). Since the subintervals \(K_j\) have disjoint interiors and their union is \(I\), we can write the total length of \(I\) and group the sum according to the covering interval index:
\[
v(I) = \sum_{j=1}^m v(K_j) = \sum_{i=1}^k \left( \sum_{j: f(j)=i} v(K_j) \right).
\]
For any fixed \(i\), the intervals \(\{K_j\}_{j: f(j)=i}\) are a collection of disjoint intervals all contained within \((I_i')^\circ\). Therefore, the sum of their lengths cannot exceed the length of \((I_i')^\circ\):
\[
\sum_{j: f(j)=i} v(K_j) = v\left(\bigcup_{j: f(j)=i} K_j\right) \le v((I_i')^\circ) = v(I_i').
\]
Applying this inequality to our grouped sum, we obtain a single chain of inequalities:
\[
v(I) = \sum_{i=1}^k \left( \sum_{j: f(j)=i} v(K_j) \right) \le \sum_{i=1}^k v(I_i') \le \sum_{i=1}^k \left( v(I_i) + \frac{\epsilon}{2^i} \right) = \sum_{i=1}^k v(I_i) + \epsilon \sum_{i=1}^k \frac{1}{2^i}.
\]
Since \(\sum_{i=1}^k \frac{1}{2^i} < 1\), we have \(v(I) < \sum_{i=1}^k v(I_i) + \epsilon\). As \(\epsilon > 0\) was arbitrary, we conclude that
\[
v(I) \le \sum_{i=1}^k v(I_i).
\]

\end{proof}
\begin{exercise}{}{}
Let $E \subset \mathbb{R}$, and for each real number $t$, let $E + t = \{x + t : x \in E\}$.
Prove that $\lambda^*(E) = \lambda^*(E + t)$. From this show that if $E$ is Lebesgue measurable, then so is $E + t$.
\end{exercise}

\begin{proof}

     \lambda ^{*}\left( E+ t\right)\le  \lambda ^{*}\left( E\right),\quad \forall t

    \]Then we get \(   \lambda ^{*}\left( E \right)=  \lambda ^{*}\left( E+ t \right)    \).  To show this, for each \(   \varepsilon > 0  \), we take closed intervals \(  I_1,\cdots   \) such that \[

    \] Then \[
    E+ t\subseteq \bigcup _{ k= 1}^{\infty} \left( I _{k}+ t \right),\quad \sum _{k = 1}^{\infty} v\left( I _{k}+ t \right)= \sum _{k = 1}^{\infty} v\left( I _{k} \right)   <  \lambda \left( E \right)+  \varepsilon  . 
    \]    Thus \[
     \lambda \left( E+ t \right)\le  \lambda \left( E \right)+  \varepsilon   .
    \] Since \(   \varepsilon > 0  \) is arbitrary, \(   \lambda \left( E+ t \right)\le  \lambda \left( E \right)    \). 
    
    To show that \(  E+ t  \) is Lebesgue measurable. There is a \(  G_{ \delta }  \) set \(  G  \), such that  \[
    G\supseteq E,\quad  \lambda ^{*}\left( G\setminus E \right)= 0 .

    \] Then it follows from Theorem \ref{thm:8-12-1} that \(  E+ t  \) is measurable.  
\end{proof}

\section{The Contor Set}

\begin{definition}{Contor set}{8-15-1}
    The Cantor set is a subset of the interval \(  \left[ 0,1 \right]   \). We will desccribe it by constructing its complement in \(  \left[ 0,1 \right]   \). 
    \begin{enumerate}
        \item At the fist step, let \(  I_{1,1}  \) denote the open interval \(  \left( \frac{1}{3},\frac{2}{3} \right)   \). Thus \(  I_{1,1}  \) is the open middle third of the interval \(  I= \left[ 0,1 \right]   \). 
        \item The second step involves performing the first tep on each of the two remaining intervals of \(  I-I_{1,1}  \). That is, We produce two open intervals, \(  I_{2,1}  \) and \(  I_{2,2}  \), each being the open middle third of one of the two intervals that constitute \(  I-I_{1,1}  \).
        \item At the \(  i  \)th step we produce \(  2^{i-1}  \) open intervals \(  I_{i,1},I_{i,2},\cdots ,I_{i,2^{i-1}}  \), each of length \(  \left( \frac{1 }{3 }  \right)^{i}   \). The \(  \left( i+ 1 \right)   \) th step consists in producing midddle thirds of each of the intervals of \[
        I- \bigcup _{j= 1}^{i}\bigcup _{k = 1}^{2^{j-1}}I_{j,k}. 
        \]         
        \item With \(  C  \) denoting the Cantor set, we define its complement by \[
        I\setminus C= \bigcup _{j = 1}^{\infty}\bigcup _{k = 1}^{2^{j-1}}I_{j,k}. 
        \] 
    \end{enumerate}
      
\end{definition}

\begin{proposition}{}{}
    Let \(  C  \) be the Contor set. Then the following holds: 
    \begin{enumerate}
        \item \(  C  \) is closed, that is \(  C = \bar{C}  \). 
        \item Let \(   \lambda   \) be the Lebesgue measure, then \[
         \lambda \left( \bar{C} \right)=  \lambda \left( C \right)= 0  
        \]
        \item \(  C  \) is nowhere dense.    
        \item The cardinality of \(  C  \) is \[
        \left| C \right|= 2^{\aleph_0} = \left| R \right|= c 
        \] 
    \end{enumerate}
     
\end{proposition}

\begin{proof}
    \begin{enumerate}
        \item \(  I\setminus C  \) is open by definition. Thus \(  C  \) is closed.
        \item  \[
         \lambda \left( I\setminus C \right)=  \sum _{j = 1}^{\infty}\sum _{k = 1}^{2^{j-1}}  \lambda \left( I_{j,k} \right)= \sum _{j = 1}^{\infty}2^{j-1}\left( \frac{1 }{3 }  \right)^{j}= \frac{1 }{2 } \sum _{j = 1}^{\infty}\left( \frac{2 }{3 }  \right)^{j}= 1     
        \]  Thus \[
         \lambda \left( C \right)= 1- \lambda \left( I\setminus C \right)= 1-1= 0  
        \]
        \item 
        

        
        \begin{note}
            \(  \left[ 0,1 \right]   \)上的每一个值通过修饰存在唯一的二进制表示. 同样地也存在唯一的三进制表示. Contor集就是把每个三进制表示中, 字符含1的数都去掉.  
        \end{note}Every number \(  x \in \left[ 0,1 \right]   \) has a ternary expansion of the form \[
        x = \sum _{i = 1}^{\infty} \frac{x_{i} }{3^{i} } 
        \]  where each \(  x_{i}  \) is \(  0,1  \), or \(  2  \) and we write \(  x = 0,x_1x_2\cdots   \) This expansion   unique except when \[
        x = \frac{a }{3^{n} } 
        \] (Since \(  \frac{1 }{3^{n} }   \) has another expresion \(  \sum _{i = n+ 1}^{\infty}\frac{2 }{3^{i} }   \)  )where \(  a  \) and \(  n  \) are  positive integers with \(  0< a< 3^{n}  \) and where \(  3  \) does not divide \(  a  \). In this case, \(  x  \) has the form \[
        x = \sum _{i = 1}^{n} \frac{x_{i} }{3^{i} } 
        \]where \(  x_{i}  \) is either \(  1  \) or \(  2  \). If \(  x_{n}= 2  \), we will use this expression to represent \(  x  \). However, if \(  x_{n}= 1  \), we will use the following representation for \(  x  \): \[
        x = \frac{x_1 }{3 }+ \frac{x_2 }{3^{2} }+ \cdots +  \frac{x_{n-1} }{3^{n-1} }+ \frac{0 }{3^{n} }+ \sum _{i = n+ 1}^{\infty}\frac{2 }{3^{i} }     .
        \] Thus , with this convention, each number \(  x \in \left[ 0,1 \right]   \) has a uniue ternary expansion.          
        
        Let \(  x \in I  \) and conssider its ternary expansion \(  x = 0.x_1x_2\cdots   \), bearing in mind the convention we have adopted above. Obseve that \(  x\not \in I_{1,1}  \) if and only if \(  x_1\neq 1  \). Also , if \(  x_1\neq 1  \), then \(  x\not \in  I_{2,1}\cup I_{2,2}  \) if and only if \(  x_2\neq 1  \).       Continuing in this way, we see that \(  x \in C  \) if and only if \(  x_{i}\neq 1  \) for each positive integer \(  i  \). Thus, there is a one-to-one correspondence between elements of  \(  C  \) and all sequences \(  \left\{ x_{i} \right\}   \) in which each \(  x_{i}   \) is either \(  0  \) or \(  2  \).        The cardinality of the latter is \(  2^{\aleph_0}  \), which is \(  \left| \mathbb{R}  \right|= c   \).   
    \end{enumerate}
    
\end{proof}

\begin{lemma}{}{8-16-1}

\end{lemma}
\begin{proofsketch}
    Since \[
    \begin{aligned}
    x \in D_{S}&\iff  S\cap \left( S+ x \right)
     \neq \varnothing\\ &\iff  \text{There exists } T\subseteq S\text{ such that }   T\cap \left(T+ x \right)\neq \varnothing    
    \end{aligned}
    \] Only need to find a such \(  T  \), and a interval \(  I  \), such that \(  T\cap \left( T+ x \right)\neq \varnothing   \) for \(  \forall x \in I  \).
    
    When \(  T\cap \left( T+ x \right)= \varnothing   \), we have \(   \lambda \left( T\cup \left( T+ x \right)  \right)= 2 \lambda \left( T \right)    \)  . Once we find a not too large (less than \(  2 \lambda \left( T \right)   \) ) set containing \(  T\cup \left( T+ x \right)   \) for all \(  x \in I  \).   From another perspective, we need to find an \(  T  \) occupise at least half of a set (relative to Lebesgue measure) and dose not escape it under a tiny transformation. 
\end{proofsketch}
\begin{proof}
    For each \(   \varepsilon > 0  \), there exists an open set \(  U\supseteq S  \) with \(   \lambda \left( U \right)< \left( 1+  \varepsilon  \right) \lambda \left( S \right)     \). We write \(  U  \) as a countable disjoint union of open intervals: \[
    U = \bigcup _{k = 1}^{\infty}I_{k}
    \]Then \[
     \lambda \left( S \right)=  \lambda \left( S\cap U \right)= \sum _{k = 1}^{\infty} \lambda \left( S\cap I_{k} \right)   
    \] We have \[
    \sum _{k = 1}^{\infty} \lambda \left( I_{k} \right) < \left( 1+  \varepsilon  \right) \sum _{k= 1}^{\infty} \lambda \left( S\cap I_{k} \right)   
    \] Then there is some \(   k_0 \) such that \[
     \lambda \left( I_{k} \right)< \left( 1+  \varepsilon  \right) \lambda \left( S\cap I_{k_0} \right)   
    \]     We take \(   \varepsilon = \frac{1 }{3 }   \), then \[
     \lambda \left( S\cap I_{k_0} \right)> \frac{3 }{4 }  \lambda \left( I_{k_0} \right)

    \left( -\frac{1}{2},\frac{1}{2} \right)\subseteq  D_{S\cap I_{k_0}}\subseteq D_{S}

\end{proof}

\section{Lebesgue-Stieljes Measure}

\begin{definition}{}{}
    If \(  f  \) is a nondecreasing function defined on \(  \mathbb{R}   \), then the ``length'' of a half-open interval \(  (a,b]  \), denoted by \(  \alpha _{f}\left( (a,b] \right)   \), can be defined by \[
     \alpha _{f}\left( (a,b] \right)= f\left( b \right)-f\left( a \right)   
    \]    
\end{definition}

\begin{definition}{Lebesgue-Stieltjes outer measure}{}
    The Lebesgue-Stieltjes outer measure of an arbitrary set \(  E\subseteq \mathbb{R}   \) is defined by \[
     \lambda _{f}^{*}\left( E \right): = \inf \left\{ \sum _{h_{k}\in \mathcal{F}}\alpha _{f}\left( h_{k} \right)  \right\} 
    \] where the infimum is take over all countable collections \(  \mathcal{F}  \) of half-open intervals \(  h_{k}  \) of the form \(  (a_{k},b_{k}]  \) such that \[
    E\subseteq \bigcup _{ h _{k}\in \mathcal{F}}h_{k}
    \]   
\end{definition}

\begin{remark}
    If \(  f  \) is right-continuous, then the (Lebesgue) length of each interval \(  (a_{k},b_{k}]  \) that appears above can be assumed to be arbitrarily small, beacause \[
    \alpha _{f}((a,b])= f\left( b \right)-f\left( a \right)= \sum _{k = 1}^{N}\left[ f\left( a_{k} \right)-f\left( a_{k-1} \right)   \right]= \sum _{k = 1}^{N} \alpha _{f}\left( (a_{k-1},a_{k}] \right)    
    \] whenever \(  a= a_0< a_1< \cdots < a_{N}= b  \).  
\end{remark}

\begin{theorem}{}{}
    If \(  f:\mathbb{R} \to \mathbb{R}   \) is a nondecreasing function, then \(   \lambda _{f}^{*}  \) is a Carath\'eodory outer measure.  We denote the restriction of \(   \lambda _{f}^{*}  \) on measruable sets by \(   \lambda _{f}  \).  
\end{theorem}
\begin{proof}
    To show that \(   \lambda _{f}^{*}  \) is a outer measure, we only need to show the it is monotone and countably subadditive. 

    For the proof of monotonicity, let \(  A_1\subseteq A_2  \) be any subset of \(  \mathbb{R}   \). Take any \(   \varepsilon > 0  \). There is a countable collections \(  h_1,h_2,\cdots   \) of \(  \mathcal{F}  \), such that   \[
    A_2\subseteq \bigcup _{k = 1}^{\infty}h_{k},\quad \sum _{k = 1}^{\infty}\alpha _{f}\left( h_{k} \right)<  \lambda ^{*}_{f}\left( A_2 \right)+  \varepsilon   
    \] Then \[
    A_1\subseteq \bigcup _{k = 1}^{\infty}h_{k},\quad \sum _{k = 1}^{\infty}\alpha _{f}\left( h_{k} \right)<  \lambda _{f}^{*}\left( A_2 \right)+  \varepsilon   
    \]That is \[
     \lambda _{f}^{*}\left( A_1 \right) \le  \lambda _{f}^{*}\left( A_2 \right)+  \varepsilon  
    \]Since \(   \varepsilon > 0  \) is arbitrary, \(   \lambda _{f}^{*}\left( A_1 \right)\le  \lambda _{f}^{*}\left( A_2 \right)    \)  .

    The proof of countable subadditivity is virtually identical to the proof the corresponding result of Lebesgue measure given in Theorem  \ref{thm:8-17-1}, since we didn't use any explicit expresion of the length of an interval there.

    The same idea of prooving Lebesgue outer measure to be Carathéodory is effective for that of the Lebesgue-Stieljes oute measure, since the only thing we should check to repeate that proof here is whether we can subdividing each \(  h_{k}  \) into smaller members of \(  \mathcal{F}  \) whose diameter less than \(  \mathbf{d}\left( A,B \right)   \). But that was shown in the remark of the definition.   
\end{proof}


\begin{theorem}{}{}
    If \(  f:\mathbb{R} \to \mathbb{R}   \) is nondecreasing and right-continuous then \[
     \lambda _{f}\left( (a,b] \right) =  f\left( b \right)-f\left( a \right)   
    \] 
\end{theorem}
\begin{note}
    回忆Lebesgue测度的情况, 我们将闭集构成的覆盖, 通过微小的几乎不影响测度的(向两边)扩张, 其内部形成一组开覆盖. 我们是利用这组开覆盖来覆盖身为紧集的闭区间, 以处理问题的.

    这里, 我们由于选择了半开区间, 只需要对右侧进行微小的扩张, 作为代价, 我们研究的集合本身不再是紧集.  这里假设\(  f  \)是右连续, 就同时解决了两个问题: 
    \begin{enumerate}
        \item 向右微小扩张, 得益于右连续, 对测度的影响也变成是无限小的.
        \item 得益于右连续, 用闭集 \(  [a^{\prime} ,b]  \),向左逼近 \(  (a,b], a^{\prime} \to a  ^{+ }\)的行为也是良好的. 
    \end{enumerate}
     
\end{note}
\begin{proof}
    It suffices to show that \[
     \lambda _{f}\left( (a,b] \right)\ge f\left( b \right)-f\left( a \right)   
    \]For each \(   \varepsilon > 0  \), we take a sequence of half-invervals \(  h_1= (a_1,b_1],h_2= (a_2,b_2],\cdots   \) such that \[
    (a,b] \subseteq \bigcup _{k = 1}^{\infty}h_{k},\quad \sum _{k = 1}^{\infty}\alpha _{f}\left( h_{k} \right)<  \lambda _{f}\left( (a,b] \right)+  \varepsilon   .
    \]  Since \(  f  \) is right-continuous , we have \[
    \lim_{t \to 0^{+ }} \alpha _{f}\left( (a_{k},b_{k}+ t] \right)= \alpha _{f}((a_{k},b_{k}])
    \] Consequently, we may replace each \(  (a_{k},b_{k}]  \) with \(  (a_{k},b_{k}^{\prime} ]  \)  such that   \[
    b_{k}< b_{k}^{\prime} ,\quad   \alpha _{f}((a_{k},b_{k}^{\prime} ])< \alpha _{f}((a_{k},b_{k}]) + \frac{ \varepsilon  }{2^{k} } 
    \] 
    
    Let $a' \in (a, b)$. The compact set $[a', b]$ is covered by a finite family of open intervals, which we denote $\mathcal{C}_{fin} = \{ C_j \}_{j=1}^N$, where $C_j = (a_j, b'_j)$.
$[a', b] \subseteq \bigcup_{j=1}^N C_j$
. By the Lebesgue number lemma, there exists an $\eta > 0$ such that any subset of $[a', b]$ with diameter less than $\eta$ is contained in at least one $C_j$. We partition $[a', b]$ into subintervals $I_k = [t_{k-1}, t_k]$ for $k=1, \dots, m$, each with length less than $\eta$. Thus, for each $I_k$, we can choose an interval from $\mathcal{C}_{fin}$ that contains it.

Let's formalize this choice. We define an assignment function $\phi: \{1, \dots, m\} \to \{1, \dots, N\}$ such that for each $k$, $I_k \subseteq C_{\phi(k)}$. 

We can now write the total increase of $f$ over $[a', b]$ and group the terms according to our assignment:
$$f(b) - f(a') = \sum_{k=1}^m (f(t_k) - f(t_{k-1})) = \sum_{j=1}^N \left( \sum_{k \text{ s.t. } \phi(k)=j} (f(t_k) - f(t_{k-1})) \right)$$
The outer sum is over the indices of the open cover $\mathcal{C}_{fin}$, ensuring no open interval is counted more than once.

Now, let's analyze the inner sum for a fixed \( j \). Let \( \mathcal{P}_j = \{ I_k \mid \phi(k)=j \} \) be the set of subintervals assigned to \( C_j \). These are a finite number of disjoint subintervals, all contained within \( C_j = (a_j, b'_j) \). Let \( t_{\min, j} \) be the minimum of all left endpoints of intervals in \( \mathcal{P}_j \), and \( t_{\max, j} \) be the maximum of all right endpoints. Then \( \bigcup_{I_k \in \mathcal{P}_j} I_k \subseteq [t_{\min, j}, t_{\max, j}] \).
Since \( f \) is nondecreasing, the sum of increases over disjoint intervals is less than or equal to the increase over their convex hull:
\[ \sum_{I_k \in \mathcal{P}_j} (f(t_k) - f(t_{k-1})) \le f(t_{\max, j}) - f(t_{\min, j}) \]
Furthermore, since all these \( I_k \) are in \( C_j = (a_j, b'_j) \), we have \( a_j < t_{\min, j} \) and \( t_{\max, j} < b'_j \). The nondecreasing property of \( f \) implies:
\[ f(t_{\max, j}) - f(t_{\min, j}) \le f(b'_j) - f(a_j) \]
Combining these, for each \( j \), we have established:
\[ \sum_{k \text{ s.t. } \phi(k)=j} (f(t_k) - f(t_{k-1})) \le f(b'_j) - f(a_j) \]
Substituting this back into our grouped sum:
\[ f(b) - f(a') \le \sum_{j=1}^N (f(b'_j) - f(a_j)) \]
This sum is over a finite subset of the original infinite cover used to define the \( 2\varepsilon \) bound. Therefore:
\[ \sum_{j=1}^N (f(b'_j) - f(a_j)) \le \sum_{i=1}^\infty (f(b'_i) - f(a_i)) < \lambda_f((a, b]) + 2\varepsilon \]
So we have \( f(b) - f(a') < \lambda_f((a, b]) + 2\varepsilon \).
 

Since \( \varepsilon \) is arbitrary, we have \( f(b) - f(a') \le \lambda_f((a, b]) \).
Furthermore, the right continuity of \( f \) implies \( \lim_{a' \to a^+} f(a') = f(a) \), and hence
\[ f(b) - f(a) \le \lambda_f((a, b]) \]
as desired.
\end{proof}


\begin{theorem}{}{}
Suppose $\mu$ is a finite Borel outer measure on $\mathbb{R}$ and let
\[ f(x) = \mu((-\infty, x]). \]
Then the Lebesgue-Stieltjes measure, $\lambda_f$, agrees with $\mu$ on all Borel sets.
\end{theorem}

\begin{proof}*
  \(  f  \) is nondecreasing from definition.  Note that \[
  (-\infty,x]= \bigcap_{n \in \mathbb{N} }(-\infty,x+ \frac{1 }{n } ]
  \] Since \(  \mu   \) is finite, the continuity from below gives that  \[
  \lim_{n\to \infty}\mu \left( (-\infty,x+ \frac{1 }{n } ] \right) = \mu\left(  (-\infty,x] \right) 
  \]That is \[
  \lim_{n\to \infty}f\left( x+ \frac{1 }{n }  \right)= f\left( x \right)  
  \] Since \(  f  \) is nondecreasing , \[
  \lim_{t \to 0^{+ }} f\left( x+ t \right)= f\left( x \right)  
  \] 

  Since \[
  (-\infty,a]\cup (a,b]= (-\infty,b]
  \] then \[
  f\left( b \right)-f\left( a \right)= \mu ((-\infty,b])-\mu ((-\infty,a])= \mu ((a,b])
  \] thus proving that \(  \mu   \) and \(   \lambda _{f}  \) agree on all half-open intervals.   Since very open set in \(  \mathbb{R}   \) is  a countable union of disjoint half-open intervals, it follows that \(  \mu   \) and \(   \lambda _{f}  \) agree on all open sets.   It will be shown later that \(  \mu , \lambda _{f}  \) agree on all Borel sets . 
\end{proof}

   \lim_{n\to \infty}\left(    f\left( x_0+ \frac{1 }{n }  \right)-f\left( x_0 \right) \right) =  \lambda _{f} \left( \left\{ x_0 \right\} \right)

     \]  We have \[
     \lim_{t\to 0^{+ }}f\left( x_0+ t \right)-f\left( x_0 \right) =  \lambda _{f} \left( \left\{ x_0 \right\} \right)  
     \] The LHS\(  = 0  \) iff \(  f  \) is left-continuous, which completes the proof.  
\end{proof}
\begin{exercise}{}{}
    Let $f$ be a nondecreasing function defined on $\mathbb{R}$. Define a Lebesgue-Stieltjes-type measure as follows: For $A \subset \mathbb{R}$ an arbitrary set,

\Lambda_{f}^{*}(A) = \inf \left\{ \sum_{h_k \in \mathcal{F}} [f(b_k) - f(a_k)] \right\},
\end{aligned}\]

E \subset \bigcup_{h_k \in \mathcal{F}} h_k.

In other words, the definition of $\Lambda_f^*(A)$ is the same as $\lambda_f^*(A)$ except that closed intervals $[a_k, b_k]$ are used instead of half-open intervals $(a_k, b_k]$.
As in the case of Lebesgue-Stieltjes measure it can be easily seen that $\Lambda_f^*$ is a Carathéodory measure. (You need not prove this.)

\item[(b)] Prove that $\Lambda_f^*(B) = \lambda_f^*(B)$ for all Borel sets $B$ if $f$ is left-continuous.
\end{enumerate}
\end{exercise}

\begin{proof}
    \begin{enumerate}
        \item Let \(  (a_1,b_1],(a_2,b_2],\cdots   \) be a sequence of  half-open intervals , such  that \[
        A\subseteq \bigcup _{k = 1}^{\infty}(a_{k},b_{k}],\quad \sum _{k = 1}^{\infty} (f\left( b_{k} \right)-f\left( a_{k} \right)  ) <  \lambda ^{*}_{f}\left( A \right)+  \varepsilon

         \Lambda _{f}^{*}\left( A \right) \le  \lambda _{f}^{*}\left( A \right) 
        \]
        \item   Let \( \left[ a_1,b_1 \right],\left[ a_2,b_2\right],\cdots      \) be a sequence of closed intervals , such that \[
        B\subseteq \bigcup _{k = 1}^{\infty}\left[ a_{k},b_{k} \right],\quad \sum _{k = 1}^{\infty}\left( f\left( b_{k} \right)-f\left( a_{k} \right)   \right)<  \Lambda _{f}^{*}\left( B \right)+  \varepsilon

        \] 
    \end{enumerate}
    
\end{proof}

\section{Hausdorff Measure}
\begin{definition}{Hausdorff Outer Measure}{8-19-1}
Hausdorff measure is defined in terms of an auxiliary set function that we introduce first. Let $0 \le s < \infty$, $0 < \varepsilon \le \infty$, and let $A \subset \mathbb{R}^n$. Define
\[
H_{\varepsilon}^{s}(A) = \inf \left\{ \sum_{i=1}^{\infty} \alpha(s) 2^{-s} (\operatorname{diam} E_i)^s : A \subset \bigcup_{i=1}^{\infty} E_i, \operatorname{diam} E_i < \varepsilon \right\},
\]
where $\alpha(s)$ is a normalization constant defined by
\[
\alpha(s) = \frac{\pi^{\frac{s}{2}}}{\Gamma(\frac{s}{2}+1)},
\]
with
\[
\Gamma(t) = \int_{0}^{\infty} e^{-x} x^{t-1} \,dx, \quad 0 < t < \infty.
\]It follows from the definition that if $\varepsilon_1 < \varepsilon_2$, then $H_{\varepsilon_1}^s(E) \ge H_{\varepsilon_2}^s(E)$. This allows the following, which is the definition of \dfntxt{s-dimensional Hausdorff measure}:
\[
H^s(A) = \lim_{\varepsilon \to 0} H_{\varepsilon}^s(A) = \sup_{\varepsilon > 0} H_{\varepsilon}^s(A).
\]

\end{definition}

\begin{remark}
    \begin{itemize}
        \item Hausdorff Measure can be defined in any metric space.
        \item The sets \(  E_{i}  \) in the definition of \(  H_{ \varepsilon }^{s}\left( A \right)   \) are arbitrary subsets of \(  \mathbb{R} ^{n}  \). However, they could be taken to be closed sets, since \(  \operatorname{diam}E_{i}= \operatorname{diam} \overline{E_{i}}  \).     
    \end{itemize}
    
\end{remark}


\begin{theorem}{}{}
    For each nonnegative number \(  s  \), \(  H^{s}  \) is a Carathéodory outer measure.  
\end{theorem}

\begin{proof}
    First we show that \(  H^{s}  \)  is an outer measure. We only show that \(  H^{s}  \) is subadditive, the other properties are obvious. Let \(  A_1,A_2,\cdots   \) be a sequence of sets with \(  H^{s}  \)- outer measure finite. Let \(  A= \bigcup _{k = 1}^{\infty}A_{k}  \). For any \(   \varepsilon > 0  \), and for each \(  k  \), there are sets \(  E^{\left( k \right) }_{1},\cdots   \) such that \[
    A_{k}\subseteq  \bigcup _{i = 1}^{\infty}E_{i}^{\left( k \right) },\quad  \sum _{i = 1}^{\infty}\alpha \left( s \right)2^{-s} \left( \operatorname{diam}E_{i}^{\left( k \right) } \right)^{s}< H_{ \varepsilon }^{s}\left( A_{k} \right)+ \frac{ \varepsilon  }{2^{k} } 
    \]       We can arange \(  \left\{ E_{i}^{\left( k \right) } \right\}_{i,k \in \mathbb{Z} _{> 0}}  \) to be a sequence of sets such that  \[
    A\subseteq \bigcup _{i,k \in \mathbb{Z} _{> 0}}E_{i}^{\left( k \right) },\quad  \sum _{i,k \in \mathbb{Z} _{> 0}} \alpha \left( s \right)2^{-s}\left( \operatorname{diam}E_{i}^{\left( k \right) } \right)^{s}< \sum _{k= 1}^{\infty}H_{ \varepsilon }^{s}\left( A_{k} \right)   +  \varepsilon 
    \]  Thus \[
    H_{ \varepsilon }^{s}\left( A \right)\le \sum _{k = 1}^{\infty} H_{ \varepsilon }^{s}\left( A_{k} \right)+  \varepsilon  
    \] Since \(  H_{ \varepsilon }^{s}\left( A_{k} \right)\le  H^{s}\left( A_{k} \right)    \) for each \(  k  \), we have \[
    H_{ \varepsilon }^{s}\left( A \right)\le  \sum _{k = 1}^{\infty}H^{s}\left( A_{k} \right)+  \varepsilon   
    \]   Let \(   \varepsilon > 0  \), we have \[
    H^{s}\left( A \right)\le \sum _{k = 1}^{\infty}H^{s}\left( A_{k} \right)  
    \]
    Now we show that \(  H^{s}  \) is a Carathéodory outer measure. Suppose that \(  A,B  \) be two sets with \(  \mathbf{d}\left( A,B \right)> 0   \), and suppose \(  0<  \varepsilon < \mathbf{d}\left( A,B \right)   \).    
    Let \(  \left\{ E_{k} \right\}  \) be a covering of \(  A\cup B  \)   with \(  \operatorname{diam}\left( E_{i} \right)< \varepsilon    \) .Then each \(  E_{k}  \) cannot intersect with both \(  A  \) and \(  B  \). Let \(  \mathcal{A}  \) be the collection of  \(  E_{k}  \) intersecting with \(  A  \) , and Let \(  \mathcal{B}  \) be that of intersecting with \(  B  \). Then  \[
   \begin{aligned}
    \sum _{k = 1}^{\infty} \alpha \left( s \right)2^{-s}\left( \operatorname{diam}E_{i} \right)^{s}&  = \sum _{ E_{k}\in \mathcal{A}}\alpha \left( s \right)2^{-s} \left( \operatorname{diam}E_{k} \right)^{s} + \sum _{ E_{k}\in \mathcal{B}}\alpha \left( s \right)2^{-s}\left( \operatorname{diam}E_{k} \right)^{s}   \\ 
     &\ge H_{ \varepsilon }^{s}\left( A \right)+ H_{ \varepsilon }^{s}\left( B \right)  
   \end{aligned}
    \]  We can take \(  \left\{ E_{i} \right\}  \) such that \[
    \sum _{k = 1}^{\infty}\alpha \left( s \right)2^{-s}\left( \operatorname{diam}E_{i} \right)^{s}< H_{ \varepsilon }^{s}\left( A\cup B \right)   +  \varepsilon 
    \]     Thus we have \[
    H_{ \varepsilon }^{s}\left( A\cup B \right)\ge H_{ \varepsilon }^{s}\left( A \right)+ H_{ \varepsilon }^{s}\left( B \right)+  \varepsilon    
    \]Since \(  0<  \varepsilon < \mathbf{d}\left( A,B \right)   \) is arbitrary, take \(   \varepsilon \to 0^{+ }  \), it follows that \[
    H^{s}\left( A\cup B \right)\ge H^{s}\left( A \right)+ H^{s}\left( B \right)   
    \], which shows that \(  H^{s}  \) is a Carathéodory outer measure.      
\end{proof}

\begin{theorem}{}{}
For each $A \subset \mathbb{R}^n$, there exists a Borel set $B \supset A$ such that
\[H^s(B) = H^s(A).\]
\end{theorem}

\begin{proof}
    Let \(  \left\{  \varepsilon _{j} \right\}  \) be a decreasing suquence of positive numbers such that \(   \varepsilon _{j}\to 0   \) as \(  j\to \infty  \). For each \(  j  \), there is a suquence of closed set (Remark of Definition \ref{dfn:8-19-1}) \(  E_{i,j}  \) , such that \[
    B\subseteq \bigcup _{i= 1}^{\infty}E_{i,j},\quad \sum _{i= 1}^{\infty}\alpha \left( s \right)2^{-s} \left( \operatorname{diam}\left( E_{i,j} \right)  \right)^{s}<  H_{ \varepsilon_{j} }^{s}\left( B \right)+  \varepsilon _{j}   
    \] Let \(  A_{j}: = \bigcup _{i = 1}^{\infty}E_{i,j}  \), and let \(  A: = \bigcap_{j= 1}^{\infty}A_{j}   \). Then   \(  A  \) is a Borel set such that \(  B\subseteq A  \). Note that \[
    H_{ \varepsilon _{j}}^{s}\left( B \right)+  \varepsilon _{j}>  \sum _{i = 1}^{\infty} \alpha \left( s \right)2^{-s}\left( \operatorname{diam}\left( E_{i,j} \right)  \right)^{s}\ge H_{ \varepsilon _{j}}^{s}\left( A_{j} \right)\ge  H_{ \varepsilon _{j}}^{s} \left( A \right) 
    \]    Let \(  j\to \infty  \), we have \[
    H^{s}\left( B \right)\ge H^{s}\left( A \right)  \implies H^{s}\left( B \right)= H^{s}\left( A \right)  
    \] 
\end{proof}

\begin{theorem}{}{8-19-2}
Suppose $A \subset \mathbb{R}^n$ and $0 \le s < t < \infty$. Then
\begin{enumerate}
\item[(i)] If $H^s(A) < \infty$ then $H^t(A) = 0$.
\item[(ii)] If $H^t(A) > 0$ then $H^s(A) = \infty$.
\end{enumerate}
\end{theorem}
\begin{proof}
    (ii) is just a restatement of (i). We now show that (i) holds.
    Take any \(   \varepsilon > 0  \), there are \(  \left\{ E_{i} \right\}  \) with \(  \operatorname{diam}\left( E_{i} \right)\le  \varepsilon    \) such that \[
    A\subseteq \bigcup _{i= 1}^{\infty}E_{i},\quad  \sum _{i = 1}^{\infty}\alpha \left( s \right) 2^{-s} \left( \operatorname{diam}\left( E_{i} \right)  \right)^{s}< H_{ \varepsilon }^{s}\left( A \right)+  \varepsilon  \le H^{s}\left( A \right)+  \varepsilon    
    \]Then \[
    \begin{aligned}
    H_{ \varepsilon }^{t}\left( A \right)&\le \sum _{i = 1}^{\infty}\alpha \left( t \right)2^{-t}\left( \operatorname{diam}\left( E_{i} \right)  \right)^{t}\\ 
     &= \frac{\alpha \left( t \right)  }{\alpha \left( s \right)  }2^{s-t}  \sum _{i = 1}^{\infty}\alpha \left( s \right)2^{-s}    \left( \operatorname{diam}\left( E_{i} \right)  \right)^{s} \left( \operatorname{diam}\left( E_{i} \right)  \right)^{t-s}\\ 
      &\le \frac{ \alpha \left( s \right)  }{ \alpha \left( t \right)  } 2^{s-t}  \varepsilon ^{t-s}   \left( H^{s}\left( A \right)+  \varepsilon   \right) 
    \end{aligned}
    \]   Let \(   \varepsilon \to 0  \), we have \[
    H^{t}\left( A \right)\le 0 \implies H^{t}\left( A \right)= 0  
    \] 
\end{proof}

\begin{exercise}{}{}
    If $A \subset \mathbb{R}$ is an arbitrary set, show that $H^1(A) = \lambda^*(A)$.
\end{exercise}

\begin{remark}{}{}
In this problem, you will see the importance of the constant that appears in the definition of Hausdorff measure. The constant $\alpha(s)$ that appears in the definition of $H^s$-measure is equal to 2 when $s = 1$. That is, $\alpha(1) = 2$, and therefore, the definition of $H^1(A)$ can be written as
\[
H^1(A) = \lim_{\varepsilon\to 0} H_{\varepsilon}^1(A),
\]
where
\[
H_{\varepsilon}^1(A) = \inf \left\{ \sum_{i=1}^{\infty} \operatorname{diam} E_i : A \subset \bigcup_{i=1}^{\infty} E_i \subset \mathbb{R}, \operatorname{diam} E_i < \varepsilon \right\}.
\]
This result is also true in $\mathbb{R}^n$ but is more difficult to prove. The isodiametric inequality $\lambda^*(A) \le \alpha(n) \left(\frac{\operatorname{diam} A}{2}\right)^n$ (whose proof is omitted in this book) can be used to prove that $H^n(A) = \lambda^*(A)$ for all $A \subset \mathbb{R}^n$.
\end{remark}

\begin{proof}
    For any \(   \varepsilon > 0  \), there are closed intervals \(  I_1,\cdots   \) with \(  v\left( I_{i} \right)<  \varepsilon    \) such that \[
    A\subseteq \bigcup _{i = 1}^{\infty}I_{i},\quad \sum _{i= 1}^{\infty}v\left( I_{i} \right)<  \lambda ^{*}\left( A \right)+  \varepsilon   
    \]   Then \[
    \sum _{i= 1}^{\infty} \alpha \left( 1 \right)2^{-1} \left( \operatorname{diam}I_{i}  \right)^{1}= \sum _{i= 1}^{\infty}v\left( I_{i} \right)<  \lambda ^{*}\left( A \right)+  \varepsilon    
    \] Thus \[
    H_{ \varepsilon }^{1}\left( A \right) <  \lambda ^{*}\left( A \right)+  \varepsilon   
    \] Take \(   \varepsilon \to 0  \), we have \[
    H^{1}\left( A \right)\le   \lambda ^{*}\left( A \right)  
    \] 
    The other hand, take \(   \varepsilon > 0  \),  there are closed set \(  E_{i}  \) with \(  \operatorname{diam}E_{i}<  \varepsilon   \) , such that \[
    A\subseteq \bigcup _{i= 1}^{\infty}E_{i},\quad \sum _{i= 1}^{\infty} \operatorname{diam}E_{i}<  H^{1}_{ \varepsilon }\left( A \right) +  \varepsilon  
    \]  Note that \[
    \operatorname{diam}E_{i}=  \sup E_{i}-\inf E_{i}
    \]Let \[
    I_{i} : =  \left[\inf E_{i}-\frac{ \varepsilon  }{2^{i+ 1} }, \sup E_{i}+ \frac{ \varepsilon  }{2^{i+ 1} }   \right]
    \] Then \[
    E_{i}\subseteq I_{i},\quad  v\left( I_{i} \right)<  \operatorname{diam}E_{i}+ \frac{ \varepsilon  }{2^{i} } 
    \] Thus \[
    A\subseteq \bigcup _{i = 1}^{\infty}I_{i},\quad \sum _{i= 1}^{\infty}v\left( I_{i} \right) <  H^{1}_{ \varepsilon }\left( A \right)  + 2 \varepsilon < H^{1}\left( A \right)+ 2 \varepsilon  
    \] Since \(   \varepsilon > 0  \) is arbitrary,  we have \[
     \lambda ^{*}\left( A \right)\le  H^{1}\left( A \right)  
    \] 
\end{proof}

\section{Hausdorff Dimension of Cantor Sets}

\begin{definition}{General Cantor set}{}
Let $0 < \gamma < 1/2$ and set $I_{0,1} = [0, 1]$. Let $I_{1,1}$ and $I_{1,2}$ denote the intervals $[0, \gamma]$ and $[1 - \gamma, 1]$ respectively. They result by deleting the open middle interval of length $1 - 2\gamma$. At the next stage, delete the open middle interval of length $\gamma(1 - 2\gamma)$ of each of the intervals $I_{1,1}$ and $I_{1,2}$. There remain $2^2$ closed intervals each of length $\gamma^2$. Continuing this process, at the $k$th stage there are $2^k$ closed intervals each of length $\gamma^k$. Denote these intervals by $I_{k,1}, \dots, I_{k,2^k}$. We define the generalized Cantor set as
\[
C(\gamma) = \bigcap_{k=0}^{\infty} \bigcup_{j=1}^{2^k} I_{k,j}.
\]
\end{definition}


\begin{theorem}{Hausdorff Dimension of Cantor Sets}{}
    The Hausdorff Diemension of \(  C\left(  \gamma  \right)   \), denoted by  \(  s\), is \[
   s= \log 2/ \log\left( \frac{1 }{ \gamma  }  \right) 
    \] that is the unique \(  s  \) such that   \[
    2 \gamma ^{s}= 1
    \] 
\end{theorem}

\begin{note}
    通过选取特定的 \(  s  \), 使得 \(  C\left(  \gamma  \right)   \)构造过程中的每一个 ``丢弃''中间区间的过程都是恰好 完全不损失``\(  s  \)-长度的 ''.  这样, \(  C\left(  \gamma  \right)   \)中 \(  I_{k,j}  \)的 \(  s  \)-长度可以完全它内部更小的组件的 \(  s  \)-长度和来代替.    

    由于\(  C\left(  \gamma  \right)   \)的自相似性, 每个区间 \(  I  \), 在盖住了最大的一个小组件 \(  I_{k,k_{1}}  \)时,  也会同时连同被跟它一样大的另外三个组件 \(  I_{k,k_2},I_{k,k_3},I_{k,k_4}  \)  所含住.这样就提供了上下两边包夹的不等式关系.
\end{note}

\begin{proof}
    Since \(  C\left(  \gamma  \right)\subseteq \bigcup _{j= 1}^{2^{k}}I_{k,j}   \)  for each \(  k  \). We have  \[
    H^{s}_{ \gamma ^{k}}\left( C\left(  \gamma  \right)  \right) \le \sum _{j= 1}^{2^{k}} \ell\left( I_{k,j} \right) ^{s}= 2^{k}\left(  \gamma ^{k} \right)^{s}=  \left( 2 \gamma ^{s} \right)^{k}  
    \] We choose \(  s  \) such that \(  2 \gamma ^{s}= 1  \), then    \[
    H^{s}_{ \gamma ^{k}}\left( C\left(  \gamma  \right)  \right)\le  1,\quad \forall k 
    \] Since \(  \lim_{k\to \infty} \gamma ^{k}= 0  \), we have \[
    H^{s}\left( C\left(  \gamma  \right)  \right)= \lim_{k\to \infty}H^{s}_{ \gamma ^{k}}\left( C\left(  \gamma  \right)  \right)\le 1  
    \]  

    It is important to observe that we choose our \(  s  \) such that \[
    \sum _{j= 1}^{2^{k}}  \ell\left( I_{k,j} \right)^{s}= 1 
    \] 

    Next, we show that \(  H^{s}\left( C\left(  \gamma  \right)  \right)\ge \frac{1 }{4 }    \) , which along with above implies that the Hausdorff dimension of \(  C\left(  \gamma  \right)   \) equals \(  \log 2 / \log\left( \frac{1 }{ \gamma  }  \right)   \) . We will establish this by showing that if \[
    C\left(  \gamma  \right)\subseteq \bigcup _{j= 1}^{\infty}J_{i} 
    \] is an open covering \(  C\left(  \gamma  \right)   \) by intervals \(  J_{i}  \), then \[
    \sum _{i= 1}^{\infty} \ell\left( J_{i} \right)^{s}\ge \frac{1 }{4 }  
    \] Firstly, we will show that for any open interval \(  I  \) and fixed \(   l  \) that \[
    \sum _{I_{i,l}\subseteq I} \ell\left( I_{l,i} \right)^{s}\le 4 \ell\left( I \right)^{s}  
    \]    Let \(  k  \) be the smallest number    such that \(  I  \) contains some \(  I_{k,j}  \). Then there are at most for of \(  I_{k,j}  \)'s intersect with \(  I  \), otherwise there are some \(  I_{k-1,j}  \) contained in \(  I  \). We call them \(  I_{k,k_{m}},m = 1,2,3,4  \).  Then \[
    4 \ell\left( I \right)^{s} \ge \sum _{m= 1}^{4} \ell\left( I_{k,k_{m}} \right)^{s}=  \sum _{m= 1}^{4}\sum _{I_{l,i}\subseteq I_{k,k_{m}}}  \ell\left( I_{l,i} \right)^{s} 
    \]    Since \[
     \ell\left( I_{k,k_{m}} \right)^{s}= \frac{1 }{2^{k} },\quad  \sum _{I_{l,i}\subseteq I_{k,k_{m}}}   \ell\left( I_{l,i} \right)^{s}= 2^{l-k}\frac{1 }{2^{l} }= \frac{1 }{2^{k} }   
    \]  Thus \[
    4 \ell\left( I \right)^{s}\ge \sum _{I_{l,i}\subseteq I} \ell\left( I_{l,i} \right)^{s}  
    \] From this, we can get \[
    \sum _{i= 1}^{\infty} \ell\left( J_{i} \right)^{s}\ge  \frac{1 }{4 }\sum _{i= 1}^{\infty}\sum _{I_{l,k}\subseteq J_{i}} \ell\left( I_{l,k} \right)^{s}  \ge \frac{1 }{4 }\sum _{k= 1}^{2^{l}} \ell\left( I_{l,k} \right)^{s}= \frac{1 }{4 }   
    \] since the covering has an Lebesgue number , thus for sufficiently large \(  l  \), each \(  I_{l,k}  \) is contained in some \(  J_{i}  \).   
\end{proof}
