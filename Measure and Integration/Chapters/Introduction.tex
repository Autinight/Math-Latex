\subsection{Equivalence between sets}

\begin{lemma}{}{}
    Suppose \(  f:X\to Y  \), \(  g:Y\to X  \), then there are decompositions \[
    X= A\cup \tilde{A},\quad Y= B\cup \tilde{B}
    \]such that \(  A\cap \tilde{A}= \varnothing, B\cap \tilde{B}= \varnothing  \) and \(  f\left( A \right)= B   \), \(  g\left( \tilde{B} \right)= \tilde{A}  \).     
\end{lemma}

\begin{note}
    我们从随便一个 \(  A  \)出发, 先让一些简单的条件得到满足, 比如 \(  B= f\left( A \right), \tilde{B}= Y\setminus B   \) , \(  \tilde{A}= g\left( Y\setminus f\left( E \right)  \right)   \), 就它们确定地写出来. 剩下 \(  A\cap \tilde{A} = \varnothing \)和 \(  A\cup \tilde{A}= X  \)需要满足.   直接找到特定的目标一开始是困难的, 于是我们先把满足 \(  A\cap \tilde{A}= \varnothing  \)的``好东西''都收集起来, 即 \(  \mathcal{S}  \). 再在 \(  \mathcal{S}  \)中寻找一个满足 \(  A\cup \tilde{A}= X  \)的候选. ``极大元''就是一个看起来非常可疑的对象(如果它也落在 \(  \mathcal{S}  \), 事实上也的确如此 ). 
\end{note}

\begin{proof}
    Let  \[
    \mathcal{S}: = \left\{ E\subseteq X: E\cap g\left( Y\setminus f\left( E \right)  \right) = \varnothing  \right\}.
    \] We define \[
    A: = \bigcup _{ E \in \mathcal{S}}E
    \] Let \[
    B: = f\left( A \right),\quad \tilde{B}: = Y\setminus f\left( A \right),\quad \tilde{A}: =  g\left( \tilde{B} \right)= g\left( Y\setminus f\left( A \right)  \right)    
    \] It is obvious that \(  B\cap \tilde{B}= \varnothing, B\cup \tilde{B}= Y  \). We should show \(  A\cap \tilde{A}= \varnothing  \), which is equivalent to \(  A\in \mathcal{S}  \),  and   \(  A\cup \tilde{A}= X  \).   First , take any \(  E \in \mathcal{S}  \), we have \[
    E\cap g\left( Y\setminus f\left( A \right)  \right)\subseteq E\cap g\left( Y\setminus f\left( E \right)  \right)= \varnothing  .
    \]Then we concludes that  \[
    A\cap g\left( Y\setminus f\left( A \right)  \right)= \varnothing\implies A\in \mathcal{S} 
    \] To show that \(  A\cup \tilde{A}= X  \), we suppose conversely that there is \(  x\in X  \) such that \[
    x \not \in A\cup \tilde{A}
    \]  Let \(  A^{\prime} : =  A\cup \left\{ x \right\}  \), then \[
   \begin{aligned}
   A^{\prime} \cap g\left( Y\setminus f\left( A^{\prime}  \right)  \right)&= \left( A\cup \left\{ x \right\} \right)\cap g\left( Y\setminus \left( f\left( A \right)\cup \left\{ x \right\}  \right)  \right)  \\ 
    &\subseteq \left( A\cup \left\{ x \right\} \right)\cap \tilde{A} \\ 
    &= \left( A\cap \tilde{A}  \right) \cup \left( \left\{ x \right\}\cap \tilde{A} \right)     \\ 
     &= \varnothing
   \end{aligned}
    \] Thus \(  A^{\prime} \in \mathcal{S}  \), which is a contradiction to the definition of \(  A  \).  
\end{proof}


\begin{theorem}{Cantro-Bernstein}{}
    If \(  X\supseteq A, Y\supseteq B  \),  and if \(  X\sim B  \), \(  Y\sim A  \), then \(  X \sim Y  \).    
\end{theorem}

\begin{proof}
    We know that there are injection \(  f :X\to Y   \) and \(  g:Y\to X  \). There are decompositions  \[
    X= A\cup \tilde{A},\quad Y= B\cup \tilde{B}
    \]such that \(  A\cap \tilde{A}= \varnothing, B\cap \tilde{B}= \varnothing  \) and \(  f\left( A \right)= B   \), \(  g\left( \tilde{B} \right)= \tilde{A}  \).     Then we define \(  F:X\to Y  \) \[
    F\left( x \right)\coloneqq \begin{cases} f\left( x \right),& x \in A\\ 
     \left( g|_{\tilde{B}} \right) ^{-1} \left( x \right) ,& x \in \tilde{A}   \end{cases}  
    \] is a bijection.
\end{proof}

        \item Let \(  f:\mathbb{R} ^{1}\to \mathbb{R} ^{1}  \) be a monotone function, then \[

        \item Let \(  f: \mathbb{R} \to \mathbb{R}   \), define \[

\end{example}
\begin{proof}

        \item We take a rational number \(  r_{x}  \) in \(  \left( f\left( x-0 \right), f\left( x+ 0 \right)   \right)   \) for each \(  x  \) that \(  f  \) is not continuous on it.      
    \end{enumerate}
    
\end{proof}

\begin{theorem}{}{}
    Let \(  A  \) be a infinite set and \(  \left| A \right|= \alpha    \). If \(  B  \) is countable, then \[
    \left| A\cup B \right|= \alpha  
    \]   
\end{theorem}
\begin{proof}
    Since \(  B^{\prime} \coloneqq B\setminus A  \) is countable, and \(  A\cup B^{\prime}= A\cup \left( B\setminus A \right)= A    \), we may assume that \(  A\cup B= \varnothing  \).   There exists countable set \(  A_1\subseteq A  \). Then \(  A_1\cup B  \) is countable. We have \(  A_1\cup B\sim A_1  \). There exists bijection \(  f:A_1\cup B\to A_1  \). Since \[
    A= A_1\cup \left( A\setminus A_1 \right),\quad A\cup B= \left( A_1\cup B \right)\cup \left( A\setminus A_1 \right)   
    \]Define \(  F:A\cup B\to A   \)  \[
    F\left( x \right)= \begin{cases} f\left( x \right),&x \in A_1\cup B \\ 
     x,&x \in  A\setminus A_1   \end{cases}  
    \]     Then \(  F  \) is a bijection. Thus \(  \left| A\cup B \right|= \left| A \right|= \alpha     \). 
\end{proof}

\begin{theorem}{}{}

   A\text{ is a infinite set} \iff  \exists A_1 \text{ is a proper subset of }A, s.t. \left| A_1 \right|= \left| A \right|  
    \]
\end{theorem}

\begin{theorem}{}{}
    \(  \left[ 0,1 \right]= \left\{ x:0\le x\le 1 \right\}   \) is a uncoutable set. 
\end{theorem}

\end{proofsketch}

\begin{definition}{}{}

    \] called the \dfntxt{cardinality of the continuum}.

\begin{theorem}{}{}

\end{proofsketch}

\section{Metric Space}
\begin{lemma}{}{}
    \(  X  \) is a metric space, \(  x \in X  \), \(  B_1,B_2  \) are twos balls in \(  X  \). If \(  x  \in B_1\cap B_2  \) , then \(  x  \) is the center of an open ball \(  B\subseteq  B_1\cap B_2  \).       
\end{lemma}
\begin{proof}
    We may assume that \(  B_1,B_2  \) have  different center. Let  \(  y_1\neq y_2  \), \[
B_1= \left\{ x: d\left( x,y_1 \right)< r_1  \right\},\quad B_2= \left\{ x:d\left( x,y_2 \right)< r_2  \right\}
\] 
Take  \(  x \in B_1\cap B_2  \), we claim that there exists  \(  r_0> 0  \) such that  \(  B: =  \left\{ y: d\left( x,y \right)< r_0  \right\}\subseteq B_1\cap B_2  \). Otherwise , for all \(  r > 0 \),  there exists a point \(  y_{r}  \) such that   the following two hold at the same time \begin{enumerate}
    \item \(  d\left( x,y_{r} \right)< r   \)
    \item \(  d\left( y_{r},y_1 \right)\ge r_1   \) or \(  d\left( y_{r},y_2 \right)\ge r_2   \).   
\end{enumerate}
Thus \[
d\left( x,y_1 \right)\ge d\left( y_{r},y_1 \right)-d\left( x,y_{r} \right)>  r_1-r   ,\quad  \text{or} \quad  d\left( x,y_2 \right)> r_2-r 
\] The above shows that \[
r> \min \left\{ r_1-d\left( x,y_1 \right),r_2-d\left( x,y_2 \right)   \right\},\quad \forall r> 0
\] which is a contradiction.
\end{proof}

\section{Limits of Set Sequences}

\begin{definition}{Limit Inferior and Limit Superior}{}
    Let $\{A_n\}_{n=1}^\infty$ be a sequence of sets.
    \begin{enumerate}
        \item By the \dfntxt{Limit Inferior of the sequence}, we mean \[
        \liminf_{n \to \infty} A_n = \bigcup_{N=1}^\infty \bigcap_{n=N}^\infty A_n = \left\{ x:  \exists N_0\in \mathbb{N} , \forall n> N_0,x \in A_{n}\right\}
        \]
        \item By the \dfntxt{Limit Superior of the sequence},we mean \[
        \limsup_{n \to \infty} A_n = \bigcap_{N=1}^\infty \bigcup_{n=N}^\infty A_n =  \left\{ x: \forall N\in \mathbb{N} , \exists n\ge N, x\in A_{n} \right\}
        \]
    \end{enumerate}
    
\end{definition}
\begin{note}
    \begin{enumerate}
        \item Limit Inferior包含那些“最终稳定下来”的元素，即从某个点之后就永远属于序列中所有后续集合的元素。
        \item  Limit Superior 包含那些“反复出现”的元素，即在无限多个集合中出现的元素。
    \end{enumerate}

\end{note}

\begin{proposition}{}{}
    For any sequence of sets $\{A_n\}$, it always holds that:
$$\liminf_{n \to \infty} A_n \subseteq \limsup_{n \to \infty} A_n$$
\end{proposition}
\begin{proofsketch}
    直觉上是显然的,因为 “最终稳定下来”的元素一定会“反复出现”.
\end{proofsketch}

\begin{proof}
    It is obvious by the \(  \left\{ x:x \in P \right\}  \) form representation of the Limit Inferior and Superior.
\end{proof}

\begin{definition}{Limit of a Set Sequence}{}
    If the limit inferior and limit superior of a set sequence $\{A_n\}_{n=1}^\infty$ are equal, i.e., $\liminf_{n \to \infty} A_n = \limsup_{n \to \infty} A_n$, then we say the \dfntxt{limit} of the sequence exists, and it is defined as:
$$\lim_{n \to \infty} A_n = \liminf_{n \to \infty} A_n = \limsup_{n \to \infty} A_n$$
\end{definition}

\begin{proposition}{}{}
     Let $\{A_n\}_{n=1}^\infty$ be a sequence of sets.
     \begin{enumerate}
        \item If $\{A_n\}$ is an increasing sequence, then the limit of \(  \left\{ A_{n} \right\}   \) exists and \[
        \lim_{n\to \infty}A_{n}= \bigcup _{n = 1}^{\infty}A_{n}
        \]
        \item If \(  \left\{ A_{n} \right\}  \) is a decreasing sequence, then the limit of \(  \left\{ A_{n} \right\}   \) exists and  \[
        \lim_{n\to \infty}A_{n}= \bigcap_{n = 1}^{\infty}A_{n} 
        \] 
     \end{enumerate}
     
\end{proposition}
\begin{proof}
    \begin{enumerate}
        \item If \(  \left\{ A_{n} \right\}  \) is increasing , then  \[
        \bigcap_{n = N}^{\infty}A_{n}= A_{N} 
        \]and\[ 
       \bigcup _{n = N}^{\infty}A_{n}= \bigcup _{n =  1}^{\infty}A_{n},\forall N\in \mathbb{N} .
        \] Thus \[
        \liminf_{n \to \infty} A_n = \bigcup_{N=1}^\infty \bigcap_{n=N}^\infty A_n = \bigcup _{N = 1}^{\infty}A_{N}
        \] and\[
        \limsup_{n \to \infty} A_n = \bigcap_{N= 1}^{\infty}\bigcup _{n = N}^{\infty}A_{n}=  \bigcap_{N = 1}^{\infty} \bigcup _{n = 1}^{\infty}A_{n}=  \bigcup _{n = 1}^{\infty}A_{n}  
        \]which are the same.
        \item If \(  \left\{ A_{n} \right\}  \) is decreasing, then \[
        \bigcap_{n = N}^{\infty}A_{n}= \bigcap_{n = 1}^{\infty}A_{n}  
        \] and \[
        \bigcup _{n = N}^{\infty}A_{n}= A_{N}
        \]Thus  \[
        \liminf_{n \to \infty} A_n = \bigcup_{N=1}^\infty \bigcap_{n=N}^\infty A_n = \bigcup _{N = 1}^{\infty}\bigcap_{n = 1}^{\infty}A_{n}= \bigcap_{n = 1}^{\infty}A_{n}  
        \] and\[
        \limsup_{n \to \infty} A_n = \bigcap_{N= 1}^{\infty}\bigcup _{n = N}^{\infty}A_{n}= \bigcap_{N = 1}^{\infty}A_{N}   
        \]which are the same.
    \end{enumerate}
    
\end{proof}

\begin{proposition}{}{}
    For a sequence of sets $\{A_n\}_{n=1}^\infty$ and their corresponding sequence of indicator functions $\{\chi_{A_n}(x)\}_{n=1}^\infty$:
    \begin{itemize}
        \item 
$\chi_{\liminf_{n \to \infty} A_n}(x) = \liminf_{n \to \infty} \chi_{A_n}(x)$
\item $\chi_{\limsup_{n \to \infty} A_n}(x) = \limsup_{n \to \infty} \chi_{A_n}(x)$
\item  $\lim_{n \to \infty} A_n$ exists, then
$\chi_{\lim_{n \to \infty} A_n}(x) = \lim_{n \to \infty} \chi_{A_n}(x)$
    \end{itemize}
    
\end{proposition}
\begin{proofsketch}
    \begin{itemize}
        \item 若 \(  x  \)在limit inferior 里面,则\(  x  \)是“最终稳定”的, \(  \chi _{A_{n}}\left( x \right)   \)是关于 \(  n  \) “最终”恒为1的.
        \item 若 \(  x  \)在 limit superior里面,则 \(  x  \)是  “反复出现”的,即相当于 \(  N  \)多大,总会出现之后的某个 \(  n  \)使得 \(  \chi _{A_{n}}\left( x \right)= 1   \).
        \item 当极限存在时,函数列 \(  \left\{ \chi _{A_{n}}\left( x \right)  \right\}_{n = 1}^{\infty}  \)的limit inferior 和 supperior根据上两条相等,等于极限集合的 \(  \chi   \). 
    \end{itemize}
    
\end{proofsketch}

\section{Ordinal and Transfinite Induction}

\begin{definition}{Ordinal}{}
    A set \(  \alpha   \) is said to be an \dfntxt{ordinal}, if the following two hold:
    \begin{enumerate}
        \item Transitive Set: If \(  x \in \alpha   \), then \(  x\subseteq \alpha   \). (Equivalently, if \(  z \in y  \), \(  y \in \alpha   \), then \(  z \in \alpha   \)   ).
        \item Well-ordered by \(  \in   \): For any \(  x,y \in \alpha   \), either \(  x \in y  \) or \(  y \in x  \).     And there is a \(  \in   \)-smallest element in any nonempty subset of \(  \alpha   \).       
    \end{enumerate}
     
\end{definition}

\begin{lemma}{}{}
    If \(   \alpha   \) is an ordinal, then \(  \alpha \not \in \alpha   \).  
\end{lemma}
\begin{proof}
    If \(  \alpha \in \alpha   \), then \(  \left\{  \alpha  \right\}  \) is a subset of \(  \alpha   \). But there is no smallest  ordinal in \(  \left\{ \alpha  \right\}  \), which is a contradiction.  
\end{proof}
\begin{definition}{Classification of Ordinal}{}
    \begin{enumerate}
        \item \dfntxt{Zero Ordinal}: Only \(  0= \varnothing  \). 
        \item  \dfntxt{Successor Ordinal}: Ordinal numbers of the form $\alpha+1 = \alpha \cup \{\alpha\}$
        \item \dfntxt{Limit Ordinal}: Ordinal that is neither zero ordinal nor successor ordinal.
    \end{enumerate}
    
\end{definition}

\begin{example}{Ordinal}{}
    \begin{enumerate}
        \item \[
         \omega = \left\{ 0,1,2,3,\cdots  \right\}
        \] is the smallest infinite ordinal, which is a limit ordinal.
        \item \[
         \omega \cdot 2:= \left\{ 0,1,2,\cdots , \omega , \omega + 1, \omega + 2,\cdots  \right\}
        \] is a limit ordinal. 
        \item \[
         \omega _{1}: = \left\{ \alpha :  \alpha  \text{ is a ountable ordinal} \right\}
        \] is the smallest uncountable ordinal. 

        \begin{enumerate}
            \item \dfntxt{Unountable}: If \(   \omega _1   \) is countable, then \(   \omega _{1} \in  \omega _{1}  \) by definition,  which is a contradiction.
            \item \dfntxt{Smallest}: If there is a uncountable ordinal \(  \beta   \) such that \(  \beta <  \omega _1   \). Then \(  \beta \in  \omega _1   \), which is a contradiction to the uncountability of \(  \beta   \).     
        \end{enumerate}
        
    \end{enumerate}
    
\end{example}

\begin{theorem}{Transfinite Induction}{}
     For any ordinal \(  \alpha   \), let \(  P\left( \alpha  \right)   \) be a property of \(  \alpha   \). If the following holds : 
    \begin{itemize}
        \item For any ordinal \(  \alpha   \), if \(  P\left( \beta  \right)   \) holds for any \(  \beta < \alpha   \) , then \(  P\left( \alpha  \right)   \) holds.    
    \end{itemize}
       Then \begin{itemize}
        \item For any ordinal \(  \alpha   \), \(  P\left( \alpha  \right)   \) is true.  
       \end{itemize}
       
\end{theorem}

\begin{proof}
    Let \(  S  \) be the collection of \(  \alpha   \) such that \(  P\left( \alpha  \right)   \) is false. Then if the transfinite indunction is not true, \(  S  \) is nonempty. Since the class of ordinal is well-ordered, there is a smallest ordinal \(   \gamma   \) in \(  S  \). Then for any \(  \beta <  \gamma   \), \(  \beta \not \in S  \) , that is \(  P\left( \beta  \right)   \) is true.   But it contradicts to what we supposed at first.      
\end{proof}

\section{Meager Sets in Topology}
\begin{definition}{DEFINITIONS}{}
\begin{itemize}
    \item A subset $E$ of a metric space $X$ is said to be \dfntxt{dense in an open set} $U$ if $\overline{E} \supset U$.
    \item  A set $E$ is defined to be \dfntxt{nowhere dense} if it is not dense in any open subset $U$ of $X$. Alternatively, we could say that $E$ is nowhere dense if $\overline{E}$ does not contain any open set.
    \item A set $E$ is said to be of \dfntxt{first category} in $X$ if it is the union of a countable collection of nowhere dense sets. 
    \item  A set that is not of the first category is said to be of the \dfntxt{second category} in $X$.
\end{itemize}

\end{definition}

\begin{theorem}{Baire category theorem}{}
A \dfntxt{complete metric space} X is not the union of a \dfntxt{countable collection of nowhere dense sets}. That is, a \dfntxt{complete metric space} is of the \dfntxt{second category}.
\end{theorem}
