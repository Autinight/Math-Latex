\begin{problem}{Mobius band}{}
    Let \(  M  \) b the Mobius band ,that is, the quotient of \(  \left[ 0,1 \right]^{2}   \) under the equivalence relation generated by \(  \left( x,0 \right)\sim \left( 1-x,1 \right)    \) for all \(  x \in \left[ 0,1 \right]   \) . Prove that \(  M  \) is homotopy equivalent to \(  \mathbb{S}^{1}  \) .       
\end{problem}
\begin{proof}
    Construct  \[
    \begin{aligned}
    f: \left[ 0,1 \right]^{2}\times \left[ 0,1 \right]&\to \mathbb{S}^{1}   \\ 
     \left( x,y,t \right)& \mapsto  \left( \frac{1 }{2 }t+ \left( 1-t \right)x, y   \right)  
    \end{aligned}
    \] Denote \(  f_{t}\left( x,t \right)= f\left( x,y,t \right)    \) ,then \[
    f_0\left( x,y \right)=\left( x,t \right),\quad f_1\left( x,t \right)= \left( \frac{1 }{2 },y  \right) ,\quad f_{t}\left( \frac{1 }{2 },y  \right)\equiv \left( \frac{1 }{2 },y  \right)      
    \] Which is a SDR to \(  \mathbb{S}^{1}  \) . 
\end{proof}
\begin{problem}{}{}

\begin{enumerate}
    \item Prove that the inclusion $O_n(\mathbb{R}) \to GL_n(\mathbb{R})$ is a homotopy equivalence. (Use the Gram-Schmidt orthonormalization procedure to construct the reverse map).
    \item Among the following matrix groups, determine which ones are compact, and determine their $\pi_0$:
    \[
        GL_n(\mathbb{C}), GL_n(\mathbb{R}), O_n(\mathbb{R}), SO_n(\mathbb{R}), U_n(\mathbb{C}), SU_n(\mathbb{C}).
    \]
\end{enumerate}

\end{problem}
\begin{proof}
    Take \(   A \in \operatorname{GL} \left( n,\mathbb{R}  \right)   \). Let \(  A= \left(  v_1,\cdots,v_n  \right)   \). Let \[
    \begin{aligned}
    e_1&=  \frac{v_1 }{\left\| v_1 \right\| }\\ 
     e_2&= \frac{v_2-\left<v_2,e_1 \right>e_1 }{\left\| v_2-\left<v_2,e_1 \right>e_1 \right\| }\\ 
      &\cdots    
    \end{aligned}
    \] \( \det Q\det \left( t \operatorname{Id}+  \left( 1-t \right)R  \right) \neq 0    \)  . Let \(  Q = \left(  e_1,\cdots,e_n  \right)   \). \(  A= QR  \) for some  upper triangular matrix \(  R  \).     Let \[
    \begin{aligned}
    f_{t}: \operatorname{GL} _{n}\left( \mathbb{R}  \right)&\to  \operatorname{GL} _{n}\left( \mathbb{R}  \right) \\ 
     A&\mapsto Q\left( t \operatorname{Id}+  \left( 1-t \right)R   \right)    
    \end{aligned}
    \]which is a SDR from  \(  A  \) to \(  Q  \).  
\end{proof}
