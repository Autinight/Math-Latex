\section{Quotient Spaces}

\begin{example}{}{}
Let $\mathbb{D}^n = \{x \in \mathbb{R}^n \mid \|x\| \leq 1\}$ be the $n$-disk and let $\mathbb{S}^{n-1} = \{x \in \mathbb{R}^n \mid \|x\| = 1\}$ be the $(n-1)$-sphere. Then $\mathbb{D}^n/\mathbb{S}^{n-1}$ is homeomorphic to $\mathbb{S}^n$.
\end{example}	

\begin{proof}
Given $x = (x_1, \dots, x_n) \in \mathbb{R}^n$, we let $i(x) = (x_1, \dots, x_n, 0) \in \mathbb{R}^{n+1}$. We let $u = (0, \dots, 0, 1) \in \mathbb{R}^{n+1}$.
Then we define:
\[
f : \mathbb{D}^n \to \mathbb{S}^n, f(x) := \cos(\pi\|x\|)u + \sin(\pi\|x\|)\frac{i(x)}{\|x\|}.
\]
Then $f$ is a continuous map which induces a continuous map $\bar{f} : \mathbb{D}^n / \mathbb{S}^{n-1} \to \mathbb{S}^n$. We can then check that $\bar{f}$ is a bijection with a compact source and Hausdorff target, thus it is a homeomorphism. 
\end{proof}

\begin{definition}{Pushout}{}
Let $A, X, Y$ be topological spaces and $f : A \to X, g : A \to Y$ be continuous maps. We denote by $X \cup_A Y$ the \dfntxt{pushout} of $f$ and $g$, defined as the quotient space $(X \sqcup Y)/\sim$ of the disjoint union $X \sqcup Y$ by the equivalence relation generated by $f(a) \sim g(a)$ for all $a \in A$.
\end{definition}
