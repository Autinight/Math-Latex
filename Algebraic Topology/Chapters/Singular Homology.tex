\chapter{ Singular Homology }

\section{The Construction}

\begin{definition}{}{}
Let $X$ be any topological space and $n \ge 0$ be an integer. By a \dfntxt{singular n-simplex} in $X$, we mean a continuous map $\sigma : |\Delta_n| \to X$. By a \dfntxt{singular n-chain} in $X$, we mean a finite sum $\sum_i n_i\sigma_i$, where $n_i \in \mathbb{Z}$ and $\sigma_i$ are singular $n$-simplices in $X$. We can add any two chains in an obvious way to obtain another chain by simply following the rule
\[
n\sigma + m\sigma = (n+m)\sigma \tag{4.5}
\]
Then the collection of all singular $n$-chains in $X$ becomes a free abelian group, with the empty sum playing the role of the zero-element. The collection of all singular $n$-simplices is then a basis for this group. We denote this group by $S_n(X)$.
\end{definition}

\begin{proposition}{Functoriality}{}
    If $f: X \to Y$ is a map of topological spaces, then $\sigma \mapsto f \circ \sigma$ extends to define a graded group homomorphism $f. : S.(X) \to S.(Y)$. Since $(g \circ f) \circ \sigma = g \circ (f \circ \sigma)$, it follows that the association $X \mapsto S.(X)$ defines a functor.

\end{proposition}

\begin{definition}{Relative Chain Complex}{}
    If \(  A  \) is a subspace of \(  X  \) and \(   \sigma   \) is a singular simplex in \(  A  \)  then it can be considered as a singular simplex in \(  X  \) via the inclusion map \(  A\hookrightarrow X  \) .    In this way, \(  S_{n}\left( A \right)   \) becomes a subgroup of \(  S_{n}\left( X \right)   \) . We denote the quotient group \(  S_{n}\left( X \right)/ S_{n}\left( A \right)    \) by \(  S_{n}\left( X,A \right)   \).        
\end{definition}
\begin{remark}
    \begin{enumerate}
        \item \(  S_{n}\left( X,A \right)   \) is a free abelian group with the collection of all singular n-simplexex \(   \sigma    \) in \(  X  \) whose image is not contained in \(  A  \) as a basis.    
        \item If \(  A = \varnothing  \), then clearly \(  S_{n}\left( X,A \right)=  S_{n}   \left( X \right) \).  
    \end{enumerate}
    
\end{remark}
\begin{definition}{}{}
For each integer $r \ge 0$, the \dfntxt{face map} or the \dfntxt{face operator} $F^r : \mathbb{R}^\infty \to \mathbb{R}^\infty$ is defined by
\[
F^r(\mathbf{e}_s) = \begin{cases} \mathbf{e}_s, & \text{if } s < r, \\ \mathbf{e}_{s+1}, & \text{if } s \ge r \end{cases}
\]
and extended linearly.
\end{definition}

\begin{remark}{}{}
\begin{enumerate}
    \item Note that for each $n \ge r$, $F^r$ carries $\Delta_{n-1}$ onto the $(n-1)$-face of $\Delta_n$ opposite to the vertex $\mathbf{e}_r$, in an \dfntxt{order preserving} manner.
    \item We shall denote each $F^r$ restricted to any $\Delta_n, n \ge r$ also by $F^r$, the exact meaning being understood from the context. 
\end{enumerate}
 
\end{remark}

\begin{lemma}{}{}
$F^r \circ F^{s-1} = F^{s} \circ F^r$ if $r < s$. \footnote{这里修正了shastri书中的一个笔误,他可能写错了\(  s-1  \)    的位置}
\end{lemma}
\begin{proof}
    Since \(  r< s  \), \(  r\le s-1  \)  , we fully expand the LHS by definition
   \[
   \begin{aligned}
   F^{r}\circ F^{s-1}\left( \mathbf{e}_{k} \right)&= \begin{cases} F^{r}\left( \mathbf{e}_{k} \right),& k< s-1\\ 
    F^{r}\left( \mathbf{e}_{k+ 1} \right),& k\ge s-1   \end{cases}\\ 
    &= \begin{cases} \mathbf{e}_{k},&k< s-1 , k< r\\ 
     \mathbf{e}_{k+ 1},&  r\le k< s-1\\ 
      \mathbf{e}_{k+ 1},& s-1\le k< r-1 \,(\text{empty set})\\ 
       \mathbf{e}_{k+ 2},& k\ge s-1, k\ge r-1 \end{cases}\\ 
        &= \begin{cases} \mathbf{e}_{k},&k< r\\ 
         \mathbf{e}_{k+ 1},&r\le k< s-1\\ 
          \mathbf{e}_{k+ 2},&k\ge s-1 \end{cases}     
   \end{aligned}
   \] Similarly,  \[
\begin{aligned}
   F^{s}\circ F^{r}\left( \mathbf{e}_{k} \right)&= \begin{cases}
F^{s}\left( \mathbf{e}_{k} \right),& k< r\\ 
 F^{s}\left( \mathbf{e}_{k+ 1} \right),&k\ge r    \end{cases}\\ 
  &= \begin{cases} \mathbf{e}_{k},&k<  r,k\le s\\ 
   \mathbf{e}_{k+ 1},&k< r,k\ge s \,(\text{empty set})\\ 
    \mathbf{e}_{k+ 1},&k\ge  r,k< s-1\\ 
     \mathbf{e}_{k+ 2}, &k\ge r,k\ge s -1\end{cases} \\ 
      &= \begin{cases} \mathbf{e}_{k},& k<  r\\ 
        \mathbf{e}_{k+ 1},& r\le  k< s-1\\ 
         \mathbf{e}_{k+ 2},&k\ge s-1
         \end{cases} 
\end{aligned}
   \]
\end{proof}

\begin{definition}{}{}
For $n \geq 1$ and for any singular $n$-simplex $\sigma$ in $X$, we define
\[
\partial_n(\sigma) = \sum_{r=0}^{n}(-1)^r \sigma \circ F^r
\]
and extend it linearly to obtain homomorphisms $\partial_n : S_n(X, A) \to S_{n-1}(X, A)$. We define $S_n(X, A) = (0)$ for $n \leq -1$ and $\partial_n = 0$ for $n \geq 0$.
\end{definition}
\begin{note}
    \(   \sigma   \) is a map from \(   \Delta _{n}  \) to \(  X  \).  \(   \sigma \circ F^{r}  \) is the restriction of \(   \sigma   \) on its face opposite to \(  \mathbf{e}_{r}  \). Then \(   \partial _{n}\left(  \sigma  \right)   \) is something like a closed \(  \left( n-1 \right)   \) - surface consists of  \(   \sigma   \) 's faces.  
\end{note}
\begin{proposition}{Singular Chain Complex}{}
$(S(X, A), \partial) = \{S_n(X, A), \partial_n\}$ is a chain complex and the association $(X, A) \leadsto (S.(X, A), \partial)$ is a functor.
\end{proposition}
\begin{proof}
    We shall first show that \(   \partial ^{2}\left(  \sigma  \right)= 0   \) for any singular n-simplex, \(  n\ge 2  \): \[
     \begin{aligned}
     \partial ^{2}\left(  \sigma  \right)&=  \partial \left( \sum _{s= 0}^{n}\left( -1 \right)^{s} \sigma \circ F^{s}  \right) \\ 
      &= \sum _{s= 0}^{n} \left( -1 \right)^{s} \partial \left(  \sigma \circ F^{s} \right)  \\ 
       &= \sum _{s= 0}^{n}\sum _{r= 0}^{n-1}\left( -1 \right)^{s}\left( -1 \right)^{r} \sigma \circ F^{s}\circ F^{r} \\ 
        &= \sum _{0\le r< s\le n}\left( -1 \right)^{s+ r} \sigma \circ F^{r}\circ F^{s-1}+ \sum _{n-1\ge r\ge s\ge 0}\left( -1 \right)^{s+ r} \sigma \circ F^{s}\circ F^{r}\\ 
         &= \sum _{0\le r< s\le n}\left( -1 \right)^{s+ r}\circ F^{r}\circ F^{s-1}+ \sum _{n-1\ge s\ge r\ge 0}\left( -1 \right)^{s+ r} \sigma \circ F^{r}\circ F^{s}    \\ 
          &= \sum _{0\le r< s\le n}\left( -1 \right)^{s+ r}\circ F^{r}\circ F^{s-1}-\sum _{n\ge s>  r\ge 0}  \left( -1 \right)^{s+ r} \sigma \circ F^{r}\circ F^{s-1}\\ 
           &= 0 
     \end{aligned} 
    \]  

    Given a map \(  f: \left( X,A \right)\to \left( Y,B \right)   \) . Since \(  \left( f\circ  \sigma  \right)\circ F^{r}= f\circ \left(  \sigma \circ F^{r} \right)    \) , we define \(  f.  \) as \(  f_{.}\left(  \sigma  \right):= f\circ  \sigma    \) extended by linearlity.  Then  \(   \partial \left( f.\left(  \sigma  \right)  \right)= f.\left(  \partial \left(  \sigma  \right)  \right)    \)  by the definition of \(    \partial   \) and \(  f.  \)'s linearlity.   The rest of the details are ommited.
\end{proof}

\begin{definition}{}{}
The \dfntxt{relative singular homology groups} of a topological pair $(X,A)$ are the homology groups $H_*(S.(X,A))$ and are denoted by $H_*(X,A)$. Taking $A = \varnothing$, we get the homology groups $H_*(X)$.
\end{definition}
\begin{remark}{}{}
Given any commutative ring $R$
\begin{enumerate}
    \item We could have formed the \dfntxt{free module} $S_n(X;R)$ of all the singular $n$-chains in $X$.
    \item Exactly as before, we would have got a chain complex $S.(X;R)$, subchain complex $S.(A;R)$ and the quotient complex $S.(X,A;R)$, etc. 
    \item The corresponding homology groups $H_*(X;R)$, etc. , are then called the homology groups $X$ with coefficients in $R$. Clearly they form a \dfntxt{graded $R$-module} by themselves. It is customary to drop the notation $R$ when $R=\mathbb{Z}$.
\end{enumerate}

\end{remark}

\section{Basic Properties}
\begin{remark}
    \begin{itemize}
          \item  It is easily seen that $\{S_.(X,A), \partial\}$ is a functor from the category of topological pairs to the category of chain complexes; thus $H_*(X,A)$ and $H_*(X)$ are also functors. 
        \item Given $f : (X,A) \to (Y,B)$, we shall denote by $f_.,$ the map induced by $f$ on $S_.(X,A)$ and by $f_*$ the map induced by $f$ on $H_*(X,A)$.
    \end{itemize}
    
\end{remark}
\begin{lemma}{}{}
There exist functorial chain homotopies
$$ h : S_q(X, A) \to S_{q+1}((X, A) \times \mathbb{I}), q \geq 0, $$
,called \dfntxt{prism operator }, between the two inclusion induced maps $\eta_0, \eta_1 : S.(X, A) \to S.((X, A) \times \mathbb{I})$, where for any \(  t\in \mathbb{I}  \), \(  \eta _{t}  \) is the inclusion map \[
        \eta _{t}\left( x \right)=  \left( x,t \right) ,\quad t \in \mathbb{I}
        \]  
\end{lemma}

\begin{theorem}{Homotopy invariance}{}
Let $f, g : (X, A) \to (Y, B)$ be \dfntxt{homotopic} to \dfntxt{each other}. Then $f_* = g_*$ on $H_*(X, A)$.
\end{theorem}

\begin{proof}
    Let \(  G  : \left( X,A \right)\times \mathbb{I} \to \left( Y,B \right)   \) be a homotopy between \(  f  \) and \(  g  \). For any \(  t \in \mathbb{I}  \), let \(  \eta _{t}:\left( X,A \right)\to \left( X,A \right)\times \mathbb{I}    \) be inclusion map \(  \eta _{t}\left( x \right)= \left( x,t \right)    \). Then \(  G\circ \eta _{0}= f  \), \(  G\circ \eta _{1}= g  \) . Only need to show that chain maps \(   \left( \eta _0  \right).  \) and \(  \left( \eta _1  \right).   \)  are chain homotopic, thus \(  \left( \eta _{0} \right)_{*}= \left( \eta _{1} \right)_{*}    \), \(  f_{*}= g_{*}  \), which is implied by the lemma above.  
\end{proof}

\begin{corollary}{}{}
Homotopy-equivalent topological spaces have isomorphic homology groups; in particular, a contractible space has the homology groups of a point-space.
\end{corollary}

\begin{example}{}{}
    Let \(  \star  \) be a point space. We have \[
    H_{k}\left( \star \right) = \begin{cases} \mathbb{Z} ,&k= 0\\ 
     0,&k \ge 1 \end{cases}  
    \] 
\end{example}
\begin{proof}
    Let $n$ be any positive integer. There is a unique singular $n$-simplex $\sigma : |\Delta_n| \to \{\star\}$ and hence $S_n(\star) \cong \mathbb{Z}$ for each $n \geq 0$. Observe that $\partial : S_{2n+1}(\star) \to S_{2n}(\star)$ is the zero-map because $\Delta_{2n+1}$ has an even number of $(2n)$-faces and the terms in $\partial(\sigma)$ cancel in pairs. However, similar reasoning tells us that $\partial : S_{2n}(\star) \to S_{2n-1}(\star)$ is an isomorphism. Thus we have a chain complex
\begin{center}
\begin{tikzcd}
... \arrow[r, "0"] & \mathbb{Z} \arrow[r, "\cong"] & \mathbb{Z} \arrow[r, "0"] & \mathbb{Z} \arrow[r, "\cong"] & \mathbb{Z} \arrow[r] & 0
\end{tikzcd}
\end{center}
Thus it follows that $H_0(\star) \cong \mathbb{Z}$ and $H_i(\star) = (0)$ for all $i > 0$.
\end{proof}

\begin{example}{}{}
    Let $\{X_j : j \in J\}$ be the set of path connected components of $X$ so that $X = \bigsqcup_{j \in J} X_j$. Then \[
    H_{*}\left( X \right)= \bigoplus _{j \in J}H_{*}\left( X_{j} \right)  
    \]
\end{example}
\begin{remark}
    Thus in practice, we can often assume that the space itself is path connected, while dealing with its homology groups.
\end{remark}
\begin{proof}
 Since $|\Delta_n|$ is a path connected space, it follows that any singular $n$-simplex in $X$ is contained completely in one of the path components $X_j$. Therefore, it follows that the entire chain complex $S_*(X)$ is the direct sum of the subchain complexes $S_*(X_j)$. Hence the homology is also a direct sum of $H_*(X_j), j \in J$ by Theorem \ref{thm:7-9}. 
\end{proof}

\begin{theorem}{Homology Long Exact Sequence of the pair \(  \left( X,A \right)   \) }{}
    By definition of $S_.(X,A)$, we have an exact sequence of chain complexes
\[
0 \to S_.(A) \to S_.(X) \to S_.(X, A) \to 0.
\]By Theorem \ref{thm:7-9-2}, this yields the long exact sequence of homology groups
\begin{gather*}
    \left\{
    \begin{gathered}
        \dots \to H_i(A) \to H_i(X) \to H_i(X,A) \xrightarrow{\delta} H_{i-1}(A) \to \dots \\
        \dots \to H_1(X,A) \xrightarrow{\delta} H_0(A) \to H_0(X) \to H_0(X,A),
    \end{gathered}
    \right.
    \tag{4.6}
\end{gather*}
which is functorial.
\end{theorem}
\begin{remark}
     This sequence comes handy in many computations as we shall see soon.
\end{remark}

\section{Homology Excision and Mayer-Vietoris}

\begin{lemma}{}{}
Let $X = A \cup B$. Then the following statements are equivalent.
\begin{enumerate}
    \item $S_.(A) + S_.(B) \to S_.(X)$ induces isomorphisms in homology.
    \item $[S_.(A) + S_.(B)]/S_.(B) \to S_.(X)/S_.(B)$ induces isomorphisms in homology.
    \item $[S_.(A) + S_.(B)]/S_.(A) \to S_.(X)/S_.(A)$ induces isomorphisms in homology.
    \item $S_.(A)/S_.(A \cap B) \to S_.(X)/S_.(B)$ induces isomorphisms in homology.
    \item $S_.(B)/S_.(A \cap B) \to S_.(X)/S_.(A)$ induces isomorphisms in homology.
\end{enumerate}
\end{lemma}
\begin{note}
    1,2等价的好处就在于, 它告诉我们, 我们所需要的同调群的同构对于一部分的信息是不敏感的. 
\end{note}
\begin{proofsketch}
    对于1,2的等价性,涉及四种同调群,我们借助 \(  B  \)的同调将它们相互联系.  用短正合列到长正合列的诱导行为将 四种同调群两两联系, 并以 \(  B  \)的同调为桥梁, 利用诱导行为的函子性将两组同调群联系起来.  \(  B  \)到自身的恒等给出一组同构, 配合已知的一组同构可以利用Five lemma 夹出另一组同构.

    2,4的等价性直接就是第二同构定理的结果, 本质上与同调关系不大.
\end{proofsketch}
\begin{proof}
The equivalence of (1) and (2) is a consequence of the Five lemma applied to the ladder of homology long exact sequences given by the ladder of short exact sequences
\begin{center}
\begin{tikzcd}
0 \arrow[r] & S.(B) \arrow[r] \arrow[d] & S.(A) + S.(B) \arrow[r] \arrow[d] & {[S.(A) + S.(B)]/S.(B)} \arrow[r] \arrow[d] & 0 \\
0 \arrow[r] & S.(B) \arrow[r] & S.(X) \arrow[r] & S.(X)/S.(B) \arrow[r] & 0.
\end{tikzcd}
\end{center}

The equivalence of (2) and (4) follows from the commutative diagram below in which the horizontal arrow is the isomorphism given by the Noether's isomorphism theorem.
\begin{center}
\begin{tikzcd}[/tikz/commutative diagrams/diagrams={nodes={text depth=0.25ex, text height=1.75ex}}]
  S.(A)/S.(A \cap B) & {[S.(A) + S.(B)]/S.(B)} \\
  & S.(X)/S.(B)
  \arrow[from=1-1, to=1-2]
  \arrow[from=1-1, to=2-2]
  \arrow[from=1-2, to=2-2]
\end{tikzcd}
\end{center}


\end{proof}

\begin{definition}{}{}
Let $X$ be a topological space, $A, B$ be two subspaces such that $X = A \cup B$. We then say that the inclusion map $(A, A \cap B) \hookrightarrow (X, B)$ an excision map. A pair $\{A, B\}$ of subspaces of a space $X$ is called an \dfntxt{excisive couple} for singular homology, if the inclusion map
$$ S_.(A) + S_.(B) \hookrightarrow S_.(A \cup B) $$
induces isomorphisms in homology or if any of the other four equivalent conditions in the above lemma is satisfied.
\end{definition}
\begin{note}
    excisive couple是一对能代表\(  \left( X,A \right)   \)和 \(  \left( X,B \right)   \)的  同调性质的子集 \(  \left( A,B \right)   \) ,是 \(  X  \)的一个良好分解.  它 使得全空间\(  X  \) 在切除\(  X\setminus A  \) 或 \(  X\setminus B  \) 的切除行为上, 在同调的意义下是无损的.
\end{note}

\begin{theorem}{Homology excision}{}
The inclusion map $(A, A \cap B) \hookrightarrow (A \cup B, B)$ of an excisive couple $\{A, B\}$ induces an isomorphism of homology groups $H_*(A, A \cap B) \to H_*(A \cup B, B)$.
\end{theorem}

\begin{theorem}{}{}
If $X = X_1 \cup X_2 = \text{int}_X(X_1) \cup \text{int}_X(X_2)$ then, $\{X_1, X_2\}$ is an excisive couple for the singular homology.
\end{theorem}
\begin{note}
    也就是说, 当分解的一个组件的边界能被另一个组件的内部“吃掉”时, 这组分解就是切除对. 
\end{note}
\begin{example}{}{}
Here are a few examples of excisive couples which occur in practice.
\begin{enumerate}
\item $\{\mathbb{D}^n, \mathbb{R}^n \setminus \{0\}\}$ is an excisive couple and hence the inclusion map $(\mathbb{D}^n, \mathbb{D}^n \setminus \{0\}) \hookrightarrow (\mathbb{R}^n, \mathbb{R}^n \setminus \{0\})$ will induce isomorphisms in all singular homology groups.
\item Let $N$ denote the north pole on $\mathbb{S}^n$ and $\bar{U}$ denote the closed upper hemisphere. Then $\{\mathbb{S}^n \setminus \{N\}, \bar{U}\}$ is an excisive couple, which gives isomorphisms of $H_*(\bar{U}, \bar{U} \setminus \{N\}) \to H_*(\mathbb{S}^n, \mathbb{S}^n \setminus \{N\})$.
\end{enumerate}
\end{example}

\begin{theorem}{}{}
    Given subsets $X_1$ and $X_2$ of a topological space $X$, let us denote the inclusion maps of $X_i$ in to $X$ by $\eta_i, i = 1, 2$. They induce inclusion maps of the chain complexes which we shall denote again by $\eta_i: S.(X_i) \to S.(X), i = 1, 2$. Now consider the chain map
\[
(\eta_1, -\eta_2): S.(X_1) \oplus S.(X_2) \to S.(X).
\] The submodule generated by $S.(X_1)$ and $S.(X_2)$ in $S.(X)$ is denoted by $S.(X_1) + S.(X_2)$ . Then we have a short exact sequence of chain complexes
\[
0 \to S.(X_1 \cap X_2) \to S.(X_1) \oplus S.(X_2) \to S.(X_1) + S.(X_2) \to 0 
\]If $\{X_1, X_2\}$ is an excisive couple , then we can obtain the \dfntxt{Mayer-Vietoris sequence}:
\[
\left.
\begin{array}{l}
\cdots \to H_{i+1}(X_1 \cup X_2) \xrightarrow{\partial} H_i(X_1 \cap X_2) \xrightarrow{i_*} H_i(X_1) \oplus H_i(X_2) \xrightarrow{j_*} H_i(X_1 \cup X_2) \to \cdots \\
\cdots \to H_1(X_1 \cup X_2) \xrightarrow{\partial} H_0(X_1 \cap X_2) \to H_0(X_1) \oplus H_0(X_2) \to H_0(X_1 \cup X_2)
\end{array}
\right\}
\]
with
\[
i_*(z) = (i_{1*}z, -i_{2*}z); \quad j_*(z_1, z_2) = j_{1*}z_1 + j_{2*}z_2 
\]
where,
\begin{align*}
i_1 &: X_1 \cap X_2 \hookrightarrow X_1, \quad i_2 : X_1 \cap X_2 \hookrightarrow X_2 \\
j_1 &: X_1 \hookrightarrow X_1 \cup X_2, \quad j_2 : X_2 \hookrightarrow X_1 \cup X_2
\end{align*}
are inclusion maps.
\end{theorem}
\begin{remark}
     The Mayer-Vietoris sequence is one of the major ready-to-use-tools in computing homology.
\end{remark}
\begin{proof}
    The kernal of \(  \left( \eta _1 ,-\eta _2  \right)   \) is \(  S.\left( X_1 \right)\cap S.\left( X_2 \right)= S.\left( X_1\cap X_2 \right)     \)  . The image is clearly \(  S.\left( X_1 \right)+ S.\left( X_2 \right)    \). Thus the short sequence is exact. Then we obtain the long exact sequence \[
\left.
\begin{array}{l}
\cdots \to H_{i+1}( S.(X_1) + S.(X_2) ) \xrightarrow{\partial} H_i(X_1 \cap X_2) \xrightarrow{i_*} H_i(X_1) \oplus H_i(X_2) \xrightarrow{j_*} H_i( S.(X_1) + S.(X_2) ) \to \cdots \\
\cdots \to H_1( S.(X_1) + S.(X_2) ) \xrightarrow{\partial} H_0(X_1 \cap X_2) \to H_0(X_1) \oplus H_0(X_2) \to H_0( S.(X_1) + S.(X_2) )
\end{array}
\right\}
\]
By the definition of excisive couple , we can replace \(   H_i( S.(X_1) + S.(X_2) )   \) by \(  H_{i}\left( X_1\cup X_2 \right)   \) to obtain Mayer-Vietoris.  
\end{proof}

\section{Problems}
\begin{problem}{}{}
Compute the homology of $\mathbb{S}^p \times \mathbb{S}^q$ using Mayer-Vietoris.
\end{problem}
\begin{proof}
   We assume that \( p, q\ge 1  \).   Let \(  U^{+ }  \), \(  U^{-}  \) be  \(  \mathbb{S}^{q}  \) without the north pole and south pole  respectivily.  Then \( \left( A,B \right): =   \left( \mathbb{S}^{p}\times U^{+ },\mathbb{S}^{p}\times U^{-} \right)   \) is a excisive couple of \(  \mathbb{S}^{p}\times \mathbb{S}^{q}  \). Since \(  U^{+ }\cap U^{-}  \)        can deformation retract to the equator of \(  \mathbb{S}^{q}  \), which is \(  \mathbb{S}^{q-1}  \), then  \(  A\cap B  \) can deformation retract to  \(  \mathbb{S}^{p}\times \mathbb{S}^{q-1}  \).    

   \[\begin{array}{l}
\cdots \xrightarrow{i^{\prime} _{*}}  H_{i+ 1}\left( \mathbb{S}^{p} \right)\oplus H_{i+ 1}\left( \mathbb{S}^{q} \right) \xrightarrow{j_*} H_{i+1}(\mathbb{S}^{p}\times \mathbb{S}^{q}) \xrightarrow{\partial} H_i(\mathbb{S}^{p}\times \mathbb{S}^{q-1}) \xrightarrow{i_*} H_i(\mathbb{S}^{p}) \oplus H_i(\mathbb{S}^{q})  \to \cdots 
\end{array}
\] THe following short exact sequence seperates \[
0\to \operatorname{ker} \partial \xrightarrow{\iota }H_{i+ 1}\left( \mathbb{S}^{p}\times \mathbb{S}^{q} \right) \xrightarrow{ \partial } \operatorname{Im} \partial \to 0
