\section{ Vector Fields on Manifold}
\begin{definition}{}{}
    If $M$ is a smooth manifold with or without boundary, a \dfntxt{vector field on $M$} is a section of the map $\pi : TM \to M$. More concretely, a vector field is a continuous map $X : M \to TM$, usually written $p \mapsto X_p$, with the property that
\begin{equation}
\pi \circ X = \text{Id}_M,
\end{equation}
or equivalently, $X_p \in T_p M$ for each $p \in M$.
\end{definition}

\begin{remark}
   \begin{itemize}
    \item For a  ``vector field'' that may be not contiuous, we call it \dfntxt{ rough vector field on \(  M  \) }.
    \item  Under the canonical smooth structure of \(  M  \) and \(  TM  \) , we can define  the smoothness of \(  X  \). 
   \end{itemize}
   
\end{remark}
\begin{definition}{component functions of a vector field}{}
    Suppose $M$ is a smooth $n$-manifold (with or without boundary). If $X: M \to TM$ is a rough vector field and $(U, (x^i))$ is any smooth coordinate chart for $M$, we can write the value of $X$ at any point $p \in U$ in terms of the coordinate basis vectors:
\begin{equation}
X_p = X^i(p) \frac{\partial}{\partial x^i}\bigg|_p .
\end{equation}
This defines $n$ functions $X^i: U \to \mathbb{R}$, called the \dfntxt{component functions of  \(  X  \) } in the given chart.
\end{definition}

\begin{proposition}{Smoothness Criterion for Vector Fields}{}
Let $M$ be a smooth manifold with or without boundary, and let $X: M \to TM$ be a rough vector field. If $(U, (x^i))$ is any smooth coordinate chart on $M$, then the restriction of $X$ to $U$ is smooth if and only if its component functions with respect to this chart are smooth.
\end{proposition}

\begin{proposition}{}{}
Let $M$ be a smooth manifold with or without boundary, and let $X : M \to TM$ be a rough vector field. The following are equivalent:
\begin{enumerate}
\item[(a)] $X$ is smooth.
\item[(b)] For every $f \in C^\infty(M)$, the function $Xf$ is smooth on $M$.
\item[(c)] For every open subset $U \subseteq M$ and every $f \in C^\infty(U)$, the function $Xf$ is smooth on $U$.
\end{enumerate}
\end{proposition}
\begin{definition}{Vector Field as a Derivation}{}
    A smooth vector field \(  X \in \mathfrak{X}\left( M \right)   \) defines a map  \[
    \begin{aligned}
    X:C^{\infty}\left( M \right) & \to C^{\infty}\left( M \right)\\ 
     f & \mapsto  Xf   
    \end{aligned}
    \] which is linear over \(  \mathbb{R}   \) and following the product rule for vector fields \[
    X\left( fg \right)= f Xg+ gXf 
    \] We call this map a \dfntxt{Derivation on \(  C^{\infty}\left( M \right)   \) }.    
\end{definition}

\begin{proposition}{}{}
Let $M$ be a smooth manifold with or without boundary. A map $D: C^{\infty}(M) \to C^{\infty}(M)$ is a derivation if and only if it is of the form $Df = Xf$ for some smooth vector field $X \in \mathfrak{X}(M)$.
\end{proposition}

\begin{proposition}{}{}
Suppose \(M\) and \(N\) are smooth manifolds with or without boundary, and \(F:M\to N\) is a diffeomorphism. For every \(X\in\mathfrak{X}(M)\), there is a unique smooth vector field on \(N\) that is \(F\)-related to \(X\).
\end{proposition}

\section{Vector Field and Smooth Maps}

\begin{definition}{\(  F  \) -related vector fields}{}
    Suppose $F:M\to N$ is smooth and $X$ is a vector field on $M$, and suppose there happens to be a vector field $Y$ on $N$ with the property that for each $p\in M$, $dF_{p}(X_{p})=Y_{F(p)}$. In this case, we say the vector fields $X$ and $Y$ are \dfntxt{$F$-related}.
\end{definition}

\begin{remark}
    这是\(  F  \)-相关向量场的逐点定义. 
\end{remark}

\begin{proposition}{}{}
    Suppose \(  F:M\to N  \) is a smooth map between manifolds with or without boundary, \(  X \in \mathfrak{X}\left( M \right)   \) and \(  Y\in \mathfrak{X}\left( N \right)   \). Then \(  X  \) and \(  Y  \) are \(  F  \)-related if and only if for every smooth real-vealued function \(  f  \) defined on an open subset of \(  N  \), \[
    X\left( f\circ F \right)= \left( Yf \right)\circ F  
    \]        
\end{proposition}

\begin{remark}
    \begin{itemize}
        \item 这是\(  F  \)-相关向量场的\(  C_{F\left( p \right) }^{\infty}\left( M \right)   \)-导子刻画.    
        \item \(  F  \)参与了以下活动 : 将 函数芽\(  f \in C_{F\left( p \right) }^{\infty}\left( N \right)   \) 的定义域拉回到 \(  M  \)上以便于 \(  X  \)作用, 以及将 \(  \left( Yf \right)   \)的定义域(\(  Y  \)的取值) 拉回到 \(  M  \)上以便于比较.
    \end{itemize}
\end{remark}

\begin{proof}
    For any \(  p \in M  \) and any \(  f\in C^{\infty}_{F\left( p \right) }\left( M \right)   \)  , \[
    X\left( f\circ F \right) \left( p \right) = X_{p}\left( f\circ F \right)= \,\mathrm{d} F_{p}\left( X_{p} \right)f  
    \]while \[
    \left( Yf \right)\circ F\left( p \right)= \left( Yf \right)\left( F\left( p \right)  \right)= Y_{F\left( p \right) }f    
    \]
\end{proof}
