\chapter{ Metric Spaces }

\section{Convergence of Sequences}

\begin{definition}{Convergence}{}
Let $(X,d)$ be a metric space and $x_1,\dots,x_n,\dots$ a sequence of points in $X$. Then the sequence is said to \dfntxt{converge to} $x \in X$ if given any $\varepsilon > 0$ there exists an integer $n_0$ such that for all $n \ge n_0$, $d(x,x_n) < \varepsilon$.
This is denoted by $x_n \to x$.

The sequence $y_1,y_2,\dots,y_n,\dots$ of points in $(X,d)$ is said to be \dfntxt{convergent} if there exists a point $y \in X$ such that $y_n \to y$.
\end{definition}

\begin{theorem}{}{}
    Let \(  \left( X,d \right)   \) be a metric space. \(  A\subseteq X  \) is a subset, \(  p \in A  \).   Then \(  p  \) is a limit point of \(  A  \), if and only if there exists a  sequence \(    \left\{ a_{n} \right\}\) composed with distinct points in \(  A  \) converging to \(  p  \).  
\end{theorem}

\begin{proof}
    If \(  p  \) is a limit point of \(  A  \). Then \(  B_{d}\left( p,1 \right)\cap \left( A\setminus \left\{ p \right\} \right)\neq \varnothing    \), we choose a point \(  a_1  \) there. If we have points \(  a_1,a_2,\cdots ,a_{n}  \) such that  \(  a_{i}\in B_{d}\left( p,\frac{1 }{i }  \right)\cap B_{d}\left( p, m_{i} \right)    \), where \[
    m_{i}\coloneqq \min \left\{ d\left( p, a_{j} \right)  \right\}_{1\le j\le i}
    \]   Since the sets \(  \left( B_{d}\left( p,\frac{1 }{n+ 1 }  \right)\cap B_{d}\left( p, m_{n} \right)   \right)\cap \left( A\setminus \left\{ p \right\} \right)    \) is nonempty, we can take a point in it.  We inductively give a seuquence \(  \left\{ a_{n} \right\}  \) composed with distinc points in \(  A  \).
    
    If \(  \left\{ a_{n} \right\}  \) is such sequence. Let  \(  U  \) be any open set containg \(  p  \). Since \(  \left\{ B_{d}\left( x,  \varepsilon  \right): x \in X,  \varepsilon > 0  \right\}  \) is a basis for \(  \left( X,d \right)   \). We have from  Proposition \ref{pps:9-6-5} that there exists a ball \(  B  \) such that \(  p \in B\subseteq U  \). Then it is easy to verify that there exists \(  B_{d}\left( p, \varepsilon  \right)   \) such that \(   p \in B_{d}\left( p, \varepsilon  \right)\subseteq B   \). Since the sequence  \(  \left\{ a_{n} \right\}  \) in \(  A  \)  converges to \(  p  \), there      exists some (in fact, infinitely many)  \(  a_{n} \in B_{d}\left( p, \varepsilon  \right)   \) that is not equal to \(  p  \).  i.e, \(  p \in B_{d}\left( p, \varepsilon  \right)\cap \left( A\setminus \left\{ p \right\} \right)\subseteq U\cap \left( A\setminus \left\{ p \right\} \right)    \neq \varnothing \). 
\end{proof}

\begin{proposition}{}{}
Let $(X, d)$ be a metric space. A subset $A$ of $X$ is closed in $(X, d)$ if and only if every convergent sequence of points in $A$ converges to a point in $A$. (In other words, $A$ is closed in $(X, d)$ if and only if $a_n \to x$, where $x \in X$ and $a_n \in A$ for all $n$, implies $x \in A$.)
\end{proposition}

\section{Completeness}

\begin{definition}{Cauchy sequence}{}
A sequence $x_1, x_2, \dots, x_n, \dots$ of points in a metric space $(X, d)$ is said to be a \dfntxt{Cauchy sequence} if given any real number $\varepsilon > 0$, there exists a positive integer $n_0$, such that for all integers $m \ge n_0$ and $n \ge n_0$, $d(x_m, x_n) < \varepsilon$.
\end{definition}
\begin{proposition}{}{}
Let $(X,d)$ be a metric space and $x_1,x_2,\dots,x_n,\dots$ a sequence of points in $(X,d)$. If there exists a point $a \in X$, such that the sequence converges to $a$, that is, $x_n \to a$, then the sequence is a Cauchy sequence.
\end{proposition}

\begin{definition}{}{}
A metric space $(X,d)$ is said to be \dfntxt{complete} if every Cauchy sequence in $(X,d)$ converges to a point in $(X,d)$.
\end{definition}

\begin{proposition}{}{}
Let $(X,d)$ be a metric space, $Y$ a subset of $X$, and $d_1$ the metric induced on $Y$ by $d$.
\begin{enumerate}
    \item[(i)] If $(X,d)$ is a complete metric space and $Y$ is a closed subspace of $(X,d)$, then $(Y,d_1)$ is a complete metric space.
    \item[(ii)] If $(Y,d_1)$ is a complete metric space, then $Y$ is a closed subspace of $(X,d)$.
\end{enumerate}
\end{proposition}


\section{The Metric Topology}
\begin{theorem}{standard bounded metric}{}
Let $X$ be a metric space with metric $d$. Define $\bar{d}: X \times X \to \mathbb{R}$ by the equation

\[
\bar{d}(x, y) = \min\{d(x, y), 1\}.
\]

Then $\bar{d}$ is a metric that induces the same topology as $d$.
\end{theorem}

\begin{remark}
    
The metric $\bar{d}$ is called the \dfntxt{standard bounded metric} corresponding to $d$.
\end{remark}
\begin{proof}
    In any metric space, the collection of \(   \varepsilon   \)-balls with \(   \varepsilon < 1  \) forms a basis for the metri topology. \(  d  \) and \(  \bar{d}  \) induce the same topology on \(  X  \) since there \(   \varepsilon   \)- balls with \(   \varepsilon < 1  \) under these two metrics are the same collection.       
\end{proof}

\begin{theorem}{}{}
Let $\bar{d}(a,b) = \min\{|a-b|, 1\}$ be the standard bounded metric on $\mathbb{R}$. If $\mathbf{x}$ and $\mathbf{y}$ are two points of $\mathbb{R}^{\omega}$, define

\[
D(\mathbf{x}, \mathbf{y}) = \sup\left\{\frac{\bar{d}(x_i, y_i)}{i}\right\}.
\]

Then $D$ is a metric that induces the product topology on $\mathbb{R}^{\omega}$.
\end{theorem}


\begin{definition}{}{}
Given an index set $J$, and given points $\mathbf{x} = (x_\alpha)_{\alpha \in J}$ and $\mathbf{y} = (y_\alpha)_{\alpha \in J}$ of $\mathbb{R}^J$, let us define a metric $\bar{\rho}$ on $\mathbb{R}^J$ by the equation

\[
\bar{\rho}(\mathbf{x}, \mathbf{y}) = \sup\{\bar{d}(x_\alpha, y_\alpha) \mid \alpha \in J\},
\]

where $\bar{d}$ is the standard bounded metric on $\mathbb{R}$. It is easy to check that $\bar{\rho}$ is indeed a metric; it is called the \dfntxt{uniform metric} on $\mathbb{R}^J$, and the topology it induces is called the \dfntxt{uniform topology}.
\end{definition}

\begin{theorem}{}{}
The uniform topology on $\mathbb{R}^J$ is finer than the product topology and coarser than the box topology; these three topologies are all different if $J$ is infinite.
\end{theorem}
