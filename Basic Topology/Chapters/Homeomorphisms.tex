\chapter{ Homeomorphisms }

\section{Subcspaces}

\begin{definition}{}{}
Let $Y$ be a non-empty subset of a topological space $(X, \mathcal{T})$. The collection $\mathcal{T}_Y = \{O \cap Y : O \in \mathcal{T}\}$ of subsets of $Y$ is a topology on $Y$ called the \dfntxt{subspace topology} (or the \dfntxt{relative topology} or the \dfntxt{induced topology} or the \dfntxt{topology induced on $Y$ by $\mathcal{T}$}). The topological space $(Y, \mathcal{T}_Y)$ is said to be a \dfntxt{subspace} of $(X, \mathcal{T})$.
\end{definition}
