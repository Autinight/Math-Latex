\section{The Concept of Measurbility}

\begin{definition}{}{}
    \begin{enumerate}
        \item A collection \(  \mathfrak{M}  \) of subsets of a set \(  X  \) is said to be a \dfntxt{\(   \sigma   \)-algebra} in \(  X  \) if \(  \mathfrak{M}  \) has the following properties: \begin{enumerate}
            \item \(  X \in \mathfrak{M}  \). 
            \item If \(  A \in \mathfrak{M}  \)  , then \(  A^{c} \in  \mathfrak{M}  \).
            \item If \(  A =  \bigcup _{n = 1}^{\infty} A_{n}  \) and \(  A_{n} \in  \mathfrak{M}  \) for \(  n = 1,2,3,\cdots   \) , then \(  A \in \mathfrak{M}  \).      
        \end{enumerate}
        \item If \(  \mathfrak{M}  \) is a \(   \sigma   \)-algebra in \(  X  \), then \(  X  \) is called a \dfntxt{measurable} provided that \(  f^{-1} \left( V \right)   \) is measurable set in \(  X  \) for every open set \(  V  \) in \(  Y  \).         
        \item If \(  X  \) is a measurable space, \(  Y  \) is a topological space, and \(  f   \) is a mapping of \(  X  \) into Y, then \(  f  \) is said to be \dfntxt{measurable} provided that \(  f^{-1} \left( V \right)   \) is a measurable set in \(  X  \) for every open set \(  V  \) in \(  Y  \).          
             
    \end{enumerate}
    
\end{definition}

\begin{note}
    对比topological space的定义, \(   \sigma   \)-algebra 的定义是对称的, 即可测集的补集仍是可测的, 而对于拓扑空间, 开集的补集不是开集, 而是被定义为了闭集这样的对象.
\end{note}

\begin{lemma}{}{7-22-1}
    \begin{enumerate}
        \item Let  \(  X  \) be a measurable space , \(  Y ,Z  \) be topological spaces. If \(  f:X\to Y  \) is measurable , and if  \(  g:Y\to Z  \) is continuous, then     \(  g\circ f :X\to Z  \) is measurable. 
        \item Let \(  u,v  \) be real measurable functions on a measurable space \(  X  \), let \(  \Phi   \) be a continuous mapping of the plane into a topological space \(  Y  \), and define \[
        h\left( x \right)=  \Phi \left( u\left( x \right),v\left( x \right)   \right)  
        \]for  \(  x \in X  \). Then  \(  h: X\to Y  \) is measurable.       
    \end{enumerate}
    
\end{lemma}

\begin{proof}
    \begin{enumerate}
        \item Take any open subset \(  V\subseteq Z  \) .  Then \(  g^{-1} \left( V \right)    \)  is an open subset of \(  Y  \) . Since \(  f  \) is measurable,  \[
        \left( g\circ f \right)^{-1} \left( V \right)=  f^{-1} \left( g^{-1} \left( V \right)  \right)   
        \] is a mearuable set. 
        \item    Let \(  V\subseteq Y   \) be a open subset.  Then \(  \Phi ^{-1} \left( V \right)   \) is a open subset of  \(  \left[ -\infty,\infty \right]^{2}   \) . There exists countably many open sets of \(  \left[ -\infty,\infty \right]   \)   :\(  U_{i},V_{i}, i \in \mathbb{N}   \) , such that  \[
        \Phi ^{-1} \left( V \right)= \bigcup _{i}U_{i}\times V_{i} 
        \]   Thus \[
        \begin{aligned}
        h^{-1} \left( V \right)&= \left( u,v \right)^{-1} \left( \bigcup _{i}U_{i}\times V_{i} \right)\\ 
         &=  \bigcup _{i} \left( u,v \right)^{-1} \left( U_{i}\times V_{i} \right)\\ 
          &= \bigcup _{i}\left( u^{-1} \left( U_{i} \right)\cap v^{-1} \left( V_{i} \right)   \right)    
        \end{aligned}  
        \] which is a measurable set.
    \end{enumerate}
    
\end{proof}
\begin{corollary}{}{}
    Let \(  X  \) be a measurable space. The following propositions hold
    \begin{enumerate}
        \item If \(  f =  u+ iv  \), where \(  u  \) and \(  v   \) are real measurable functions on \(  X  \), then \(  f   \) is a  complex measurable function on \(  X  \). 
        \item If \(  f= u+ iv  \) is a complex measurable function on \(  X  \), then \(  u,v  \), and \(  \left| f \right|   \) are real measurable functions on \(  X  \). 
        \item If \(  f  \) and \(  g  \) are complex measurable functions on \(  X  \), then so are \(  f+ g  \) and \(  fg  \).
        \item If \(  E  \) is a measurable set in \(  X  \) and if \[
        \chi _{E}\left( x \right)= \begin{cases} 1& x\in E\\ 
         0&x\not\in  E \end{cases}  
        \] then \(  \chi _{E}  \) is a measurable function.
        \item If \(  f  \) is a complex measurable function on \(  X  \) , there is a complex measurable function \(  \alpha   \) on \(  X  \) such that \(  \left| \alpha  \right|=  1   \)                         and \(  f= \alpha \left| f \right|   \) .  
    \end{enumerate}
     
    
\end{corollary}

\begin{theorem}{}{}
    If \(  \mathscr{F}  \) is any collection of  subsets of \(  X  \) , there exists a smallest \(   \sigma   \)-algebra \(  \mathfrak{M}^{*}  \) such that \(  \mathscr{F}\subseteq  \mathfrak{M}^{*}  \).     
\end{theorem}

\begin{definition}{Borel}{}
    Let \(  X  \) be a topological space. 
    \begin{enumerate}
        \item By \dfntxt{Borel \(   \sigma   \)-algebra} , we mean the smallest \(   \sigma   \)-algebra \(  \mathscr{B}  \) in \(  X  \) such that every open set in \(  X  \) belongs to \(  \mathscr{B}  \).   The members of \(  \mathscr{B}  \) are called the \dfntxt{Borel sets} of \(  X  \).  
        \item All countable unions of closed sets and all countable intersections of open sets are Borel sets, which we called \dfntxt{\(  F_{ \sigma } \)'s  and \(  G_{ \delta }  \)'s}, respectively. 
        \item By a \dfntxt{Borel function}, we mean a measurable function on the measurable space \(  \left( X,\mathscr{B} \right)   \) .
    \end{enumerate}
         
\end{definition}
\begin{remark}
   \begin{enumerate}
    \item  The letters \(  F  \) and \(  G  \) were used for colsed and opensets,respectively, and \(   \sigma   \) refers to union, \(   \delta   \) to intersection. 
    \item A continuous function is Borel-measurable, since the preimage of any open set is open and therefore a Borel set.
   \end{enumerate}
      
\end{remark}

\begin{theorem}{}{}
    Suppose \(  \mathfrak{M}  \) is a \(   \sigma   \)-algbera in \(  X  \) ,and \(  Y  \) is a topological space. Let \(  f   \) map \(  X  \) into \(  Y  \).
    \begin{enumerate}
        \item If \(   \Omega   \) is the collection of all sets \(  E\subseteq Y  \) such that \(  f^{-1} \left( E \right)\in  \mathfrak{M}   \), then \(   \Omega   \) is a \(   \sigma   \)- algebra in \(  Y  \). 
        \item If \(  f  \) is measurable and \(  E  \) is a Borel set in \(  Y  \) , then \(  f^{-1} \left( E \right) \subseteq \mathfrak{M}   \). 
        \item If \(  Y= [-\infty,\infty]  \) and \(  f^{-1} \left( (\alpha ,\infty] \right)\subseteq  \mathfrak{M}   \) for every real \(  \alpha   \), then \(  f   \)  is measurable . 
        \item If \(  f  \) is measurable, if \(  Z  \) is topological space, if \(  g:Y\to Z  \) is a Borel mapping, and if \(  h= g\circ f  \), then \(  h:X\to Z  \) is measurable.                   
    \end{enumerate}
           
\end{theorem}


\begin{theorem}{}{7-5-1}
    If \(  f_{n}:X\to [-\infty,\infty]  \) is measurable,for \(  \neq 1,2,3,\cdots   \) , and \[
    g =  \sup _{n\ge 1}f_{n},\quad h= \lim_{n\to \infty}\sup f_{n},
    \] then \(  g  \) and \(  h  \) are measurable.    
\end{theorem}

\begin{corollary}{}{}
    \begin{enumerate}
        \item The limit of every pointwise convergent sequence of complex measurable functions is measurable.
        \item If \(  f  \) and \(  g  \) are measurable(with range in \(  [-\infty,\infty]  \) ), then so are \(  \max \left\{ f,g \right\}  \) and \(  \min \left\{ f,g \right\}  \). In particular , this is true of the functions \[
        f^{+ }= \max \left\{ f,0 \right\},\quad f^{-}= -\min \left\{ f,0 \right\}.
        \]    which are called the \dfntxt{positive part} and \dfntxt{negative part} of \(  f  \) , respectively. 
    \end{enumerate}
    
\end{corollary}
\begin{remark}
   \begin{enumerate}
    \item  There are the standrd representation \[
    \left| f \right|= f^{+ }+ f^{-},\quad  f= f^{+ }-f^{-} 
    \]
    \item And easy but (may) useful observation is : If \(  f= g-h, g \ge 0, h \ge 0  \), then \(  f^{+ }\le g  \) and \(  f^{-} \le h  \).   
   \end{enumerate}
   
\end{remark}

\section{Simple Functions}

\begin{definition}{}{}
    A complex function \(  s  \) on a measurable space \(  X  \) whose range consists of only finitely many points will be called a \dfntxt{simple function}. Among these are the nonnegative simple functions, whose range is a finite subset of \(  [0,\infty)  \).    

   Specifically, if \(   \alpha_1,\cdots,\alpha_n   \) are distinct values of a simple function \(  s  \), and if we set \(  A_{i}= \left\{ x:s\left( x \right)= \alpha _{i}  \right\}  \) , then clearly \[
    s= \sum _{i= 1}^{n}\alpha _{i}\chi _{A_{i}}.
    \]Where \(  \chi _{A_{i}}  \) is the characteristic function of \(  A_{i}  \).      
\end{definition}
\begin{remark}
  \begin{itemize}
    \item   Here, we explicity exclude \(  \infty  \) from the values of a simple function. 
    \item It is clear that \(  s  \) is mearuable if and only if each of the sets \(  A_{i}  \) is measurable.  
  \end{itemize}
  
\end{remark}

\begin{theorem}{}{7.19}
    Let \(  f: X\to [0,\infty]  \) be measurable space. There exists simple measurable functions \(  s_{n}  \) on \(  X  \) such that 
    \begin{enumerate}
        \item \(  0\le s_1\le s_2\le \cdots \le f  \). 
        \item \(  s_{n}\left( x \right)\to f\left( x \right)    \) as \(  n\to \infty  \) , for every \(  x \in X  \).     
    \end{enumerate}
       
\end{theorem}

\begin{proofsketch}
    We construst a sequence of Borel  simple functions \( \left\{  \varphi _{n}\left( x \right) \right\}    \) to act as an \dfntxt{identify}. 当 \(  n  \) 增大的同时, 我们同时让 \(   \varphi _{n}\left( x \right)   \)的单位逼近精度和单位逼近范围随着 \(  n  \)提升.并且在舍弃误差时,总是向下取整,使得该单位逼近是自下而上的.   
\end{proofsketch}

\begin{proof}
    For every \(  x \in [0,\infty]  \), and for every \(  n\in \mathbb{N}   \), there exists a unique integer \(  k_{n}\left( x \right)   \), such that  \[
    k_{n}\left( x \right)2^{-n}\le x<  \left( k_{n}\left( x \right)+ 1  \right)2^{-n}  
    \]For every \(  n\in \mathbb{N}   \), we define
    \[
     \varphi _{n}\left( x \right)= \begin{cases}  k_{n}\left( x \right) 2^{-n},  &0\le x\le n\\ 
      n,& x\ge n\end{cases}  
    \]Each \(   \varphi _{n}\left( x \right)   \) is then a Borel simple function. It is not hard to show that \(  0\le  \varphi _1 \le  \varphi _2 \cdots \le \cdots  \le  \operatorname{Id}\). For each \(  n   \), we define \[
    s_{n}\left( x \right): = \left(  \varphi _{n}\circ f \right)\left( x \right)   
    \]   Then \( \left\{  s_{n}   \right\} \) is a suquence of simple measurable functions such that \(  0\le s_1\le s_2\le \cdots \le f  \) .  Since \(  \lim_{n\to \infty} \varphi _{n}\left( x \right)= x   \), then  \(  \lim_{n\to \infty}s_{n}\left( x \right)= f\left( x \right)    \).  
\end{proof}

\section{Measure}
\begin{definition}{Measure and Measure Space}{}
 \begin{enumerate}
    \item[(a)] A \dfntxt{positive measure} is a function $\mu$, defined on a $\sigma$-algebra $\mathfrak{M}$, whose range is in $[0, \infty]$ and which is \dfntxt{countably additive}. This means that if $\{A_i\}$ is a disjoint countable collection of members of $\mathfrak{M}$, then
    \[ \mu\left(\bigcup_{i=1}^{\infty} A_{i}\right)=\sum_{i=1}^{\infty} \mu\left(A_{i}\right). \]
    To avoid trivialities, we shall also assume that $\mu(A) < \infty$ for at least one $A \in \mathfrak{M}$.
    \item[(b)] A \dfntxt{measure space} is a measurable space which has a positive measure defined on the $\sigma$-algebra of its measurable sets.
    \item[(c)] A \dfntxt{complex measure} is a complex-valued countably additive function defined on a $\sigma$-algebra.
\end{enumerate}
\end{definition}

\begin{theorem}{}{6-27-1}
     Let $\mu$ be a positive measure on a $\sigma$-algebra $\mathfrak{M}$. Then
\begin{enumerate}
    \item $\mu(\emptyset) = 0.$
    \item $\mu(A_1 \cup \dots \cup A_n) = \mu(A_1) + \dots + \mu(A_n)$ if $A_1, \dots, A_n$ are pairwise disjoint members of $\mathfrak{M}$.
    \item $A \subset B$ implies $\mu(A) \le \mu(B)$ if $A \in \mathfrak{M}, B \in \mathfrak{M}$.
    \item $\mu(A_n) \to \mu(A)$ as $n \to \infty$ if $A = \bigcup_{n=1}^\infty A_n, A_n \in \mathfrak{M}$, and
    \[ A_1 \subset A_2 \subset A_3 \subset \dots \]
    \item $\mu(A_n) \to \mu(A)$ as $n \to \infty$ if $A = \bigcap_{n=1}^\infty A_n, A_n \in \mathfrak{M}$,
    \[ A_1 \supset A_2 \supset A_3 \supset \dots, \]
    and $\mu(A_1)$ is finite.
\end{enumerate}
\end{theorem}

\begin{proof}
    \begin{enumerate}
        \item Take \(  A \in \mathfrak{M}   \) such that  \(  \mu \left( A \right)< \infty   \). \footnote{That is what we supposed at the definition of measure.}  And let \(  A_2= A_3= \cdots = \varnothing  \), then \(  \mu \left( \varnothing \right)> 0   \) leads to a contradition to the countably additive.  
        \item Take \(  A_{n+ 1}= A_{n+ 2}= \cdots = \varnothing  \).
        \item Note that \(  B=\left( B\setminus A \right) \cup A  \), then   by additivity \[
        \mu \left( B \right)= \mu \left( A \right)+ \mu \left( B\setminus A \right)\ge \mu \left( A \right)    
        \] 
        \item Let \(  A_0= \varnothing  \), and let \(  B_{n}=  A_{n}\setminus A_{n-1}  \) for all  \(  n\in \mathbb{N}   \). Then \(  B_1,\cdots ,B_{n}  \) are pairwise disjoint members of \(  \mathfrak{M}  \) such that \(  A=  \bigcup _{n = 1}^{\infty}B_{n}  \)  . We have \[
        \mu \left( A \right)=  \sum _{n = 1}^{\infty}\mu \left( B_{n} \right)= \sum _{n = 1}^{\infty}\mu \left( A_{n}\setminus A_{n-1} \right)   
        \]    If one of the \(  \mu \left( A_{n} \right)   \) is \(  \infty  \), then \(  \lim_{n\to \infty}\mu \left( A_{n} \right)   \) and \(  \mu \left( A \right)   \) both are \(  \infty  \) . Otherwise, we have \[
        \mu \left( A_{n}\setminus A_{n-1} \right)= \mu \left( A_{n} \right)-\mu \left( A_{n-1} \right),\quad \forall n   \mathbb{N} 
        \] Thus \[
      \mu \left( A \right)=    \sum _{n = 1}^{\infty}\mu \left( A_{n}\setminus A_{n-1} \right)= \lim_{n\to \infty}\mu \left( A_{n} \right)  
        \]
        \item Let \(  B_{n}=  A_{n}\setminus A_{n+ 1}  \) for all \( n \in \mathbb{N}    \).   \(  B_1,\cdots ,B_{n}  \) are pariwise disjoint members of \(  \mathfrak{M}  \) with finite measure, such that\[
        A_1\setminus A= A_1\setminus \left( \bigcap_{n = 1}^{\infty}A_{n}  \right) = \bigcup _{n = 1}^{\infty}\left( A_1\setminus A_{n} \right)= \bigcup _{n = 1}^{\infty}\left( \bigcup _{k = 1}^{n}B_{k} \right) = \bigcup _{n = 1}^{\infty}B_{n}
        \] . We have \[
       \mu  \left( A_1\setminus A \right)=  \mu \left( \bigcup _{n = 1}^{\infty}B_{n} \right) , 
        \]   where the RHS is \(  \mu \left( A_1 \right)-\mu \left( A \right)    \) , and the LHS is \[
        \sum _{n = 1}^{\infty}\left( \mu \left( A_{n} \right)-\mu \left( A_{n+ 1} \right)   \right)= \mu \left( A_1 \right)-\lim_{n\to \infty}\mu \left( A_{n+ 1} \right)    
        \]Since \(  \mu \left( A_1 \right)< \infty   \), we have \[
        \mu \left( A \right)= \lim_{n\to \infty}\mu \left( A_{n} \right)  
        \] 
    \end{enumerate}
    
\end{proof}

\begin{example}{}{}
    \begin{enumerate}
\item For any $E \subset X$, where $X$ is any set, define $\mu(E) = \infty$ if $E$ is an infinite set, and let $\mu(E)$ be the number of points in $E$ if $E$ is finite. This $\mu$ is called the \dfntxt{counting measure} on $X$.
\item Fix $x_0 \in X$, define $\mu(E) = 1$ if $x_0 \in E$ and $\mu(E) = 0$ if $x_0 \notin E$, for any $E \subset X$. This $\mu$ may be called the \dfntxt{unit mass concentrated at $x_0$}.
\item Let $\mu$ be the counting measure on the set $\{1, 2, 3, \dots\}$, let $A_n = \{n, n + 1, n + 2, \dots\}$. Then $\bigcap A_n = \emptyset$ but $\mu(A_n) = \infty$ for $n = 1, 2, 3, \dots$. This shows that the hypothesis
\[ \mu(A_1) < \infty \]
is not superfluous in Theorem \ref{thm:6-27-1}(5).
\end{enumerate}
\end{example}
\section{Integration of Positive Funtions}


\begin{definition}{Integration of Positive Funtions}{}
\begin{enumerate}
    \item If $s: X \to [0, \infty)$ is a measurable simple function, of the form
\[
s = \sum_{i=1}^{n} \alpha_i \chi_{A_i},
\]
where $\alpha_1, \dots, \alpha_n$ are the distinct values of $s$ , and if $E \in \mathfrak{M}$, we define
\[
\int_E s \, d\mu = \sum_{i=1}^{n} \alpha_i \mu(A_i \cap E).
\]
The convention $0 \cdot \infty = 0$ is used here; it may happen that $\alpha_i = 0$ for some $i$ and that $\mu(A_i \cap E) = \infty$.
\item If $f: X \to [0, \infty]$ is measurable, and $E \in \mathfrak{M}$, we define
\[
\int_E f \, d\mu = \sup \int_E s \, d\mu,
\]
the supremum being taken over all simple measurable functions $s$ such that $0 \le s \le f$.
The left member above is called the \dfntxt{Lebesgue integral} of $f$ over $E$, with respect to the measure $\mu$. It is a number in $[0, \infty]$.
\end{enumerate}

\end{definition}
\begin{remark}
    We apparently have two definitions for \(  \int_{E}f\,\mathrm{d} \mu   \) if \(  f   \) is simple, they are the same.  
\end{remark}

\begin{corollary}{}{}
    \begin{enumerate}
    \item[(a)] If $0 \le f \le g$, then $\int_E f \, d\mu \le \int_E g \, d\mu$.
    \item[(b)] If $A \subset B$ and $f \ge 0$, then $\int_A f \, d\mu \le \int_B f \, d\mu$.
    \item[(c)] If $f \ge 0$ and $c$ is a constant, $0 \le c < \infty$, then
    \[
    \int_E cf \, d\mu = c \int_E f \, d\mu.
    \]
    \item[(d)] If $f(x) = 0$ for all $x \in E$, then $\int_E f \, d\mu = 0$, even if $\mu(E) = \infty$.
    \item[(e)] If $\mu(E) = 0$, then $\int_E f \, d\mu = 0$, even if $f(x) = \infty$ for every $x \in E$.
    \item[(f)] If $f \ge 0$, then $\int_E f \, d\mu = \int_X \chi_E f \, d\mu$.
\end{enumerate}
\end{corollary}

\begin{remark}
    We can also regard the \(  \int_{E} f \,\mathrm{d} \mu   \) as the restricted function \(  f|_{E}  \)  integrates on the induced measure space  \(  E  \). 
\end{remark}


\begin{proposition}{}{7-19-2}
Let $s$ and $t$ be nonnegative measurable simple functions on $X$. For $E \in \mathfrak{M}$, define
\[
\varphi(E) = \int_E s \, d\mu.
\]
Then $\varphi$ is a measure on $\mathfrak{M}$. Also
\[
\int_X (s + t) \, d\mu = \int_X s \, d\mu + \int_X t \, d\mu.
\]
\end{proposition}


\begin{note}
    \(   \varphi \left( E \right) =  \int_{E}s\,\mathrm{d} \mu    \) 就是依据 \(  s  \) 对测度 \(  \mu   \)进行一个有限的算数的加权.    自然地也就成为一个测度 . 
\end{note}
\begin{proof}
    If  \(  E_1,E_2,\cdots   \) are disjoint members of  \(  \mathfrak{M}  \) whose union is \(  E  \), the countable additivity of \(  \mu   \) shows that  \[
    \begin{aligned}
     \varphi \left( E \right)&= \sum _{i= 1}^{n} \alpha _{i}\mu \left( A_{i}\cap E \right)\\ 
      &=  \sum _{i = 1}^{n}\sum _{j = 0}^{\infty} \alpha _{i}\mu \left( A_{i}\cap E_{j} \right) \\ 
       &= \sum _{j = 0}^{\infty}\sum _{i = 1}^{n}\alpha _{i}\mu \left( A_{i}\cap E_{j} \right)\\ 
        &= \sum _{j = 0}^{\infty} \varphi \left( E_{j} \right) 
    \end{aligned} 
    \] Also, \(   \varphi \left( \varnothing \right)\neq \infty   \), \(  \mu   \) is not identically   \(  \infty  \).  

    Next, let \(  \beta _1 ,\cdots ,\beta _{m}  \) be distinct values of \(  t  \).   And let \(  B_{j}: = \left\{ x: t\left( x \right)= \beta _{j}  \right\}  \). Then \[
    \begin{aligned}
    \int_{X}\left( s+ t \right)\,\mathrm{d} \mu &= \sum _{i,j}\int_{A_{i}\cap B_{j}}\left( s+ t \right)\,\mathrm{d} \mu \\ 
     &= \sum _{i,j}\alpha _{i}\mu \left( A_{i}\cap B_{j} \right)+ \sum _{i,j}\beta _{j}\mu \left( A_{i}\cap B_{j} \right)\\ 
      &=         \sum _{i}\alpha _{i}\mu \left( A_{i} \right)+ \sum _{j}\beta _{j}\mu \left( B_{j} \right)\\ 
       &= \int_{X}s \,\mathrm{d} \mu + \int_{X} t\,\mathrm{d} \mu   
    \end{aligned}
    \] 
\end{proof}

\begin{theorem}{Lebesgue's Monotone Convergence Theorem}{}
Let $\{f_n\}$ be a sequence of measurable functions on $X$, and suppose that
\begin{enumerate}
\item[(a)] $0 \le f_1(x) \le f_2(x) \le \cdots \le \infty$ for every $x \in X$,
\item[(b)] $f_n(x) \to f(x)$ as $n \to \infty$, for every $x \in X$.
\end{enumerate}
Then $f$ is measurable, and
\[
\int_X f_n \, d\mu \to \int_X f \, d\mu \quad \text{as } n \to \infty.
\]
\end{theorem}

\begin{proof}
    Since \(  \int_{X} f_{n}\,\mathrm{d} \mu \le \int_{X}f_{n+ 1}\,\mathrm{d} \mu   \) for all \(  n  \), there exists an \(  \alpha > 0  \) such that \[
    \int_{X}f_{n}\,\mathrm{d} \mu \to \alpha ,\quad \text{as } n \to \infty
    \]   By Theorem \ref{thm:7-5-1}, \(  f  \) is measurable,  then by \(  \int_{X}f_{n}\,\mathrm{d} \mu \le \int_{X} f\,\mathrm{d} \mu    \;\forall n\), \[
    \alpha \le \int_{X}f \,\mathrm{d} \mu 
    \]  Take any simple function  \(  0\le s\le f  \) , and \(  0\le c< 1  \) .  Define \[
    E_{n}^{s}: = \left\{ x \in X: f_{n}\left( x \right)\ge cs\left( x \right)   \right\}
    \] Then since \(   \varphi \left( E \right): =  \int_{E}s\,\mathrm{d} \mu    \)   is a measure , \[
    \alpha\ge   \lim_{n\to \infty} \int_{X} f_{n}\,\mathrm{d} \mu \ge  \lim_{n\to \infty} \int_{E_{n}^{s}}f_{n}\,\mathrm{d} \mu \ge  c\lim_{n\to \infty}\int_{E_{n}^{s}}s\,\mathrm{d} \mu = c \lim_{n\to \infty} \varphi \left( E_{n}^{s} \right)=  c  \varphi \left( X \right)  \] That is \[
    \alpha \ge c \int_{X}s \,\mathrm{d} \mu 
    \]Since \(  s  \) is arbitrary, \[
    \alpha \ge c \int_{X} f\,\mathrm{d} \mu 
    \]  and since \(  c  \) is arbitrary , \[
    \alpha \ge  \int_{X} f\,\mathrm{d} \mu 
    \]  which completes the proof. 
\end{proof}


\begin{corollary}{}{}
    Let \(  f,g  :X\to \left[ 0,\infty \right] \) be mearuable functions on \(  X  \) .   Then \(  f+ g  \) is mearuable and   \[
    \int_{X}\left( f+ g \right)\,\mathrm{d} \mu = \int_{X}f\,\mathrm{d} \mu + \int_{X} g\,\mathrm{d} \mu  
    \]
\end{corollary}
\begin{proof}
    Let \(  \left\{ s_{n} \right\} ,\left\{ s_{n}^{\prime}  \right\}  \) be  simple function sequences  in Theorem \ref{thm:7.19} of \(  f ,g  \) , respectively .   Then by the theorem above and the Proposition \ref{pps:7-19-2}  \[
   \begin{aligned}
    \int_{X}\left( f+ g \right)\,\mathrm{d} \mu  = \lim_{n\to \infty}  \int_{X}\left( s_{n}+ s_{n}^{\prime}  \right)\,\mathrm{d} \mu &= \lim_{n\to \infty} \int_{X} s_{n}\,\mathrm{d} \mu + \lim_{n\to \infty} \int_{X}s_{n}^{\prime} \,\mathrm{d} \mu\\ 
     &= \int_{X}f\,\mathrm{d} \mu + \int_{g}\,\mathrm{d} \mu   
   \end{aligned}
    \]
\end{proof}
 

\begin{corollary}{}{}
    If $f_n: X \to [0, \infty]$ is \dfntxt{measurable}, for $n = 1, 2, 3, \dots$, and
\[
f(x) = \sum_{n=1}^{\infty} f_n(x) \quad (x \in X),
\]
then
\[
\int_X f \, d\mu = \sum_{n=1}^{\infty} \int_X f_n \, d\mu.
\]
\end{corollary}

\begin{proof}
    Let \(  g_{n}\left( x \right)= \sum _{k= 1}^{n}f_{k}\left( x \right)    \) .  Then \(  g_1\le g_2\le \cdots \le \infty  \) and  \(  f= \lim_{n\to \infty}g_{n}\left( x \right)   \)  . Applying Monotone Convergence Theorem to \(  \left\{ g_{n} \right\}   \) and \(  f  \) , we have  \[
    \int_{X}f\,\mathrm{d} \mu = \lim_{n\to \infty} \int_{X} \sum _{k = 1}^{n}f_{k}\left( x \right)\,\mathrm{d} \mu  =  \lim_{n\to \infty}\sum _{k= 1}^{n}\int_{X}f_{k}\left( x \right)\,\mathrm{d} \mu =  \sum _{n = 1}^{\infty} \int_{X}f_{n}\,\mathrm{d} \mu   
    \]
\end{proof}


\begin{corollary}{}{}
If $a_{ij} \ge 0$ for $i$ and $j = 1, 2, 3, \dots$, then
\[ \sum_{i=1}^{\infty} \sum_{j=1}^{\infty} a_{ij} = \sum_{j=1}^{\infty} \sum_{i=1}^{\infty} a_{ij}. \]
\end{corollary}
\begin{proof}
    Let \(  \mu   \) be counting measure on \(  \mathbb{Z}   \) .     Let \[
    f_{i}\left( j \right) = a_{ij}   ,\quad f\left( k \right)= \sum _{ i= 1}^{\infty}f_{i}\left( k \right)   =  \sum _{i = 1}^{\infty}a_{ik}
    \] By Monotone Convergence Theorem , for any measurable \(  g  \) ,   \[
    \int_{\mathbb{Z} }g\,\mathrm{d} \mu  =  \int_{\mathbb{Z} } \lim_{n\to \infty}  \chi  \left( \left\{ 1,\cdots ,n \right\} \right)g \,\mathrm{d} \mu =  \lim_{n\to \infty} \int_{\mathbb{Z} } \chi  \left( \left\{ 1,\cdots ,n \right\} \right)g\,\mathrm{d} \mu =   \sum _{ k= \infty}^{n}g\left( k \right)  
    \] Thus \[
    \int_{\mathbb{Z} } f\,\mathrm{d} \mu =  \sum _{ j= 1}^{\infty}\sum _{i= 1}^{\infty}a_{ij}
    \]   And we have  \[
    \sum _{ i= 1}^{\infty} \int_{X}f_{i}\,\mathrm{d} \mu =  \sum _{i = 1}^{\infty}  \sum _{j= 1}^{\infty}a_{ij}
    \] The above  corollary shows that \[
    \int_{\mathbb{Z} }f\,\mathrm{d} \mu =  \sum _{ i= 1}^{\infty}\int_{X}f_{i}\,\mathrm{d} \mu  
    \] That is \[
    \sum _{i= 1}^{\infty}\sum _{j= 1}^{\infty}a_{ij}= \sum _{j= 1}^{\infty}\sum _{i= 1}^{\infty}a_{ij}
    \]
\end{proof}


\begin{lemma}{Fatou's Lemma}{}
If $f_n: X \to [0, \infty]$ is \dfntxt{measurable}, for each positive integer $n$, then
\[
\int_X \left( \lim_{n \to \infty} \inf f_n \right) d\mu \le \lim_{n \to \infty} \inf \int_X f_n d\mu.
\tag{1}
\]

\end{lemma}

\begin{note}
   有了Monotone Convergence Theorem 之后, 关键的不等式就剩下 \[
    \int_{X}\inf _{n\ge k} f_{n}\,\mathrm{d} \mu \le  \inf _{n\ge k} \int_{X}f_{n}\,\mathrm{d} \mu  
    \] 它的含义是 局部最小的积分小于等于积分的最小. 
\end{note}
\begin{proof}
    Let \[
    g_{k}\left( x \right) =  \inf _{n\ge k} f_{n}\left( x \right) 
    \] Then \[
    \lim_{n\to \infty} \inf  f_{n}=  \lim_{n\to \infty}g_{n}=: g
    \]where  \(  \left\{ g_{k} \right\} , g  \)  are all measurable .   It is obvious that \(  \left\{ g_{k} \right\}  \) is increasing .  Thus \[
    \lim_{n\to \infty}  \int_{X}g_{n}\,\mathrm{d} \mu =  \int_{X}g\,\mathrm{d} \mu =  \int_{X}\left( \lim_{n\to \infty}\inf f_{n} \right)\,\mathrm{d} \mu  
    \]Only need to show that \[
    \int_{X}\inf _{n\ge k} f_{n}\,\mathrm{d} \mu \le  \inf _{n\ge k} \int_{X}f_{n}\,\mathrm{d} \mu  
    \] which is true since \[
    \int_{X}\inf _{n\ge k}f_{n}\,\mathrm{d} \mu \le  \int_{X}f_{n}\,\mathrm{d} \mu  ,\text{for all   }  n\ge k
    \]
\end{proof}

\begin{theorem}{}{}
Suppose $f: X \to [0, \infty]$ is measurable, and
\[
\varphi(E) = \int_E f \, d\mu \quad (E \in \mathfrak{M}).
\]
Then $\varphi$ is a measure on $\mathfrak{M}$, and
\[
\int_X g \, d\varphi = \int_X gf \, d\mu
\]
for every measurable $g$ on $X$ with range in $[0, \infty]$.
\end{theorem}
\begin{note}
    将 measurable function \(  f  \) 视为对measure \(  \mu   \)的某种 countable次算数加权操作.   那么得益于 \(  \mu   \)对countable操作的相容性,  \(   \varphi   \)也就成为一个测度.  \(   \varphi   \)的积分行为也验证了这种看法的合理性.    
\end{note}
\begin{proof}
    \begin{itemize}
        \item Let \(  E_1,\cdots ,E_{n},\cdots    \) be disjoint measurable sets such that \(  E= \bigcup _{n = 1}^{\infty}E_{n}  \).  Then \(  \left\{ \sum _{k= 1}^{n}\chi _{E_{k}}f \right\}  \) be increasing measurable function sequence such that  \(  \lim_{n\to \infty} \sum _{k= 1}^{n}\chi _{E_{k}}f= f  \).     The Monotone Convergence Theorem shows that \[
   \begin{aligned}
    \sum _{n = 1}^{\infty} \varphi \left( E_{k} \right)= \lim_{n\to \infty} \sum _{ k= 1}^{n} \int_{E_{k}}f\,\mathrm{d} \mu &=   \lim_{n\to \infty} \int_{E} \sum _{k = 1}^{n}\chi _{E_{k}}f\,\mathrm{d} \mu  \\ 
     &=  \int_{E}f\,\mathrm{d} \mu =  \varphi \left( E \right) 
   \end{aligned}
    \]Since \(   \varphi \left( \varnothing \right) = 0   \) .  \(   \varphi   \) is a measure on \(  X  \) . 
    
    \item To show the second results. See that it is true for \(  g = \chi _{E}  \) . Thus it is true for \(  g  \) is a simple function .   And the general case follows from the Monotone Convergence Theorem. 
    \end{itemize}
    
\end{proof}


\section{Integration of Complex Functions }

\begin{definition}{Lebegsgue Integrability}{}
We define $L^1(\mu)$ to be the collection of all complex measurable functions $f$ on $X$ for which
\[
\int_X |f| \, d\mu < \infty.
\]
Note that the measurability of $f$ implies that of $|f|$, hence the above integral is defined.

The members of $L^1(\mu)$ are called \dfntxt{Lebesgue integrable functions} (with respect to $\mu$) or \dfntxt{summable functions}. 
\end{definition}

\begin{definition}{Complex Integration}{}
If $f = u + iv$, where $u$ and $v$ are real measurable functions on $X$, and if $f \in L^1(\mu)$, we define
\[
\int_{E} f \,d\mu = \int_{E} u^{+} \,d\mu - \int_{E} u^{-} \,d\mu + i \int_{E} v^{+} \,d\mu - i \int_{E} v^{-} \,d\mu \tag{1}
\]
for every measurable set $E$.
\end{definition}
\begin{remark}
 We have $u^+ \le |u| < |f|$, etc., so that each of these four integrals is finite. Thus (1) defines the integral on the left as a complex number.
\end{remark}

\begin{definition}{Real Integration}{}
Occasionally it is desirable to define the integral of a measurable function $f$ with range in $[-\infty, \infty]$ to be
\[
\int_E f \,d\mu = \int_E f^+ \,d\mu - \int_E f^- \,d\mu, \tag{2}
\]
provided that at least one of the integrals on the right of (2) is finite. The left side of (2) is then a number in $[-\infty, \infty]$.
\end{definition}


\begin{theorem}{}{}
Suppose $f$ and $g \in L^1(\mu)$ and $\alpha$ and $\beta$ are complex numbers. Then $\alpha f + \beta g \in L^1(\mu)$, and
\[
\int_{X} (\alpha f + \beta g) \,d\mu = \alpha \int_{X} f \,d\mu + \beta \int_{X} g \,d\mu
\]
\end{theorem}
\begin{proofsketch}
    可积性来自于模长的三角不等式. 主要证明可加性 . 将 \(  f+ g = h  \)的 正负部分到两边, 就都变成了非负函数. 积分即可. 
\end{proofsketch}
\begin{theorem}{}{}
If $f \in L^1(\mu)$, then
\[
\left| \int_X f d\mu \right| \le \int_X |f| d\mu.
\]
\end{theorem}
\begin{note}
    被缩放的部分实际上是 \(  f  \)因改变了方向产生的损耗.  
\end{note}
\begin{proof}
    Let \(  z =  \int_{X} f\,\mathrm{d} \mu   \) , \(  z  \) is a complex number .  There exists a complex number \(  \alpha   \) (\(  \alpha = \frac{\bar{z} }{\left| z \right|  }   \) ) , such that \(  \left| \alpha  \right|= 1   \) , and   \[
    \left| \int_{X} f\,\mathrm{d} \mu  \right|=  \alpha z= \alpha \int_{X}f\,\mathrm{d} \mu  = \int_{X} \alpha f \,\mathrm{d} \mu 
    \]   Suppose \(  \alpha f= u+ iv  \) ,  since \(  \int_{X}\alpha f\,\mathrm{d} \mu   \) is a real number , \(  \int_{X}v\,\mathrm{d} \mu = 0  \) . We have \[
    \int_{X}\alpha f\,\mathrm{d} \mu = \int_{X}u\,\mathrm{d} \mu \le \int_{X}\left| \alpha f \right|\,\mathrm{d} \mu = \int_{X}\left| f \right|\,\mathrm{d} \mu   
    \]   
\end{proof}

\begin{theorem}{Lebesgue's Dominated Convergence Theorem}{}
Suppose $\{f_n\}$ is a sequence of complex measurable functions on $X$ such that
\[
f(x) = \lim_{n \to \infty} f_n(x)
\]
exists for every $x \in X$. If there is a function $g \in L^1(\mu)$ such that
\[
|f_n(x)| \le g(x) \quad (n = 1, 2, 3, \dots; x \in X),
\]
then $f \in L^1(\mu)$,
\[
\lim_{n \to \infty} \int_X |f_n - f| \, d\mu = 0,
\]
and
\[
\lim_{n \to \infty} \int_X f_n \, d\mu = \int_X f \, d\mu.
\]
\end{theorem}

\begin{proofsketch}
     \( \left\{ f_{n} \right\}  \)不一定单调,  能将极限从积分外 塞进积分里的工具 , 目前我们只有Fatou's Lemma , 但是它原始的样子跟我们需要的不等号是反向的. 自然的想法是填一个负号, 这又会导致 函数失去非负性, 而控制函数 \(  g  \)恰好能弥补这一点不足. 实现起来, 我们从 \(  g  \)上扣去 \(  f,f_{n}  \)    , 而非直接拿出 \(  f, f_{n}  \)去做.  
\end{proofsketch}
\begin{proof}
    Since \(  \left| f \right|\le g   \)  and \(  f   \) is measurable , \(  f \in  L^{1}\left( \mu  \right)    \) .  Function \(  \left( 2g- \left| f_{n}-f \right| \right)    \)   is a  nonnegative measurable function .  Applying Fatou's Lemma to it , we have \[
    \int_{X}\lim_{n\to \infty}\inf \left( 2g-\left| f_{n}-f \right|  \right)\,\mathrm{d} \mu \le \lim_{n\to \infty}\inf \int_{X}\left( 2g-\left| f_{n}-f \right|  \right)\,\mathrm{d} \mu   
    \]where \[
    \lim_{n\to \infty} \inf \left( 2g-\left| f_{n}-f \right|  \right) = 2g- \lim_{n\to \infty}\sup \left| f_{n}-f \right| = 2g
    \] \[
    \lim_{n\to \infty}\inf \int_{X}\left( 2g- \left| f_{n}-f \right|  \right)\,\mathrm{d} \mu =  2\int_{X}g\,\mathrm{d} \mu  - \lim_{n\to \infty}\sup \int_{X}\left| f_{n}-f \right|\,\mathrm{d} \mu   
    \] The original inequality comes to  \[
    2\int_{X}g\,\mathrm{d} \mu \le  2\int_{X}g\,\mathrm{d} \mu - \lim_{n\to \infty}\sup \int_{X}\left| f_{n}-f \right|\,\mathrm{d} \mu  
    \] Since \(  \int_{X}g\,\mathrm{d} \mu   \) is finite, we can substract \(  2\int_{X}g\,\mathrm{d} \mu   \) both sides and get \[
    \lim_{n\to \infty}\sup \int_{X}\left| f_{n}-f \right|\,\mathrm{d} \mu \le 0 
    \]  Thus \[
     \lim_{n\to \infty}\int_{X}\left| f_{n}-f \right|\,\mathrm{d} \mu = 0 
    \]Finnaly , we have\[
 \begin{aligned}
    \left| \lim_{n\to \infty} \int_{X}f_{n}\,\mathrm{d} \mu - \int_{X}f\,\mathrm{d} \mu \right|  &= \left| \lim_{n\to \infty} \int_{X} \left( f_{n}-f \right)\,\mathrm{d} \mu   \right|  \\ 
     &\le  \lim_{n\to \infty} \int_{X}\left| f_{n}-f \right|\,\mathrm{d} \mu \\ 
      &= 0 
 \end{aligned}
    \] which completes the proof. 
\end{proof}

\section{The Sets of Measure Zero}

\begin{definition}{}{}
Let $P$ be a property which a point $x$ may or may not have. If $\mu$ is a measure on a $\sigma$-algebra $\mathfrak{M}$ and if $E \in \mathfrak{M}$, the statement "$P$ holds \dfntxt{almost everywhere on $E$}" (abbreviated to "$P$ holds a.e. on $E$") means that there exists an $N \in \mathfrak{M}$ such that $\mu(N) = 0$, $N \subset E$, and $P$ holds at every point of $E - N$.
\end{definition}

\begin{remark}
     This concept of a.e. depends of course very strongly on the given measure, and we shall write “a.e. $[\mu]$” whenever clarity requires that the measure be indicated.
\end{remark}

\begin{example}{}{}
    For example, if $f$ and $g$ are measurable functions and if
\[
\mu(\{x: f(x) \neq g(x)\}) = 0,
\]
we say that $f = g$ a.e. $[\mu]$ on $X$, and we may write $f \sim g$. 
\end{example}
\begin{remark}
  This is easily seen to be an equivalence relation.  The transitivity ($f \sim g$ and $g \sim h$ implies $f \sim h$) is a consequence of the fact that the union of two sets of measure 0 has measure 0.
\end{remark}

\begin{proposition}{Completion of a Measure}{}
  If $f \sim g$, then, for every $E \in \mathfrak{M}$,
\[
\int_E f d\mu = \int_E g d\mu.
\]

\end{proposition}

\begin{theorem}{}{}
Let $(X, \mathfrak{M}, \mu)$ be a measure space, let $\mathfrak{M}^*$ be the collection of all $E \subset X$ for which there exist sets $A$ and $B \in \mathfrak{M}$ such that $A \subset E \subset B$ and $\mu(B - A) = 0$, and define $\mu(E) = \mu(A)$ in this situation. Then $\mathfrak{M}^*$ is a $\sigma$-algebra, and $\mu$ is a measure on $\mathfrak{M}^*$.
\end{theorem}

\begin{remark}
    
This extended measure $\mu$ is called \dfntxt{complete}, since all subsets of sets of measure 0 are now measurable; the $\sigma$-algebra $\mathfrak{M}^*$ is called the \dfntxt{$\mu$-completion} of $\mathfrak{M}$. The theorem says that every measure can be completed, so, whenever it is convenient, we may assume that any given measure is complete; this just gives us more measurable sets, hence more measurable functions. 

\end{remark}

\begin{proof}
   We begin by checking that \(  \mu   \) is well defined for every \(  E \in \mathfrak{M}^{*}  \) . If for \(  i= 1,2  \) ,   \(  A_{i}\subseteq E\subseteq B_{i}  \),   where \(  A_{i},B_{i}\in \mathfrak{M}, \mu \left( B_{i}-A_{i}\right)= 0 \) .     We need to show that \(  \mu \left( A_1 \right)= \mu \left( A_2 \right)    \). Since \(  A_1\subseteq E\subseteq B_2  \) ,  \[
   \mu \left( A_1 \right)\le \mu \left( B_2 \right)= \mu \left( A_2 \right)+ \mu \left( B_2-A_2 \right)= \mu \left( A_2 \right)     
   \] By symmetric , \[
   \mu \left( A_2 \right)\le \mu \left( A_2 \right)  
   \]Thus \(  \mu \left( A_1 \right)= \mu \left( A_2 \right)    \), \(  \mu \left( E \right)   \) is well defined. 
    Next , let us verify that \(  \mathfrak{M}^{*}  \) has the three defining properties of a \(   \sigma   \)-algebra. 
    \begin{enumerate}
        \item It is clearly that \(  X \in \mathfrak{M}^{*}  \). 
        \item If \(  A\subseteq E\subseteq B  \) , then \(  B^{c} \subseteq E^{c} \subseteq A^{c}   \) . And since \(  \mu \left( A^{c}-B^{c} \right)= \mu \left( A^{c}\cap B \right)= \mu \left( B-A \right)= 0     \) , \(  E^{c} \in \mathfrak{M}^{*}  \). 
        \item Suppose \(  A_{i}\subseteq E_{i}\subseteq B_{i} , i =  1,2,\cdots   \), where \(  A_{i},B_{i} \in \mathfrak{M}  \),    \(  \mu \left( B_{i}-A_{i} \right)= 0   \)       .  Let \(  E= \bigcup E_{i} ,A= \bigcup A_{i} ,B= \bigcup B_{i}  \) .  Since \[
        B-A= \bigcup _{i}\left( B_{i}-A \right)\subseteq \bigcup _{i} \left( B_{i}-A_{i} \right)  
        \], \[
        \mu \left( B-A \right)\le \sum _{i}\mu \left( B_{i}-A_{i} \right)= 0  
        \]Then by \(  A\subseteq E\subseteq B  \) , it follows that  \(  E \in \mathfrak{M}^{*}  \).   

        Finally, we show the countable additive of \(  \mu   \) .  If the above sets \(  E_{i}   \)   are disjoint, the smae is true of the sets \(  A_{i}  \) , then \[
        \mu \left( E \right)= \mu \left( A \right)= \sum _{i}\mu \left( A_{i} \right)= \sum _{i}\mu \left( E_{i} \right)    
        \] 
    \end{enumerate}
       
\end{proof}

\begin{definition}{More Mearuable Functions}{}
    We say a function \(  f  \) defined on a set \(  E \in \mathfrak{M}  \) is \dfntxt{measurable on \(  X  \) } .  If \(  \mu \left( E^{c} \right)= 0   \) and if \(  f^{-1} \left( V \right)\cap E   \)     is a measurable for every open set \(  V  \) . 
\end{definition}

\begin{remark}
    \begin{itemize}
        \item If we define \(  f\left( x \right)= 0   \) for \(  x \in E^{c}   \) , we obtain a mearuable function on \(  X  \) , in the old sense. 
        \item If our measure happens to be complete , we can define \(  f  \) on \(  E^{c}  \) in a perfrctly arbitrary manner, and we still get a measurable function . The integral of \(  f  \) over any set \(  A \in \mathfrak{M}  \) is independent of the definition of \(  f  \) on \(  E^{c}  \); therefore this definition need not even be specified at all.       
    \end{itemize}
       
\end{remark}
\begin{theorem}{}{}
Suppose $\{f_n\}$ is a sequence of complex measurable functions defined a.e. on X such that
\[
\sum_{n=1}^{\infty} \int_X |f_n| \, d\mu < \infty.
\tag{1}
\]
Then the series
\[
f(x) = \sum_{n=1}^{\infty} f_n(x)
\tag{2}
\]
converges for almost all x, $f \in L^1(\mu)$, and
\[
\int_X f \, d\mu = \sum_{n=1}^{\infty} \int_X f_n \, d\mu.
\tag{3}
\]
\end{theorem}
\begin{remark}
    \begin{itemize}
        \item 现在, 考察积分和求和是否可交换, 只需要检查模长积分的级数是否收敛. 
        \item 即便 \(  f_{n}  \)在 \(  X  \)上处处被定义, 级数收敛的地方也只是\dfntxt{almost everywhere} .   
    \end{itemize}
    
\end{remark}
\begin{note}
    此时,控制函数就是绝对级数本身, 它在一个满测集上将原级数控制住. 其特殊性间接给出了原级数的几乎处处收敛性质. 
\end{note}
\begin{proof}
   Let \(  S_{n}  \) be the set on which \(  f_{n}  \) is defined, so  that \(  \mu \left( S_{n}^{c} \right)= 0   \)   . Let \(  S= \bigcap S_{n}   \).  Put \(   \varphi \left( x \right)= \sum \left| f_{n}\left( x \right)  \right|     \) , for \(  x \in S  \) .By Monotone Convergence Theorem \[
   \int_{S} \varphi \,\mathrm{d} \mu \le \sum _{n = 1}^{\infty}\int_{S}\left| f_{n} \right|\,\mathrm{d} \mu < \infty 
   \]   Let \(  E= \left\{ x\in S:  \varphi \left( x \right)< \infty  \right\}  \) . Then it follows that   \(  \mu \left( E^{c} \right)= 0   \) .    We have \[
   \sum _{n = 1}^{\infty}\left| f_{n}\left( x \right)  \right|< \infty ,\quad \forall x \in E
   \] Thus \[
   f\left( x \right)= \sum _{n = 1}^{\infty}f_{n}\left( x \right)  
   \] converges on  \(  E  \) , which is a set with its complement measure zero . That is to say \(  f  \) coverges for almost all \(  x  \) . And since \(  f \in L^{1}\left( \mu |_{E} \right)   \) , \(  \mu \left( E \right)= 0   \) , we have \(  f \in L^{1}\left( \mu  \right)   \) .      Finally , by Monotone Convergence Theorem  ,  \[
   \int_{X}f\,\mathrm{d} \mu = \int_{E} f\,\mathrm{d} \mu = \sum _{n = 1}^{\infty}\int_{E}f_{n}\,\mathrm{d} \mu =   \sum _{n = 1}^{\infty}\int_{X}f_{n}\,\mathrm{d} \mu  
   \]
\end{proof}

\begin{theorem}{}{}
    \begin{enumerate}
        \item Suppose \(  f: X\to \left[ 0,\infty \right]   \) is measurable , \(  E \in \mathfrak{M}  \) , and \(  \int_{E}f\,\mathrm{d} \mu   = 0\) . Then \(  f= 0  \) a.e. on \(  E  \).

\end{theorem}
\begin{proof}
     Let \[
     g\left( x \right)=  \sum _{k = 1}^{\infty}\chi _{E_{k}}\left( x \right)  
     \] Then \(  g   \) is measurable .  By Monotone Convergence Theorem ,  \[

\[
\int_E \infty\cdot f \, d\mu = \infty\cdot  \int_E f \, d\mu.
\]
\end{problem}
\begin{proof}
    Let \(  E_1=  f^{-1} \left( 0 \right)\cap E   \) , \(  E_2=  E\setminus E_1  \).   
    \begin{enumerate}
        \item If  \(  \mu \left( E_2 \right)> 0   \) Then \[
        \int_{E} \infty\cdot f\,\mathrm{d} \mu \ge  \int_{E_2}\infty\cdot f\,\mathrm{d} \mu = \infty
        \]   \[
        \int_{E}f\,\mathrm{d} \mu \ge \int_{E_2}f\,\mathrm{d} \mu > 0,\quad  \infty\cdot \int_{E}f\,\mathrm{d} \mu = \infty
        \]
        \item If \(  \mu \left( E_2 \right)= 0  \) , then \[
        \int_{E}\infty\cdot f\,\mathrm{d} \mu =  \int_{E_1}\infty\cdot 0\,\mathrm{d} \mu + \int_{E_2}\infty\cdot f\,\mathrm{d} \mu = 0
        \]  \[
        \int_{E}f\,\mathrm{d} \mu =  \int_{E_1}0\,\mathrm{d} \mu + \int_{E_2}f\,\mathrm{d} \mu = 0,\quad \infty\cdot \int_{E}f\,\mathrm{d} \mu = 0
        \]
    \end{enumerate}
      
\end{proof}
